 if(act===1){g=R(2,5);e=R(2,5);txt=`Vex loaded <span class="g">${g} ${c[0]}</span>. Each one holds <span class="e">${e} ${c[1]}</span>.`;}
 else if(act===2){g=R(3,9);e=R(3,9);const d=R(21,58);
   const k=R(0,2);
   if(k===0)txt=`Vex, who is ${d} years old, packed ${g} ${c[0]} with ${e} ${c[1]} each.`;
   else if(k===1)txt=`There are ${e} ${c[1]} in every ${c[0].slice(0,-1)}, and Vex grabbed ${g} ${c[0]}.`;
   else txt=`Dock ${d}: Vex filled ${g} ${c[0]}. Each ${c[0].slice(0,-1)} holds ${e} ${c[1]}.`;
 } else {
   if(Math.random()<.5){const t=R(2,4),cr=R(2,4),it=R(3,6);g=t*cr;e=it;two=true;
     txt=`Each truck carries ${cr} ${c[0]}. Each ${c[0].slice(0,-1)} holds ${it} ${c[1]}. There are ${t} trucks — crack the ${c[1]} total.`;}
   else{g=R(4,9);e=R(4,9);txt=`There are ${e} ${c[1]} in every ${c[0].slice(0,-1)}, and Vex grabbed ${g} ${c[0]}.`;}
 }
 return{type:'crack',g:g,e:e,icon:c[2],txt:txt,two:two,unit:c[1]};
}
function makeBluff(act){
 const c=CTX[R(0,CTX.length-1)];const g=R(3,act===3?9:8),e=R(3,9);const p=g*e;
 let lie=Math.random()<.55, claim=p;
 if(lie){claim=p+(Math.random()<.5?-1:1)*R(1,Math.max(2,Math.round(p*.2)));if(claim<1)claim=p+3;}
 return{type:'bluff',g:g,e:e,icon:c[2],claim:claim,lie:lie,
  txt:`Vex broadcasts about ${g} ${c[0]}, each holding ${e} ${c[1]}.`,
  say:`"Too slow! That load is mine —"`,unit:c[1]};
}

let S;
function newRun(){
 S={i:0,cr:0,ea:0,seized:0,busted:0,vex:0,stamps:[],q:null,phase:'run',times:[]};
 const list=[];
 for(let k=0;k<4;k++)list.push(makeCrack(1));
 for(let k=0;k<3;k++){list.push(makeCrack(2));if(k<2)list.push(makeBluff(2));}
 list.push(makeCrack(2));
 for(let k=0;k<3;k++)list.push(makeCrack(3));
 for(let k=0;k<3;k++)list.push(makeBluff(3));
 S.list=list;
 $('stamps').innerHTML='';renderScore();load();
}
function actOf(i){return i<4?1:(i<11?2:3);}
function load(){
 if(S.i>=S.list.length){return end();}
 const q=S.list[S.i];S.q=q;S.cr=0;S.ea=0;S.used=false;S.start=Date.now();
 const a=actOf(S.i);
 $('actlbl').textContent=a===1?'ACT 1 — HARBOR DOCKS':a===2?'ACT 2 — WAREHOUSE ROW':'ACT 3 — THE NIGHT CONVOY';
 $('msg').innerHTML=q.txt;
 $('vexbox').classList.toggle('hide',q.type!=='bluff');
 $('vexbox').classList.remove('crack');
 $('dockPanel').classList.toggle('hide',q.type==='bluff');
 if(q.type==='bluff'){$('vexsay').textContent=q.say;$('vexclaim').textContent=`${q.g} × ${q.e} = ${q.claim}`;}
 $('lock').classList.toggle('hide',q.type==='bluff');
 $('confirmB').classList.toggle('hide',q.type!=='bluff');
 $('bustB').classList.toggle('hide',q.type!=='bluff');
 $('ans').value='';$('ans').disabled=q.type==='bluff';
 $('hint').textContent=a===1?'Turn the CRATES dial to match the message.':(q.type==='bluff'?'Build the true array first — then judge his claim.':'Read carefully. Which number is the groups?');
 // vex position for this shipment
 S.vex=0;placeVex();
 draw();buildTally();
 if(a===1&&q.type==='crack'){S.cr=1;draw();$('dCr').classList.add('hot');} else $('dCr').classList.remove('hot');
 startVexTimer();
}
function buildTally(){

 if(actOf(S.i)===3&&q.type==='crack'){$('tally').innerHTML='<span style="color:#7f9;font-size:15px">Tally strip encrypted — trust your reasoning.</span>';return;}
 for(let k=0;k<n;k++)h+='<div class="tk"></div>';
 $('tally').innerHTML=h;
}

}

}

function toast(t,c){const e=$('toast');e.textContent=t;e.style.color=c;e.style.opacity=1;e.style.transition='none';setTimeout(()=>{e.style.transition='opacity .8s';e.style.opacity=0;},700);}

function verdict(said){
 const q=S.q;if(q.type!=='bluff')return;
 const dialsOK=(S.cr===q.g&&S.ea===q.e);
 const truthIsLie=q.lie;
 if(said===!truthIsLie){ // confirmed a true claim OR busted a lie -> said===true means CONFIRM
 }
 const right=(said==='confirm'&&!truthIsLie)||(said==='bust'&&truthIsLie);
 if(right){
  clearInterval(vt);
  if(said==='bust'){$('vexbox').classList.add('crack');beep(1200,.08,'square',.2);setTimeout(()=>beep(300,.4,'sawtooth',.16),90);
    S.busted++;S.vex=Math.max(0,S.vex-4);placeVex();toast('BLUFF BUSTED! Vex loses ground.','#ff9ce8');
    $('hint').textContent=`${q.g} rows of ${q.e} is ${q.g*q.e}, not ${q.claim}. Your array proves it.`+(dialsOK?'':' (Set the dials next time to show your proof!)');
    stamp(`BUSTED ${q.g}×${q.e}≠${q.claim}`);}
  else{beep(659,.12);setTimeout(()=>beep(880,.25),110);toast('CONFIRMED — and seized.','#7fffd4');
    S.seized++;stamp(`${q.g} × ${q.e} = ${q.g*q.e}`);$('hint').textContent=`He told the truth this once. ${q.g} × ${q.e} = ${q.g*q.e}.`;}
  renderScore();next();
 } else {
  S.vex+=2;placeVex();beep(200,.3,'sawtooth',.14);toast('WRONG CALL — Vex surges ahead','#ffb84a');
  $('hint').textContent=`Build ${q.g} rows of ${q.e} on your dials and count. Truth: ${q.g} × ${q.e} = ${q.g*q.e}.`;
  $('dockPanel').classList.remove('hide');$('vexbox').classList.add('hide');
 }
}
