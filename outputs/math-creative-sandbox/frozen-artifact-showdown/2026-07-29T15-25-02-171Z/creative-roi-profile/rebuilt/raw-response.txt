let AC=null;
function ac(){if(!AC)AC=new (window.AudioContext||window.webkitAudioContext)();return AC}
function tone(f,t,d,type,g){const a=ac(),o=a.createOscillator(),v=a.createGain();o.type=type||'sine';o.frequency.value=f;v.gain.value=0;o.connect(v);v.connect(a.destination);const s=a.currentTime+t;v.gain.setValueAtTime(0,s);v.gain.linearRampToValueAtTime(g||.14,s+.012);v.gain.exponentialRampToValueAtTime(.0008,s+d);o.start(s);o.stop(s+d+.05)}
function noise(t,d,g){const a=ac(),b=a.createBuffer(1,a.sampleRate*d,a.sampleRate),ch=b.getChannelData(0);for(let i=0;i<ch.length;i++)ch[i]=(Math.random()*2-1)*(1-i/ch.length);const s=a.createBufferSource();s.buffer=b;const f=a.createBiquadFilter();f.type='highpass';f.frequency.value=1400;const v=a.createGain();v.gain.value=g||.12;s.connect(f);f.connect(v);v.connect(a.destination);s.start(a.currentTime+t)}

/* ===== PROBLEM GENERATION ===== */
function makeExchange(rIdx,idx){
  const R=ROUNDS[rIdx];const fl=FLAVORS[(idx+season)%FLAVORS.length];const unit=fl[0],grp=fl[1];
  let type='build';
  if(rIdx===1&&idx===1)type='read';
  if(rIdx===2)type=(idx%2===1)?'boast':'build';
  if(rIdx===3){type=(idx===0)?'missing':(idx===1?'boast':(idx===2?'two':(idx===3?'missing':'build')))}
  let[g,e]=pair(R.lo,R.hi);
  const ex={type,g,e,p:g*e,unit,grp,round:R};
  if(type==='build'){
    let s;
    if(R.hl)s='I send <mark>'+g+' '+grp+'s</mark>, with '+e+' '+unit+'s in each '+grp+'. Match my force or fall.';
    else if(rIdx===1)s='In each '+grp+' march '+e+' '+unit+'s — and I command '+g+' '+grp+'s. I am '+ri(19,44)+' years unbeaten in arena '+ri(3,9)+'. Match me.';
    else s='My force: '+g+' '+grp+'s, each holding '+e+' '+unit+'s. Build its equal.';
    ex.scroll=s;
  } else if(type==='read'){
    ex.scroll='My formation already stands on the floor. <b>Read it</b> — how many '+grp+'s, how many in each, and the total?';
  } else if(type==='boast'){
    const honest=Math.random()<.45;
    let claim=g*e;
    if(!honest){const off=Math.max(1,Math.round(g*e*(ri(6,20)/100)));claim=Math.random()<.5?g*e-off:g*e+off;if(claim<1)claim=g*e+off}
    ex.claim=claim;ex.honest=(claim===g*e);
    ex.scroll='I boast: <b>'+g+' '+grp+'s of '+e+' '+unit+'s makes '+claim+'!</b> Build the truth, then rule.';
  } else if(type==='missing'){
    ex.given='cols';ex.scroll='My total is <b>'+ex.p+'</b> '+unit+'s in <b>'+g+' '+grp+'s</b>. How large is each '+grp+'? Build it.';
  } else if(type==='two'){
    const t=ri(2,3),c=ri(2,3),m=ri(3,5);ex.t=t;ex.c=c;ex.m=m;ex.g=t*c;ex.e=m;ex.p=t*c*m;
    ex.scroll='I field <b>'+t+' platoons</b>, each of <b>'+c+' '+grp+'s</b>, each '+grp+' holding <b>'+m+' '+unit+'s</b>. How many '+grp+'s in all — and the total force? Build every '+grp+' as a row.';
  }
  return ex;
}

    if(rows*cols!==ex.p){$('coach').textContent='Your '+rows+' rows of '+cols+' make '+(rows*cols)+'. The total must be '+ex.p+'. Adjust PER ROW.';buzz();attempts++;flick();if(attempts>=3)loseExchange();return}
    winExchange();return;
  }
  if(ex.type==='two'){
    if(isNaN(typed)){$('coach').textContent='Type the total force.';return}
    if(rows!==ex.t*ex.c||cols!==ex.m){$('coach').textContent=(attempts?'Each of the '+ex.t+' platoons holds '+ex.c+' squads → '+ex.t+' × '+ex.c+' rows, each of '+ex.m+'.':'How many squads in ALL? Platoons times squads-per-platoon. Then each row holds the members.');buzz();attempts++;flick();if(attempts>=2)loseExchange();return}
    if(typed!==ex.p){wrongProduct(typed);return}
    winExchange();return;
  }
  // build
  if(isNaN(typed)){$('coach').textContent='Type your product to arm the formation.';return}
  if(rows!==ex.g||cols!==ex.e){
    $('coach').textContent='The '+ex.grp+'s are the ROWS ('+(attempts?ex.g:'how many groups?')+'), the '+ex.unit+'s in each are PER ROW.';
    buzz();flick();attempts++;if(attempts>=3)loseExchange();return;
  }
  if(typed!==ex.p){wrongProduct(typed);return}
  winExchange();
};

function rule(confirmed){
  const typed=parseInt($('prodIn').value,10);
  if(typed!==ex.p){$('coach').textContent='Lock your proof product first.';return}
  const right=(confirmed===ex.honest);
  $('confirmBtn').disabled=true;$('bustBtn').disabled=true;
  if(right&&!ex.honest){
    $('coach').textContent='BUSTED! '+ex.g+' × '+ex.e+' = '+ex.p+', not '+ex.claim+'.';
    rowlines().forEach((l,i)=>setTimeout(()=>{l.querySelectorAll('.tile').forEach(t=>t.classList.add('shatter'));noise(0,.3,.16);tone(900-i*40,0,.25,'triangle',.09)},i*220));
    setTimeout(()=>winExchange(true),rowlines().length*220+400);
  } else if(right){
    $('coach').textContent='Confirmed — the boast was honest. '+ex.g+' × '+ex.e+' = '+ex.p+'.';
    setTimeout(()=>winExchange(),400);
  } else {
    $('coach').textContent='Wrong ruling. Truth: '+ex.g+' × '+ex.e+' = '+ex.p+' (claim was '+ex.claim+').';

function winExchange(skipPulse){
  $('lockBtn').disabled=true;
  hall.push({g:rows,c:cols,p:rows*cols,label:rows+' × '+cols+' = '+(rows*cols),hard:(ex.type==='two'||ex.type==='missing'),season:season});
  const finish=()=>{
    gong();state.win++;pips();
    if(state.roundIdx===3){state.cracks++;}
    if(state.win>=state.need)setTimeout(matchWon,900);
    else{state.exIdx++;$('coach').textContent='Exchange won! '+rows+' × '+cols+' = '+rows*cols+'.';setTimeout(nextExchange,1200)}
  };
  if(skipPulse){finish()}else{firePulses(rows*cols,finish)}
}

}

  }
  if(state.roundIdx===3){payoff();return}

}

}
