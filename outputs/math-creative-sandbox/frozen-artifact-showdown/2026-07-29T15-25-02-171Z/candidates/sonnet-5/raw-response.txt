function rnd(a,b){ return Math.floor(Math.random()*(b-a+1))+a; }

function pickBounds(){
  if(act===1){ maxCrates=5; maxPer=5; }
  else if(act===2){ maxCrates=9; maxPer=9; }
  else { maxCrates=9; maxPer=9; }

function buildMessage(){
  pickBounds();
  const ctx = contexts[rnd(0,contexts.length-1)];
  let g = rnd(2, maxCrates);
  let p = rnd(2, maxPer);
  currentGroupsVal = g; currentPerVal = p;
  currentAnswer = g*p;
  colorCode = (act===1);
  distractor = (act>=2 && Math.random()<0.5);
  let distractNum = rnd(1,99);

  let groupsFirst = act===1 ? true : Math.random()<0.6;
  let sentence;
  let gSpan = colorCode ? `<span class="grp">${g} ${ctx.noun}</span>` : `${g} ${ctx.noun}`;
  let pSpan = colorCode ? `<span class="pg">${p} ${ctx.item} each</span>` : `${p} ${ctx.item} each`;

  if(groupsFirst){
    sentence = `Vex loaded ${gSpan}. Each ${ctx.noun.slice(0,-1)} holds ${pSpan.replace(' each','')}.`;
  } else {
    sentence = `There are ${pSpan.replace(' each',' in every '+ctx.noun.slice(0,-1))}, and Vex grabbed ${gSpan}.`;
  }
  if(distractor){
    sentence = `Vex, who is ${distractNum} years old, packed ${gSpan} with ${pSpan}.`;
  }
  return {sentence, g, p, ctx};
}

function doVexBroadcast(){
  const m = buildMessage();
  curCtx = m.ctx;
  let claimedG = m.g, claimedP = m.p, claimedAns;
  let bluffType = rnd(0,2);
  let trueAns = m.g*m.p;
  if(bluffType===0){
    claimedAns = trueAns; // true claim
  } else if(bluffType===1){
    claimedAns = trueAns + rnd(1,5); // wrong product
  } else {
    // swapped interpretation, same product coincidentally sometimes, else wrong
    claimedAns = m.p*m.g + (Math.random()<0.5?0:rnd(1,4));
  }
  currentGroupsVal=m.g; currentPerVal=m.p; currentAnswer=trueAns;
  window._vexIsTrue = (claimedAns===trueAns);
  vexClaim.textContent = `"Vex claims: ${claimedG} ${m.ctx.noun} of ${claimedP} ${m.ctx.item} each = ${claimedAns}!"`;
  vexBroadcast.classList.remove('cracked');
  vexBroadcast.style.display='flex';
  note.innerHTML = `Intercepted: ${m.sentence}<br><b>Build the true array to check Vex's math.</b>`;
  lockBtn.style.display='none';
  verdictBtns.style.display='none';
}

let audioCtx;
function playTone(freq){
  try{
    audioCtx = audioCtx || new (window.AudioContext||window.webkitAudioContext)();
    const o = audioCtx.createOscillator();
    const g = audioCtx.createGain();
    o.frequency.value=freq; o.type='triangle';
    g.gain.setValueAtTime(0.08, audioCtx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.2);
    o.connect(g); g.connect(audioCtx.destination);
    o.start(); o.stop(audioCtx.currentTime+0.2);
  }catch(e){}
}
function playSting(type){
  try{
    audioCtx = audioCtx || new (window.AudioContext||window.webkitAudioContext)();

      [440,554,659,880].forEach((f,i)=>{
        setTimeout(()=>playTone(f), i*70);
      });
    } else if(type==='fail'){
      playTone(140); setTimeout(()=>playTone(100),100);
    } else if(type==='bust'){
      playTone(900); setTimeout(()=>playTone(300),80);
    }
  }catch(e){}
}

function resolveBluff(userSaysTrue){
  const built = (crates===currentGroupsVal && per===currentPerVal);
  const actualTrue = window._vexIsTrue;
