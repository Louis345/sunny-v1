<script>
const svgNS = "http://www.w3.org/2000/svg";
const cx = 210, cy = 210, R = 190;
let state = "await_judge";
let sliceEls = [];
let equalness = 0;

const goodAngles = [60,60,60,60,60,60];
let currentAngles = [];

function hexPoints(radius) {
  let pts = [];
  for (let i=0; i<6; i++) {
    const ang = (i*60 - 90) * Math.PI/180;
    pts.push([cx + radius*Math.cos(ang), cy + radius*Math.sin(ang)]);
  }
  return pts.map(p => p.join(",")).join(" ");
}

function setupHexClip() {
  const pts = hexPoints(R);
  document.getElementById('hexClipPoly').setAttribute('points', pts);
  document.getElementById('hexOutline').setAttribute('points', pts);
}
setupHexClip();

function polarPoint(angleDeg, radius) {
  const rad = (angleDeg - 90) * Math.PI / 180;
  return [cx + radius * Math.cos(rad), cy + radius * Math.sin(rad)];
}

function drawHex(angles, equalize=false) {
  const group = document.getElementById('hexGroup');
  group.innerHTML = "";
  sliceEls = [];
  let start = 0;
  angles.forEach((a) => {
    const [x1,y1] = polarPoint(start, R*1.5);
    const [x2,y2] = polarPoint(start+a, R*1.5);
    const path = document.createElementNS(svgNS, "path");
    const d = `M${cx},${cy} L${x1},${y1} A${R*1.5},${R*1.5} 0 ${a>180?1:0} 1 ${x2},${y2} Z`;
    path.setAttribute("d", d);
    path.classList.add("slice");
    if (!equalize) path.classList.add("unequal");
    else path.classList.add("fixed");
    group.appendChild(path);
    sliceEls.push(path);
    start += a;
  });
}

function randomBadAngles() {
  // generate 6 angles summing to 360, with clear inequality, randomized each round
  const bigCount = 2;
  const bigAngle = 90 + Math.random()*20; // 90-110
  const remaining = 360 - bigAngle*bigCount;
  const smallBase = remaining/4;
  let angles = [bigAngle, smallBase, smallBase, bigAngle, smallBase, smallBase];
  // shuffle order slightly for variety
  const shift = Math.floor(Math.random()*6);
  angles = angles.slice(shift).concat(angles.slice(0, shift));
  return angles;
}

function resetRound() {
  state = "await_judge";
  currentAngles = randomBadAngles();
  drawHex(currentAngles, false);
  equalness = 0;
  updateMeter();
  document.getElementById('meterWrap').classList.remove('visible');
  document.getElementById('meterFill').classList.remove('locked');
  document.getElementById('fairBtn').disabled = false;
  document.getElementById('unfairBtn').disabled = false;
  document.getElementById('nextBtn').style.display = "none";
  document.getElementById('customer').classList.remove('show');
  document.getElementById('customer').textContent = "😟";
  setFeedback("Inspect the tray. Are all 6 pieces the same size?", "#fff4de");
  document.getElementById('gm-note').textContent = "A tray just came in... is it fair?";
  setGmMouth(false);
  const stamp = document.getElementById('stampMarkEl');
  if (stamp) stamp.remove();
}

function setGmMouth(happy) {
  const m = document.getElementById('gm-mouth');
  if (happy) {
    m.style.borderRadius = "0 0 20px 20px";
    m.style.borderColor = "#2a6b2a";
    m.style.bottom = "30px";
  } else {
    m.style.borderRadius = "20px 20px 0 0";
    m.style.borderColor = "#2a1a0e";
    m.style.bottom = "38px";
  }
}

function setFeedback(text, color) {
  const f = document.getElementById('feedbackText');
  f.textContent = text;
  f.style.color = color || "#fff4de";
}

function judge(choice) {
  if (state !== "await_judge") return;
  document.getElementById('fairBtn').disabled = true;
  document.getElementById('unfairBtn').disabled = true;

  if (choice === "unfair") {
    setFeedback("Correct! That split is UNFAIR — pieces are not the same size.", "#c8ffcf");
    document.getElementById('gm-note').textContent = "Good eye! Now fix it, Inspector.";
    setGmMouth(true);
    sliceEls.forEach(s => s.classList.add('flash-correct'));
    document.getElementById('hexWrap').classList.add('flash-good');
    setTimeout(()=> document.getElementById('hexWrap').classList.remove('flash-good'), 400);
    document.getElementById('customer').classList.add('show');

    document.getElementById('meterWrap').classList.add('visible');

    setTimeout(() => {
      sliceEls.forEach(s => s.classList.remove('flash-correct'));
      animateFix();
    }, 900);
  } else {
    setFeedback("Hmm, look again — are all 6 pieces really the same size?", "#ffd9c2");
    sliceEls.forEach(s => s.classList.add('pulse-wrong'));
    document.getElementById('hexWrap').classList.add('wobble-bad');
    setTimeout(() => {
      document.getElementById('hexWrap').classList.remove('wobble-bad');
      sliceEls.forEach(s => s.classList.remove('pulse-wrong'));
      document.getElementById('fairBtn').disabled = false;
      document.getElementById('unfairBtn').disabled = false;
    }, 600);
  }
}

function animateFix() {
  state = "fixing";
  setFeedback("Watch the helper hand snap every slice to equal...", "#ffe9c7");
  let steps = 24;
  let frame = 0;
  const startAngles = currentAngles.slice();
  const target = goodAngles;

  function step() {
    frame++;
    let t = frame / steps;
    let eased = t < 1 ? 1 - Math.pow(1-t, 3) : 1;
    let interp = startAngles.map((a,i) => a + (target[i]-a)*eased);
    drawHex(interp, false);
    equalness = Math.min(100, Math.round(t*100));
    updateMeter();

    if (frame < steps) {
      requestAnimationFrame(step);
    } else {
      drawHex(target, true);
      equalness = 100;
      updateMeter(true);
      onFixedComplete();
    }
  }
  requestAnimationFrame(step);
}

function updateMeter(locked) {
  const fill = document.getElementById('meterFill');
  fill.style.width = equalness + "%";
  if (locked) fill.classList.add('locked');
}

let certIndex = 0;
function onFixedComplete() {
  state = "done";
  setFeedback("Certified! 6 equal pieces — perfectly fair. 🎉", "#c8ffcf");
  document.getElementById('gm-note').textContent = "Certified Fair-Split! Well done, Inspector.";
  document.getElementById('customer').textContent = "😃";

  const hexWrap = document.getElementById('hexWrap');
  const stamp = document.createElement('div');
  stamp.id = "stampMarkEl";
  stamp.textContent = "✅ FAIR";
  stamp.style.position = "absolute";
  stamp.style.top = "170px";
  stamp.style.left = "110px";
  stamp.style.fontSize = "42px";
  stamp.style.fontWeight = "bold";
  stamp.style.color = "#ffd23a";
  stamp.style.textShadow = "2px 2px 3px #000";
  stamp.style.transform = "scale(2) rotate(-15deg)";
  stamp.style.opacity = "0";
  stamp.style.transition = "all 0.35s cubic-bezier(.4,1.6,.6,1)";
  stamp.style.pointerEvents = "none";
  hexWrap.appendChild(stamp);
  requestAnimationFrame(() => {
    stamp.style.opacity = "1";
    stamp.style.transform = "scale(1) rotate(-15deg)";
  });

  if (certIndex < 3) {
    document.getElementById('slot' + certIndex).classList.add('filled');
    certIndex++;
  }

  setTimeout(() => {
    document.getElementById('subtitle').textContent =
      "You certified this split. Tap Next Tray for another one, Inspector.";
    document.getElementById('nextBtn').style.display = "inline-block";
  }, 700);
}

function nextTray() {
  resetRound();
}

function fitStage() {
  const scale = Math.min(window.innerWidth / 1365, window.innerHeight / 768);
  document.getElementById('scene').style.transform = `scale(${scale})`;
}
window.addEventListener('resize', fitStage);
fitStage();

resetRound();
</script>
</body>
