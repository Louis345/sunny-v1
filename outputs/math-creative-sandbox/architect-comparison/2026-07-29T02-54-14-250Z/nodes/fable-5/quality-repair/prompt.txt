const foxBar = document.getElementById('fox-bar');
const badgerBar = document.getElementById('badger-bar');

let draggedFrom = null;
let roundLocked = false;

// Build pieces
function buildBar(bar, count, isFox) {
  bar.innerHTML = '';
  const widthPercent = 100 / count;
  for (let i = 0; i < count; i++) {
    const p = document.createElement('div');
    p.className = 'piece';
    p.style.width = widthPercent + '%';
    p.dataset.index = i;
    p.dataset.owner = isFox ? 'fox' : 'badger';
    p.dataset.denom = count;
    bar.appendChild(p);
  }
}
buildBar(foxBar, 3, true);
buildBar(badgerBar, 4, false);

// snap animation on load
gsap.from('#fox-bar .piece', {scale: 0.3, opacity:0, stagger:0.08, duration:0.4, ease:'back.out(2)'});
gsap.from('#badger-bar .piece', {scale: 0.3, opacity:0, stagger:0.08, duration:0.4, ease:'back.out(2)', delay:0.3});

// Mark one pickable piece on each bar (first piece)
function markPickable() {
  document.querySelectorAll('.piece').forEach(p => p.classList.remove('pickable'));
  foxBar.children[0].classList.add('pickable');
  badgerBar.children[0].classList.add('pickable');
}
markPickable();

function playGhostDemo() {
  const foxPiece = foxBar.children[0];
  const rect = foxPiece.getBoundingClientRect();
  gsap.set(ghostHand, {x: rect.left - 260, y: rect.top - 280, opacity:1});

      gsap.to(ghostHand, {opacity:0, duration:0.4, delay:0.3});
    }
  });
}

// Drag setup (mouse + touch, simplified custom drag)
let dragEl = null;
let offsetX=0, offsetY=0;

function startDrag(e, piece) {
  if (roundLocked) return;
  if (trayPieces.find(tp => tp.owner === piece.dataset.owner)) return; // already placed
  e.preventDefault();
  dragEl = piece.cloneNode(true);
  dragEl.classList.add('dragging');
  dragEl.style.position = 'fixed';
  dragEl.style.width = piece.offsetWidth + 'px';
  dragEl.style.height = piece.offsetHeight + 'px';
  document.body.appendChild(dragEl);
  const rect = piece.getBoundingClientRect();
  offsetX = (e.touches ? e.touches[0].clientX : e.clientX) - rect.left;
  offsetY = (e.touches ? e.touches[0].clientY : e.clientY) - rect.top;
  dragEl.style.left = rect.left + 'px';
  dragEl.style.top = rect.top + 'px';
  dragEl.dataset.owner = piece.dataset.owner;
  dragEl.dataset.denom = piece.dataset.denom;

  function move(ev) {
    const cx = ev.touches ? ev.touches[0].clientX : ev.clientX;
    const cy = ev.touches ? ev.touches[0].clientY : ev.clientY;
    dragEl.style.left = (cx - offsetX) + 'px';
    dragEl.style.top = (cy - offsetY) + 'px';
  }
  function up(ev) {
    document.removeEventListener('mousemove', move);
    document.removeEventListener('touchmove', move);
    document.removeEventListener('mouseup', up);
    document.removeEventListener('touchend', up);
    handleDrop(dragEl);
  }
  document.addEventListener('mousemove', move);
  document.addEventListener('touchmove', move, {passive:false});
  document.addEventListener('mouseup', up);
  document.addEventListener('touchend', up);
}

foxBar.addEventListener('mousedown', e => { if(e.target.classList.contains('pickable')) startDrag(e, e.target); });
foxBar.addEventListener('touchstart', e => { if(e.target.classList.contains('pickable')) startDrag(e, e.target); }, {passive:false});
badgerBar.addEventListener('mousedown', e => { if(e.target.classList.contains('pickable')) startDrag(e, e.target); });
badgerBar.addEventListener('touchstart', e => { if(e.target.classList.contains('pickable')) startDrag(e, e.target); }, {passive:false});

function handleDrop(el) {
  const trayRect = tray.getBoundingClientRect();
  const elRect = el.getBoundingClientRect();
  const overlap = !(elRect.right < trayRect.left || elRect.left > trayRect.right ||
                     elRect.bottom < trayRect.top || elRect.top > trayRect.bottom);
  if (overlap) {
    placeOnTray(el.dataset.owner, parseInt(el.dataset.denom));
  }
  el.remove();
}

function placeOnTray(owner, denom) {
  if (trayPieces.find(tp => tp.owner === owner)) return;
  // full-length reference: fraction of tray width (scaled up for visibility)
  const trayWidth = 360;
  const fullPieceWidth = trayWidth / denom * 1.4; // scale up relative to actual 1/denom of a bar of length trayWidth*denom/... simplified proportional
  const pieceWidth = (trayWidth) / denom + 40; // visually proportional to 1/denom
  const tp = document.createElement('div');
  tp.className = 'tray-piece';
  const idx = trayPieces.length;
  tp.style.width = (300/denom) + 'px';
  tp.style.left = (10 + idx*10) + 'px';
  tp.style.top = idx === 0 ? '20px' : '55px';
  tp.dataset.owner = owner;
  tp.dataset.denom = denom;
  tray.appendChild(tp);
  trayPieces.push({owner, denom, el: tp});
  gsap.from(tp, {x: owner==='fox' ? -150: 150, duration:0.6, ease:'power2.out'});

  // mark corresponding remaining piece un-pickable
  document.querySelector(`.piece[data-owner="${owner}"].pickable`)?.classList.remove('pickable');

  if (trayPieces.length === 2) {
    setTimeout(showTapPrompt, 500);
  }
}

function showTapPrompt() {
  document.getElementById('speech').textContent = "Now tap the piece you think is BIGGER!";
  trayPieces.forEach(tp => {
    tp.el.style.cursor = 'pointer';
    tp.el.addEventListener('click', () => onJudgeTap(tp.owner));
  });
}

let judgedOwner = null;
function onJudgeTap(owner) {
  if (roundLocked) return;
  roundLocked = true;
  judgedOwner = owner;
  trayPieces.forEach(tp => {
    if (tp.owner === owner) {
      gsap.to(tp.el, {scale:1.2, duration:0.3, yoyo:true, repeat:1, ease:'power1.inOut'});
    }
  });
  document.getElementById('speech').textContent = "Now stamp your justification!";
  stampArea.style.display = 'flex';
}

stampArea.addEventListener('click', (e) => {
  if (!e.target.classList.contains('stamp-btn')) return;
  const choice = e.target.dataset.choice;
  evaluateRound(choice);
});

function evaluateRound(choice) {
  stampArea.style.display = 'none';
  const correctOwner = 'fox'; // 1/3 > 1/4
  const predictedCorrect = judgedOwner === correctOwner;
  const justificationCorrect = choice === 'smaller';

  if (predictedCorrect && justificationCorrect) {
    sealCount++;
    sealCountEl.textContent = sealCount;
    showBanner("✅ Gold Fairness Seal!\n1/3 > 1/4 — more pieces means smaller pieces!", '#ffd93b');
    owlSay("More equal parts make smaller pieces, so one third is bigger!");
  } else {
    showBanner("🦉 Hmm... let's look closer!", '#f2b6b6');
    owlSay("Look — four cuts made skinnier pieces! Three pieces are each bigger.");
  }

  setTimeout(() => resetRound(), 2600);
}

function showBanner(text, color) {
  resultBanner.style.background = color;
  resultBanner.textContent = text;
  gsap.fromTo(resultBanner, {opacity:0, scale:0.5}, {opacity:1, scale:1, duration:0.4, ease:'back.out(2)'});
  gsap.to(resultBanner, {opacity:0, scale:0.5, duration:0.4, delay:2.0});
}

function owlSay(msg) {
  owlText.textContent = msg;
  owlText.style.display = 'block';
  setTimeout(() => owlText.style.display = 'none', 3200);
}

function resetRound() {
  roundLocked = false;
  trayPieces.forEach(tp => tp.el.remove());
  trayPieces = [];
  judgedOwner = null;
  buildBar(foxBar, 3, true);

  markPickable();
  document.getElementById('speech').textContent = "Drag a piece from each traveler to the Comparison Tray!";
}
