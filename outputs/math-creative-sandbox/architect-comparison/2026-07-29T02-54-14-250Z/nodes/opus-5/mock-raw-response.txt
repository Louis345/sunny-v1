<script>
const svgNS = "http://www.w3.org/2000/svg";
const panesSvg = document.getElementById('panes');
const glass = document.getElementById('glassRing');
const counterHud = document.getElementById('counterHud');
const nameplate = document.getElementById('nameplate');
const slotBottom = document.getElementById('slotBottom');
const plateLabel = document.getElementById('plateLabel');
const feedback = document.getElementById('feedback');
const dragHint = document.getElementById('dragHint');
const keypad = document.getElementById('keypad');
const channel = document.getElementById('channel');

let paneEls = [];
let sweepCount = 0;
let sweeping = false;
let locked = false;
let totalParts = 3;
let glowIndex = 0;

// Build a triangle divided into 3 equal panes (fresh context)
function buildTriangleThirds(){
  panesSvg.innerHTML = '';
  paneEls = [];
  const cx = 180, cy = 250, r = 150;
  // apex up triangle points
  const top = [cx, cy - r];
  const left = [cx - r*0.9, cy + r*0.55];
  const right = [cx + r*0.9, cy + r*0.55];
  const centroid = [(top[0]+left[0]+right[0])/3, (top[1]+left[1]+right[1])/3];

  const tris = [
    [top, left, centroid],
    [left, right, centroid],
    [right, top, centroid]
  ];
  glowIndex = 1; // middle bottom pane glows
  tris.forEach((pts,i)=>{
    const poly = document.createElementNS(svgNS,'polygon');
    poly.setAttribute('points', pts.map(p=>p.join(',')).join(' '));
    poly.classList.add('pane');
    if(i===glowIndex) poly.classList.add('glow');
    panesSvg.appendChild(poly);
    paneEls.push(poly);
  });
  totalParts = 3;
}

buildTriangleThirds();

// position glass initially
let glassPos = {x: 500, y: 400};
function placeGlass(x,y){
  glassPos.x = x; glassPos.y = y;
  glass.style.left = x+'px';
  glass.style.top = y+'px';
}
placeGlass(500,400);

let dragging = false;
glass.style.pointerEvents = 'auto';
glass.style.cursor = 'grab';

function within(px,py, poly){
  // point in polygon test
  const pts = poly.getAttribute('points').split(' ').map(s=>s.split(',').map(Number));
  let inside = false;
  for(let i=0,j=pts.length-1;i<pts.length;j=i++){
    const xi=pts[i][0], yi=pts[i][1], xj=pts[j][0], yj=pts[j][1];
    const intersect = ((yi>py)!=(yj>py)) && (px < (xj-xi)*(py-yi)/(yj-yi)+xi);
    if(intersect) inside=!inside;
  }
  return inside;
}

function svgLocalPoint(clientX, clientY){
  const rect = panesSvg.getBoundingClientRect();
  return {x: clientX-rect.left, y: clientY-rect.top};
}

function startDrag(e){
  if(locked) return;
  dragging = true;
  glass.style.cursor='grabbing';
}
function moveDrag(clientX, clientY){
  if(!dragging || locked) return;
  const stageRect = document.getElementById('stage').getBoundingClientRect();
  const x = clientX - stageRect.left;
  const y = clientY - stageRect.top;
  placeGlass(x,y);
  checkHover(clientX, clientY);
}
function endDrag(){
  dragging = false;
  glass.style.cursor='grab';
}

glass.addEventListener('mousedown', startDrag);
window.addEventListener('mousemove', e=>moveDrag(e.clientX, e.clientY));
window.addEventListener('mouseup', endDrag);

glass.addEventListener('touchstart', e=>{startDrag(e); e.preventDefault();}, {passive:false});
window.addEventListener('touchmove', e=>{
  if(e.touches[0]) moveDrag(e.touches[0].clientX, e.touches[0].clientY);
  e.preventDefault();
}, {passive:false});
window.addEventListener('touchend', endDrag);

let hoveredGlow = false;
function checkHover(clientX, clientY){
  const local = svgLocalPoint(clientX, clientY);
  let overGlow = false;
  paneEls.forEach(p=>{
    if(within(local.x, local.y, p) && p.classList.contains('glow')){
      overGlow = true;
    }
  });
  if(overGlow && !locked){
    glass.style.boxShadow = '0 0 45px rgba(255,220,120,0.9), inset 0 0 20px rgba(255,230,170,0.6)';
    if(!hoveredGlow){
      hoveredGlow = true;
      snapAndCount();
    }
  } else {
    glass.style.boxShadow = '0 0 25px rgba(255,200,100,0.5), inset 0 0 15px rgba(255,220,150,0.4)';
    hoveredGlow = false;
  }
}

function snapAndCount(){
  if(locked || sweeping) return;
  sweeping = true;
  locked = true;
  dragHint.style.opacity = '0';

  // snap glass to glow pane center
  const glowPane = paneEls[glowIndex];
  const bbox = glowPane.getBBox();
  const rect = panesSvg.getBoundingClientRect();
  const cx = rect.left + bbox.x + bbox.width/2 - document.getElementById('stage').getBoundingClientRect().left;
  const cy = rect.top + bbox.y + bbox.height/2 - document.getElementById('stage').getBoundingClientRect().top;
  placeGlass(cx, cy);
  glowPane.classList.add('lit');

  // outline whole lantern - pulse all panes briefly
  paneEls.forEach(p=>{ p.style.stroke = '#ffd873'; });

  // count sequence
  let count = 0;
  counterHud.textContent = 'parts:';
  const tick = setInterval(()=>{
    count++;
    counterHud.textContent = 'parts: ' + Array.from({length:count},(_,i)=>i+1).join(' … ');
    if(count>=totalParts){
      clearInterval(tick);
      setTimeout(()=>{
        revealNameplate();
      }, 500);
    }
  }, 550);
}

function revealNameplate(){
  nameplate.style.display = 'flex';
  buildKeypad();
  plateLabel.textContent = 'how many equal parts in the whole?';
}

function buildKeypad(){
  keypad.innerHTML = '';
  keypad.style.display = 'flex';
  [2,3,4,5].forEach(n=>{
    const k = document.createElement('div');
    k.className = 'key';
    k.textContent = n;
    k.addEventListener('click', ()=>submitAnswer(n));
    keypad.appendChild(k);
  });
}

function submitAnswer(n){
  if(n === totalParts){
    slotBottom.textContent = n;
    slotBottom.style.color = '#9effb0';
    showFeedback('✓ "one third" — the glass reads it true.', true);
    setTimeout(()=>{
      lightLantern();
    }, 700);
  } else {
    slotBottom.textContent = n;
    slotBottom.style.color = '#ff9e9e';
    showFeedback('The lantern flickers... "I counted ' + totalParts + '."', false);
    paneEls.forEach(p=>{ p.classList.add('flicker'); });
    setTimeout(()=>{
      slotBottom.textContent = '?';
      slotBottom.style.color = '#ffe9b8';
    }, 1200);
  }
}

function showFeedback(msg, ok){
  feedback.textContent = msg;
  feedback.style.color = ok ? '#bdf5c8' : '#f5c8c8';
  feedback.style.opacity = '1';
  setTimeout(()=>{ feedback.style.opacity = '0'; }, 2200);
}

function lightLantern(){
  paneEls.forEach(p=>p.classList.add('lit'));
  plateLabel.textContent = 'one third — 1/3';
  keypad.style.display = 'none';
  counterHud.textContent = '1 / 3 — one third';
  channel.style.width = '30%';
  document.getElementById('ferry').style.left = '35%';

  // gentle glow pulse on whole stage
  document.getElementById('stage').style.transition = 'filter 1s';
  document.getElementById('stage').style.filter = 'brightness(1.08)';
  setTimeout(()=>{document.getElementById('stage').style.filter='brightness(1)';}, 900);

  showFeedback('Lit: one third of the lantern is now named and glowing.', true);
}

document.getElementById('startBtn').addEventListener('click', ()=>{
  document.getElementById('missionCard').style.display = 'none';
});
</script>
