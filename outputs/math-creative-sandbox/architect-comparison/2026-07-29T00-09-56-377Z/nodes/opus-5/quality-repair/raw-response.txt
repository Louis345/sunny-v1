const NS = "http://www.w3.org/2000/svg";
let denom = 4, correctIdx = 0, guided = true, misses = 0, stage = 0;

const starsEl = document.getElementById('stars');
for(let i=0;i<70;i++){
  const s=document.createElement('div'); s.className='star';
  s.style.left = Math.random()*100+'%';
  s.style.top = Math.random()*50+'%';
  s.style.opacity = (Math.random()*0.5+0.3);
  starsEl.appendChild(s);

function polarToXY(cx,cy,r,angle){
  return [cx + r*Math.cos(angle), cy + r*Math.sin(angle)];
}

function buildPanes(n, pulseIdx, litIdx, targetSvgGroup, cx, cy, r){
  targetSvgGroup.innerHTML = '';
  const startAngle = -Math.PI/2 - Math.PI/n;
  for(let i=0;i<n;i++){
    const a0 = startAngle + i*(2*Math.PI/n);
    const a1 = a0 + (2*Math.PI/n);
    const [x0,y0] = polarToXY(cx,cy,r,a0);
    const [x1,y1] = polarToXY(cx,cy,r,a1);
    const large = (a1-a0) > Math.PI ? 1 : 0;
    const path = document.createElementNS(NS,'path');
    path.setAttribute('d', `M${cx},${cy} L${x0},${y0} A${r},${r} 0 ${large} 1 ${x1},${y1} Z`);
    path.setAttribute('fill', i===litIdx ? '#ffe9b0' : '#3b4a68');
    if(i===litIdx) path.classList.add('flash');
    path.setAttribute('stroke', '#1a2338');
    path.setAttribute('stroke-width','3');
    path.dataset.idx = i;
    path.style.cursor = 'pointer';
    path.style.transition = 'fill .18s';
    if(i===pulseIdx){
      path.classList.add('pulseTarget');
      path.setAttribute('fill', '#f0c060');
      pulseAnim(path);
    }
    targetSvgGroup.appendChild(path);
  }
}

let pulseInterval;
function pulseAnim(path){
  clearInterval(pulseInterval);
  let on = true;
  pulseInterval = setInterval(()=>{
    if(!document.body.contains(path)) { clearInterval(pulseInterval); return; }
    path.setAttribute('fill', on ? '#ffd27a' : '#f0c060');
    path.setAttribute('stroke', on ? '#fff2c0' : '#1a2338');
    on = !on;
  }, 380);
}

function setRoundA(n, idx, isGuided, label){
  mode = 'A';
  tilesEl.style.display = 'none';
  twinWrap.style.display = 'none';
  plateEl.classList.remove('blank');
  modeBtn.textContent = 'Mode A: Light the Pane';
  denom = n; correctIdx = idx; guided = isGuided;
  fracDen.textContent = n;
  fracNum.textContent = 1;
  roundLabel.textContent = label;
  buildPanes(n, isGuided ? idx : null, null, panesG, 150, 172, 128);
  msg.textContent = isGuided ? 'A pane is glowing on the glass. Tap it to light the lighthouse.'
                              : 'Count the panes. Tap the one pane that matches 1/'+n+'.';
  resetShip();
}

let engraveState = { correctDen: 4, litIdx: 1, numTile:null, denTile:null };
function setRoundB(n, litIdx, label){
  mode = 'B';
  tilesEl.style.display = 'flex';
  twinWrap.style.display = 'block';
  plateEl.classList.add('blank');
  modeBtn.textContent = 'Mode B: Engrave the Plate';
  roundLabel.textContent = label;
  engraveState = { correctDen: n, litIdx: litIdx, numTile:null, denTile:null };
  fracNum.textContent = '?';
  fracDen.textContent = '?';
  buildPanes(n, null, litIdx, panesG, 150, 172, 128);
  buildTiles(n);
  document.getElementById('twinLamp').innerHTML = '';
  msg.textContent = 'One pane is lit. Drag/tap numerals to engrave the matching unit fraction, then pull the lever.';
  resetShip(true);
}

function buildTiles(correctDen){
  const opts = new Set([1, correctDen]);
  while(opts.size < 3){
    opts.add(2 + Math.floor(Math.random()*7));
  }
  const arr = Array.from(opts).sort(()=>Math.random()-0.5);
  tilesEl.innerHTML = '';
  arr.forEach(v=>{
    const t = document.createElement('div');
    t.className='tile'; t.textContent = v; t.dataset.v = v;
    t.addEventListener('click', ()=>{
      if(engraveState.numTile === null){
        engraveState.numTile = v;
        fracNum.textContent = v;
        t.style.opacity = '0.35';
      } else if(engraveState.denTile === null){
        engraveState.denTile = v;
        fracDen.textContent = v;
        t.style.opacity = '0.35';
      }
    });
    tilesEl.appendChild(t);
  });
}

function resetShip(delay){
  const ship = document.getElementById('ship');
  const shipLamp = document.getElementById('shipLamp');
  ship.style.transition = 'none';
  ship.style.left = '-80px';
  shipLamp.style.opacity = 0;
  void ship.offsetWidth;
  ship.style.transition = 'left 2.2s ease-out';
  if(!delay) setTimeout(()=>{ ship.style.left = '640px'; }, 100);
}

function handleTap(i){
  if(mode !== 'A') return;
  clearInterval(pulseInterval);
  const paths = panesG.querySelectorAll('path');
  if(i === correctIdx){
    paths.forEach(p=>{ if(parseInt(p.dataset.idx) !== i) p.setAttribute('fill','#3b4a68'); });
    ignite(paths[i]);
  } else {
    paths[i].setAttribute('fill', '#6b3b3b');
    setTimeout(()=>{ paths[i].setAttribute('fill','#3b4a68'); }, 500);
    misses++;
    if(misses===1){
      msg.textContent = 'Not quite — that pane didn\\'t match 1/'+denom+'. Try counting again, or pull the lever.';
    } else if(misses===2){
      msg.textContent = 'Watch the panes number as they flash — pull the lever.';
      numberPanes();
    } else {
      msg.textContent = 'Let\\'s try guided — the right pane will glow.';
      setTimeout(()=> setRoundA(denom, correctIdx, true, roundLabel.textContent), 900);
      misses = 0;
    }
  }
}
panesG.addEventListener('click', (e)=>{
  const p = e.target.closest('path');
  if(p && mode==='A') handleTap(parseInt(p.dataset.idx));
});

function numberPanes(){
  const paths = panesG.querySelectorAll('path');
  paths.forEach((p,idx)=>{
    setTimeout(()=>{
      const t = document.createElementNS(NS,'text');
      const d = p.getAttribute('d').match(/M([\\d.\\-]+),([\\d.\\-]+)/);
      t.setAttribute('x', 150); t.setAttribute('y', 90);
      t.textContent = idx+1;
      t.setAttribute('fill', '#fff2c0');
      t.setAttribute('font-size','20');
      p.setAttribute('fill', '#e0c98a');
      setTimeout(()=> p.setAttribute('fill', idx===correctIdx? '#e0c98a':'#3b4a68'), 200);
    }, idx*220);
  });
}

function ignite(path){
  path.setAttribute('fill', '#ffe9b0');
  path.classList.add('flash');
  msg.textContent = 'Lit! One pane of ' + denom + ' — that is 1/' + denom + '.';

  const beam = document.createElementNS(NS,'path');
  const cx=150, cy=172;
  beam.setAttribute('d', `M${cx},${cy} L${cx-380},${cy-230} L${cx+380},${cy-230} Z`);
  beam.setAttribute('fill', 'rgba(255,230,160,0.18)');
  panesG.parentNode.appendChild(beam);
  setTimeout(()=>{ beam.remove(); }, 950);

  setTimeout(()=>{
    document.getElementById('shipLamp').style.opacity = 1;
    dockCount++;
    const docks = dockWrap.querySelectorAll('.dock');
    if(docks[dockCount-1]) docks[dockCount-1].classList.add('lit');
    msg.textContent = 'A ship answers from the fog and finds its way home.';
  }, 700);

  setTimeout(()=>{ advance(); }, 2600);
}

function advance(){
  stage++;
  misses = 0;
  if(stage === 1){
    setRoundA(6, 2, false, 'Beacon 2 — Ghost Trace');
    setTimeout(()=> ghostTrace(2), 500);
  } else if(stage === 2){
    setRoundA(5, 4, false, 'Beacon 2 — On Your Own');
  } else if(stage === 3){
    setRoundB(4, 1, 'Beacon 4 — Engrave the Plate');
  } else {
    setRoundA(7, 3, false, 'Beacon 5 — Denominators 7 & 8');
  }
}

function ghostTrace(idx){
  const paths = panesG.querySelectorAll('path');
  const p = paths[idx];
  if(!p) return;
  let count = 0;
  const t = setInterval(()=>{
    p.setAttribute('fill', count%2===0 ? '#f0c060' : '#3b4a68');
    p.setAttribute('stroke', count%2===0 ? '#fff2c0' : '#1a2338');
    count++;
    if(count>5){ clearInterval(t); p.setAttribute('fill', '#3b4a68'); p.setAttribute('stroke','#1a2338'); }
  }, 220);
}

document.getElementById('lever').addEventListener('click', ()=>{
  const btn = document.getElementById('leverBtn');
  btn.style.transform = 'translateY(3px)';
  setTimeout(()=>{ btn.style.transform='translateY(0)'; }, 200);
  if(mode === 'A'){
    playAnatomy();
  } else {
    stampPlate();
  }
});

function playAnatomy(){
  const paths = Array.from(panesG.querySelectorAll('path'));
  fracDen.style.fontSize='42px'; fracDen.style.transition='font-size .2s';
  msg.textContent = 'Bottom number = how many equal panes.';
  let i = 0;
  const seq = setInterval(()=>{
    if(i>0) paths[i-1].setAttribute('fill', paths[i-1].classList.contains('flash') ? '#ffe9b0':'#3b4a68');
    if(i < paths.length){
      paths[i].setAttribute('fill', '#e0c98a');
      i++;
    } else {
      clearInterval(seq);
      fracDen.style.fontSize='30px';
      fracNum.style.fontSize='42px';
      msg.textContent = 'Top number = light exactly one.';
      setTimeout(()=>{
        fracNum.style.fontSize='30px';
        paths.forEach((p,idx)=>{ p.setAttribute('fill', idx===correctIdx && p.classList.contains('flash') ? '#ffe9b0' : '#3b4a68'); });
        msg.textContent = guided ? 'Tap the glowing pane.' : 'Now find the one pane that matches 1/'+denom+'.';
      }, 900);
    }
  }, 260);
}
