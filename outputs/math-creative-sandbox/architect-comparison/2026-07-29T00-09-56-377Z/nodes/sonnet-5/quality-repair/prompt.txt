function drawStaticCut(x1,y1,x2,y2, equal){
  dragLine.setAttribute('x1',x1); dragLine.setAttribute('y1',y1);
  dragLine.setAttribute('x2',x2); dragLine.setAttribute('y2',y2);
  dragLine.setAttribute('visibility','visible');
  renderPieces(x1,y1,x2,y2,equal);
}

function renderPieces(x1,y1,x2,y2, forcedEqual){
  piecesG.innerHTML = '';
  // Approximate: just show two colored half-circles via clip using paths
  const dx = x2-x1, dy = y2-y1;
  const len = Math.sqrt(dx*dx+dy*dy) || 1;
  const nx = -dy/len, ny = dx/len;
  const far = 400;
  const p1 = `${x1+nx*far},${y1+ny*far}`;
  const p2 = `${x2+nx*far},${y2+ny*far}`;
  const p3 = `${x2-nx*far},${y2-ny*far}`;
  const p4 = `${x1-nx*far},${y1-ny*far}`;
  const cls = forcedEqual ? 'piece-good' : 'piece-bad';
  const cls2 = forcedEqual ? 'piece-good' : 'piece-bad';

  const clipA = document.createElementNS("http://www.w3.org/2000/svg","clipPath");
  clipA.setAttribute("id","clipA");
  const polyA = document.createElementNS("http://www.w3.org/2000/svg","polygon");
  polyA.setAttribute("points", `${x1},${y1} ${p1} ${p2} ${x2},${y2}`);
  clipA.appendChild(polyA);

  const clipB = document.createElementNS("http://www.w3.org/2000/svg","clipPath");
  clipB.setAttribute("id","clipB");
  const polyB = document.createElementNS("http://www.w3.org/2000/svg","polygon");
  polyB.setAttribute("points", `${x1},${y1} ${p4} ${p3} ${x2},${y2}`);
  clipB.appendChild(polyB);

  piecesG.appendChild(clipA);
  piecesG.appendChild(clipB);

  const circA = document.createElementNS("http://www.w3.org/2000/svg","circle");
  circA.setAttribute("cx",190); circA.setAttribute("cy",190); circA.setAttribute("r",150);
  circA.setAttribute("clip-path","url(#clipA)");
  circA.setAttribute("class", cls);
  circA.setAttribute("id","pieceA");

  const circB = document.createElementNS("http://www.w3.org/2000/svg","circle");
  circB.setAttribute("cx",190); circB.setAttribute("cy",190); circB.setAttribute("r",150);
  circB.setAttribute("clip-path","url(#clipB)");
  circB.setAttribute("class", cls2);
  circB.setAttribute("id","pieceB");

  piecesG.appendChild(circA);
  piecesG.appendChild(circB);

  // compute rough area ratio for meter using shoelace on polygon approx (clipped to circle bounding box is imprecise but good enough visually)
  const areaA = Math.abs(polyArea([{x:x1,y:y1}, pt(p1), pt(p2), {x:x2,y:y2}]));
  const areaB = Math.abs(polyArea([{x:x1,y:y1}, pt(p4), pt(p3), {x:x2,y:y2}]));
  const total = areaA+areaB;
  let pctA = total ? (areaA/total*100) : 50;
  pctA = Math.max(5, Math.min(95, pctA));
  segA.style.width = pctA + '%';
  segB.style.width = (100-pctA) + '%';

  if(!forcedEqual){
    circA.classList.add('wobble');
    circB.classList.add('wobble');
  } else {
    circA.classList.add('popIn');
    circB.classList.add('popIn');
  }
}

function pt(str){ const [x,y] = str.split(',').map(Number); return {x,y}; }
function polyArea(pts){
  let a = 0;
  for(let i=0;i<pts.length;i++){
    const j = (i+1)%pts.length;
    a += pts[i].x*pts[j].y - pts[j].x*pts[i].y;
  }
  return a/2;
}

// ---- interactive cut ----
function getPos(evt){
  const rect = svg.getBoundingClientRect();
  const clientX = evt.touches ? evt.touches[0].clientX : evt.clientX;
  const clientY = evt.touches ? evt.touches[0].clientY : evt.clientY;
  return { x: clientX - rect.left, y: clientY - rect.top };
}

function onDown(e){
  if(demoPlaying) return;
  e.preventDefault();
  dragging = true;
  start = getPos(e);
  dragLine.setAttribute('x1', start.x);
  dragLine.setAttribute('y1', start.y);
  dragLine.setAttribute('x2', start.x);
  dragLine.setAttribute('y2', start.y);
  dragLine.setAttribute('visibility','visible');
  piecesG.innerHTML = '';
  feedback.textContent = '';
  retryBtn.style.display = 'none';
}
function onMove(e){
  if(!dragging) return;
  e.preventDefault();
  const p = getPos(e);
  dragLine.setAttribute('x2', p.x);
  dragLine.setAttribute('y2', p.y);
  // live meter based on angle offset from center diameter
  const mid = {x:(start.x+p.x)/2, y:(start.y+p.y)/2};
  const distFromCenter = Math.sqrt((mid.x-center.x)**2 + (mid.y-center.y)**2);
  const closeness = Math.max(0, 1 - distFromCenter/120);
  const pct = 50 + (1-closeness)*30*(Math.random()>0.5?1:-1)*0.3;
  segA.style.width = pct + '%';
  segB.style.width = (100-pct) + '%';
}
function onUp(e){
  if(!dragging) return;
  dragging = false;
  const p = getPos(e);
  const x1 = parseFloat(dragLine.getAttribute('x1'));
  const y1 = parseFloat(dragLine.getAttribute('y1'));
  const x2 = p.x, y2 = p.y;

  // Determine equality: does the line pass near center (within tolerance)?
  const mid = {x:(x1+x2)/2, y:(y1+y2)/2};
  const distFromCenter = Math.sqrt((mid.x-center.x)**2 + (mid.y-center.y)**2);
  const lineLen = Math.sqrt((x2-x1)**2+(y2-y1)**2);
  const isEqual = distFromCenter < 22 && lineLen > 200;

  renderPieces(x1,y1,x2,y2, isEqual);

  if(isEqual){
    face.textContent = "😄";
    feedback.textContent = "Perfectly equal! Fair cut! 🎉";
    setCoins(coins + 10);
    patience = Math.min(100, patience + 10);
    banner.textContent = "Great! Next order coming up...";
    setTimeout(()=> resetRound(false), 1400);
  } else {
    face.textContent = "😟";
    feedback.textContent = "Not equal — one piece is bigger!";
    patience = Math.max(0, patience - 20);
    retryBtn.style.display = 'block';
    banner.textContent = "Oops! Try cutting through the center.";
  }
  updatePatience();
}

function updatePatience(){
  patienceBar.style.width = patience + '%';
  if(patience > 60) patienceBar.style.background = '#6cd66c';
  else if(patience > 30) patienceBar.style.background = '#e8c94a';
  else patienceBar.style.background = '#e05a5a';
}

function resetRound(keepFeedback){
  piecesG.innerHTML = '';
  dragLine.setAttribute('visibility','hidden');
  retryBtn.style.display = 'none';
  segA.style.width = '50%';
  segB.style.width = '50%';
  face.textContent = "😐";
  if(!keepFeedback){ feedback.textContent = ''; }
  banner.textContent = "Drag a straight line across the cookie to cut it!";
}
