<script>
let pairs = [[2,8],[3,7],[5,6],[6,8],[4,4]];
let pairIdx = 0;
let denomA, denomB, chosenBigger = null, correctSym = null;
let brewPct = 0;
let stage = 0; // 0 pick bottle, 1 pick symbol, 2 pick reasoning

const bottleA = document.getElementById('bottleA');
const bottleB = document.getElementById('bottleB');
const wrapA = document.getElementById('wrapA');
const wrapB = document.getElementById('wrapB');
const labelA = document.getElementById('labelA');
const labelB = document.getElementById('labelB');
const symbolSlot = document.getElementById('symbolSlot');
const stageHint = document.getElementById('stageHint');
const hint = document.getElementById('hint');
const reasoningRow = document.getElementById('reasoningRow');
const wheel = document.getElementById('wheel');
const whisper = document.getElementById('whisper');
const meterFill = document.getElementById('meterFill');
const brewLiquid = document.getElementById('brewLiquid');
const symBtns = document.querySelectorAll('.symBtn');

function buildBottle(el, denom){
  el.innerHTML = '';
  const segHeight = 100 / denom;
  for (let i=0;i<denom;i++){
    const seg = document.createElement('div');
    seg.className = 'segment';
    seg.style.top = (i*segHeight)+'%';
    seg.style.height = segHeight+'%';
    el.appendChild(seg);
  }
  const dose = document.createElement('div');
  dose.className = 'dose glow';
  dose.style.bottom = '0%';
  dose.style.height = segHeight+'%';
  el.appendChild(dose);
}

function setStage(s){
  stage = s;
  symBtns.forEach(b => b.classList.toggle('locked', stage !== 1));
  reasoningRow.classList.toggle('active', stage === 2);
  bottleA.classList.toggle('disabled', stage !== 0);
  bottleB.classList.toggle('disabled', stage !== 0);
  if (stage === 0) stageHint.textContent = '1. Tap the bigger sip';
  else if (stage === 1) stageHint.textContent = '2. Drag the correct sign into the slot';
  else stageHint.textContent = '3. Pick the reason';
}

function newRound(){
  chosenBigger = null;
  bottleA.classList.remove('chosen','wrongPick');
  bottleB.classList.remove('chosen','wrongPick');
  wrapA.style.transform = '';
  wrapB.style.transform = '';
  bottleA.classList.remove('overlayGhost');
  bottleB.classList.remove('overlayGhost');
  symbolSlot.textContent = '?';
  symbolSlot.style.borderColor = '#b088ff';
  hint.textContent = '';
  reasoningRow.innerHTML = '';
  setStage(0);

  const p = pairs[pairIdx % pairs.length];
  pairIdx++;
  denomA = p[0]; denomB = p[1];
  labelA.textContent = '1/' + denomA;
  labelB.textContent = '1/' + denomB;
  buildBottle(bottleA, denomA);
  buildBottle(bottleB, denomB);

  if (denomA < denomB) correctSym = '>';
  else if (denomA > denomB) correctSym = '<';
  else correctSym = '=';

  whisper.textContent = 'Bigger sip wins. Tap it.';

  wheel.style.transition = 'none';
  wheel.style.transform = 'rotate(0deg)';
  void wheel.offsetWidth;
  wheel.style.transition = 'transform 1.1s cubic-bezier(.2,.8,.2,1)';
  wheel.style.transform = 'rotate(720deg)';
  setTimeout(()=>{
    wheel.style.transition = 'none';
    wheel.style.transform = 'rotate(0deg)';
  }, 1150);
}

function pourCauldron(amount){
  brewPct = Math.min(100, brewPct + amount);
  meterFill.style.width = brewPct + '%';
  brewLiquid.style.height = Math.min(90, 10 + brewPct*0.8) + '%';
}

function overlayReveal(){
  // slide bottles together, one translucent, to show true size difference
  wrapA.style.transform = 'translateX(75px)';
  wrapB.style.transform = 'translateX(-75px)';
  bottleB.classList.add('overlayGhost');
}

function chooseBottle(which){
  if (stage !== 0) return;
  const isBigger = (which === 'A' && (correctSym === '>' || correctSym === '=')) ||
                    (which === 'B' && (correctSym === '<' || correctSym === '='));
  if (correctSym === '=' ) {
    // equal pair: no single bottle is "bigger" -- must be routed to '=' path
    (which==='A'?bottleA:bottleB).classList.add('wrongPick');
    hint.textContent = 'Same size! Tap "=" below instead of picking a bottle.';
    setStage(1);
    chosenBigger = 'EQUAL';
    return;
  }
  if (isBigger) {
    (which==='A'?bottleA:bottleB).classList.add('chosen');
    chosenBigger = which;
    hint.textContent = '';
    setStage(1);
  } else {
    (which==='A'?bottleA:bottleB).classList.add('wrongPick');
    hint.textContent = 'Tink! Count the parts — which bottle has fewer cuts?';
    overlayReveal();
    setTimeout(()=>{
      wrapA.style.transform = '';
      wrapB.style.transform = '';
      bottleB.classList.remove('overlayGhost');
      bottleA.classList.remove('wrongPick');
      bottleB.classList.remove('wrongPick');
    }, 1600);
  }
}

bottleA.addEventListener('click', ()=>chooseBottle('A'));
bottleB.addEventListener('click', ()=>chooseBottle('B'));

function applySymbol(sym){
  if (stage !== 1) return;
  symbolSlot.textContent = sym;
  if (sym === correctSym) {
    symbolSlot.style.borderColor = '#7CFC00';
    pourCauldron(10);
    setStage(2);
    showReasoning();
    whisper.textContent = 'Splash! Now say why.';
    hint.textContent = '';
  } else {
    symbolSlot.style.borderColor = '#ff4d4d';
    hint.textContent = 'Not quite — compare the number of cuts again.';
    overlayReveal();
    setTimeout(()=>{
      wrapA.style.transform = '';
      wrapB.style.transform = '';
      bottleB.classList.remove('overlayGhost');
    }, 1600);
  }
}

symBtns.forEach(btn=>{
  btn.addEventListener('click', ()=>{
    if (stage !== 1) return;
    applySymbol(btn.getAttribute('data-sym'));
  });
  btn.addEventListener('dragstart', (e)=>{
    if (stage !== 1) { e.preventDefault(); return; }
    btn.classList.add('dragging');
    e.dataTransfer.setData('text/plain', btn.getAttribute('data-sym'));
  });
  btn.addEventListener('dragend', ()=> btn.classList.remove('dragging'));
});

symbolSlot.addEventListener('dragover', (e)=>{ if (stage===1) e.preventDefault(); });
symbolSlot.addEventListener('drop', (e)=>{
  e.preventDefault();
  if (stage !== 1) return;
  const sym = e.dataTransfer.getData('text/plain');
  applySymbol(sym);
});

function showReasoning(){
  reasoningRow.innerHTML = '';
  const tiles = [
    {text:'Fewer equal parts means each part is bigger.', correct:true},
    {text:'More equal parts means each part is bigger.', correct:false},
    {text:'The bottle color decides the size.', correct:false}
  ];
  tiles.sort(()=>Math.random()-0.5);
  tiles.forEach(t=>{
    const d = document.createElement('div');
    d.className = 'reasonTile';
    d.textContent = t.text;
    d.addEventListener('click', ()=>{
      if (stage !== 2) return;
      if (t.correct){
        d.classList.add('correct');
        pourCauldron(10);
        hint.textContent = '';
        whisper.textContent = '"Fewer cuts, bigger sip!" — the imps cheer!';
        stage = 3;
        setTimeout(newRound, 1500);
      } else {
        d.classList.add('wrong');
        hint.textContent = 'Count the parts — which bottle has fewer cuts?';
      }
    });
    reasoningRow.appendChild(d);
  });
}

newRound();
</script>
</body>
