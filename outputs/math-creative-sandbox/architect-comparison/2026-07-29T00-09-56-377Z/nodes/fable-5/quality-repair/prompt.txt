<script>
let pairs = [[3,6],[2,8],[3,7],[5,6],[4,4]];
let pairIdx = 0;
let denomA, denomB, chosenBigger = null, symbolPlaced = null, correctSym = null;
let brewPct = 0;
let stage = 0; // 0 pick bottle, 1 pick symbol, 2 pick reasoning

const bottleA = document.getElementById('bottleA');
const bottleB = document.getElementById('bottleB');
const labelA = document.getElementById('labelA');
const labelB = document.getElementById('labelB');
const symbolSlot = document.getElementById('symbolSlot');
const hint = document.getElementById('hint');
const reasoningRow = document.getElementById('reasoningRow');
const wheel = document.getElementById('wheel');
const whisper = document.getElementById('whisper');
const meterFill = document.getElementById('meterFill');
const brewLiquid = document.getElementById('brewLiquid');

function buildBottle(el, denom, biggerIsThis) {
  el.innerHTML = '';
  const segHeight = 100 / denom;
  for (let i=0;i<denom;i++){
    const seg = document.createElement('div');
    seg.className = 'segment';
    seg.style.top = (i*segHeight)+'%';
    seg.style.height = segHeight+'%';
    el.appendChild(seg);
  }
  const dose = document.createElement('div');
  dose.className = 'dose glow';
  dose.style.bottom = '0%';
  dose.style.height = segHeight+'%';
  el.appendChild(dose);
}

function newRound(){
  stage = 0;
  chosenBigger = null; symbolPlaced = null;
  bottleA.classList.remove('chosen','wrongPick');
  bottleB.classList.remove('chosen','wrongPick');
  symbolSlot.textContent = '?';
  hint.textContent = '';
  reasoningRow.innerHTML = '';

  const p = pairs[pairIdx % pairs.length];
  pairIdx++;
  denomA = p[0]; denomB = p[1];
  labelA.textContent = '1/' + denomA;
  labelB.textContent = '1/' + denomB;
  buildBottle(bottleA, denomA);
  buildBottle(bottleB, denomB);

  if (denomA < denomB) correctSym = '>';
  else if (denomA > denomB) correctSym = '<';
  else correctSym = '=';

  whisper.textContent = 'Bigger sip wins. Tap it.';

  // spin animation
  wheel.style.transition = 'transform 1.1s cubic-bezier(.2,.8,.2,1)';
  wheel.style.transform = 'rotate(' + (720 + Math.random()*360) + 'deg)';
  setTimeout(()=>{ wheel.textContent = '1/'+denomA+' vs 1/'+denomB; }, 900);
}

function pourCauldron(amount){
  brewPct = Math.min(100, brewPct + amount);
  meterFill.style.width = brewPct + '%';
  brewLiquid.style.height = Math.min(90, 10 + brewPct*0.8) + '%';
}

function chooseBottle(which){
  if (stage !== 0) return;
  const isBigger = (which === 'A' && correctSym === '>') ||
                    (which === 'B' && correctSym === '<') ||
                    (correctSym === '=' );
  if (isBigger) {
    (which==='A'?bottleA:bottleB).classList.add('chosen');
    chosenBigger = which;
    hint.textContent = '';
    stage = 1;
    whisper.textContent = 'Now drag the correct sign into the slot.';
  } else {
    (which==='A'?bottleA:bottleB).classList.add('wrongPick');
    hint.textContent = 'Tink! Count the parts — which bottle has fewer cuts?';
  }
}

bottleA.addEventListener('click', ()=>chooseBottle('A'));
bottleB.addEventListener('click', ()=>chooseBottle('B'));

document.querySelectorAll('.symBtn').forEach(btn=>{
  btn.addEventListener('click', ()=>{
    if (stage !== 1) return;
    const sym = btn.getAttribute('data-sym');
    symbolSlot.textContent = sym;
    if (sym === correctSym) {
      symbolSlot.style.borderColor = '#7CFC00';
      pourCauldron(10);
      stage = 2;
      showReasoning();
      whisper.textContent = 'Splash! Now say why.';
    } else {
      symbolSlot.style.borderColor = '#ff4d4d';
      hint.textContent = 'Not quite — compare the number of cuts again.';
    }
  });
});

function showReasoning(){
  reasoningRow.innerHTML = '';
  const tiles = [
    {text:'Fewer equal parts means each part is bigger.', correct:true},
    {text:'More equal parts means each part is bigger.', correct:false},
    {text:'The bottle color decides the size.', correct:false}
  ];
  tiles.sort(()=>Math.random()-0.5);
  tiles.forEach(t=>{
    const d = document.createElement('div');
    d.className = 'reasonTile';
    d.textContent = t.text;
    d.addEventListener('click', ()=>{
      if (t.correct){
        d.classList.add('correct');
        pourCauldron(10);
        hint.textContent = '';
        whisper.textContent = '"Fewer cuts, bigger sip!" — the imps cheer!';
        setTimeout(newRound, 1400);
      } else {
        d.classList.add('wrong');
        hint.textContent = 'Count the parts — which bottle has fewer cuts?';
      }
    });
    reasoningRow.appendChild(d);
  });
}

newRound();
