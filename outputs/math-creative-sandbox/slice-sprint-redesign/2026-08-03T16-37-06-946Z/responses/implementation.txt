let AC=null;
function ac(){ if(!AC){ try{AC=new (window.AudioContext||window.webkitAudioContext)();}catch(e){} } return AC; }
function tone(f,d,type="sine",vol=.18,delay=0){
  const c=ac(); if(!c) return;
  const o=c.createOscillator(), g=c.createGain();
  o.type=type; o.frequency.value=f; g.gain.value=0;
  o.connect(g); g.connect(c.destination);
  const t=c.currentTime+delay;
  g.gain.linearRampToValueAtTime(vol,t+0.02);
  g.gain.exponentialRampToValueAtTime(0.001,t+d);
  o.start(t); o.stop(t+d+0.05);
}

const CORAL="#ff6f52", CORAL_D="#d94b31";
function rectWhole(parts, shadeIndex, w=300,h=150){
  const pw=w/parts;
  let s=`<svg viewBox="0 0 ${w} ${h}" width="${w}" height="${h}">
    <rect x="1" y="1" width="${w-2}" height="${h-2}" fill="#fdf6e6" stroke="#4b3a24" stroke-width="4"/>`;
  for(let i=0;i<parts;i++){
    if(i===shadeIndex) s+=`<rect x="${i*pw+2}" y="3" width="${pw-4}" height="${h-6}" fill="${CORAL}"/>`;
  }
  for(let i=1;i<parts;i++) s+=`<line x1="${i*pw}" y1="1" x2="${i*pw}" y2="${h-1}" stroke="#4b3a24" stroke-width="4"/>`;
  s+=`<rect x="1" y="1" width="${w-2}" height="${h-2}" fill="none" stroke="#4b3a24" stroke-width="4"/></svg>`;
  return s;
}
function sectorPath(cx,cy,r,a0,a1){
  const x0=cx+r*Math.cos(a0), y0=cy+r*Math.sin(a0);
  const x1=cx+r*Math.cos(a1), y1=cy+r*Math.sin(a1);
  const large=(a1-a0)>Math.PI?1:0;
  return `M${cx},${cy} L${x0},${y0} A${r},${r} 0 ${large} 1 ${x1},${y1} Z`;
}
function circleWhole(parts, shadeIndex, size=190){
  const r=size/2-6, c=size/2;
  let s=`<svg viewBox="0 0 ${size} ${size}" width="${size}" height="${size}">
    <circle cx="${c}" cy="${c}" r="${r}" fill="#fdf6e6" stroke="#4b3a24" stroke-width="4"/>`;
  const step=2*Math.PI/parts;
  for(let i=0;i<parts;i++){
    const a0=-Math.PI/2+i*step, a1=a0+step;
    if(i===shadeIndex) s+=`<path d="${sectorPath(c,c,r-2,a0,a1)}" fill="${CORAL}"/>`;
  }
  for(let i=0;i<parts;i++){
    const a=-Math.PI/2+i*step;
    s+=`<line x1="${c}" y1="${c}" x2="${c+r*Math.cos(a)}" y2="${c+r*Math.sin(a)}" stroke="#4b3a24" stroke-width="4"/>`;
  }
  s+=`<circle cx="${c}" cy="${c}" r="${r}" fill="none" stroke="#4b3a24" stroke-width="4"/></svg>`;
  return s;
}

function scoreRule(t){
  const s=t.toLowerCase();
  const saysThird = /1\/3|one ?third|thirds?\b/.test(s) && !/1\/4 is (larger|bigger)/.test(s);
  const dirRule = /(more|4|four).{0,26}(smaller)|(fewer|less|3|three).{0,26}(bigger|larger)|smaller.{0,20}(more|four)|bigger.{0,20}(fewer|three)/.test(s)
                 || /(more equal parts).{0,30}(smaller)/.test(s) || /(fewer equal parts).{0,30}(bigger|larger)/.test(s);
  const sameWhole=/same size|same whole|both.{0,12}same|equal size whole/.test(s);
  const numeralOnly=/bottom number|denominator.{0,20}(bigger|larger)/.test(s) && !dirRule;
  let pts=0;
  if(saysThird) pts+=0.35;
  if(dirRule) pts+=0.4;
  if(sameWhole) pts+=0.25;
  if(numeralOnly) pts=Math.max(0,pts-0.25);
  pts=Math.round(Math.min(1,pts)*100)/100;
  let feedback, spoken;
  if(pts>=0.85){ feedback="Full credit: names 1/3, uses part size, and states the same-size-whole condition."; spoken="Perfect rule! Same size wholes, fewer equal parts, bigger pieces. The beacon lights!"; }
  else if(pts>=0.5){ feedback="Partial-to-strong credit: the direction of your rule is right. Remember to say the wholes are the same size."; spoken="Good rule — your direction is right. Say the wholes are the same size and it's airtight. Beacon lit!"; }
  else if(numeralOnly){ feedback="Careful: this decides by the bottom number, not by the size of one part."; spoken="Careful — that decides by the bottom number. The real rule is about the size of one part."; }
  else { feedback="The rule needs the part-size idea: more equal parts in the same whole makes each part smaller."; spoken="The rule is about part size: more equal parts in the same whole makes each part smaller."; }
  return {score:pts, feedback, spoken};
}

  right:{parts:2,tag:"1/2",correct:true,label:"1/2"},wrong:{parts:4,tag:"1/4",correct:false,label:"1/4"},
  spoken:"Quick round. Two circles are exactly the same size. One shows one of two equal parts shaded. The other shows one of four equal parts shaded. Tap the crate with the larger shaded part."};

function runFlow(){

    if(M["node.N3B.item.N3B-I1.correct"]===false && !ghostUsed){
      enableGhost(true);
      gsap.fromTo(btnGhost,{scale:1},{scale:1.25,duration:.35,yoyo:true,repeat:3});
      say("Want the ghost check? Tap the ghost — or just look and pick.");
    }
    setTimeout(()=>comparisonBeat(b2, ()=>{
      setTimeout(()=>{
        say("Whistle drops. Compass slice time.");
        setTimeout(()=>shadingBeat(()=>{
          gsap.to("#beatArea",{opacity:.6,duration:.6,yoyo:true,repeat:1});
          say("Quiet now. Walk to the lighthouse door.");
          setTimeout(()=>explainBeat(finale),2000);
        }),900);
      },400);
    }),300);
  });
}
