const AC=new (window.AudioContext||window.webkitAudioContext)();
function tone(f,d,type,v){const o=AC.createOscillator(),g=AC.createGain();o.type=type;o.frequency.value=f;g.gain.setValueAtTime(v,AC.currentTime);g.gain.exponentialRampToValueAtTime(.001,AC.currentTime+d);o.connect(g);g.connect(AC.destination);o.start();o.stop(AC.currentTime+d);}
function drum(){tone(90,.25,'square',.3);}
function horn(){tone(140,.6,'sawtooth',.25);setTimeout(()=>tone(110,.6,'sawtooth',.2),120);}
function chime(){tone(880,.15,'sine',.2);setTimeout(()=>tone(1200,.2,'sine',.2),150);}

const stages=[
 {rows:3,cols:4,title:'Training Yard Scroll',text:'Form 3 rows of 4 soldiers.'},
 {rows:5,cols:6,title:'Dispatch Scroll',text:'"5 wagons, 6 soldiers each." Build the ranks!',decoy:{text:'General Vex sneers: "Ha! My scouts counted 6 rows of 5 — that\\'s 35 soldiers!"',claimed:35}}
];
let stageIdx=0,filled=0,remaining=0,rows=0,cols=0,progressPct=2,reinforcements=0,firstTap=true;

function loadStage(i){
  const s=stages[i];rows=s.rows;cols=s.cols;filled=0;remaining=rows*cols;
  document.getElementById('scrollTitle').textContent=s.title;
  document.getElementById('scrollText').textContent=s.text;
  document.getElementById('trayCount').textContent=remaining;
  document.getElementById('skipcount').innerHTML='';
  document.getElementById('feedback').innerHTML='';
  document.getElementById('mentorHint').textContent='';
  const btn=document.getElementById('sendBtn');
  btn.classList.add('hidden');btn.disabled=true;btn.textContent='Send Formation ⚔️';btn.onclick=sendFormation;
  const grid=document.getElementById('grid');
  grid.style.gridTemplateColumns='repeat('+cols+',46px)';
  grid.innerHTML='';
  for(let k=0;k<rows*cols;k++){const d=document.createElement('div');d.className='cell';d.id='cell'+k;grid.appendChild(d);}
  glowNext();
}
function glowNext(){
  document.querySelectorAll('.cell.glow').forEach(c=>c.classList.remove('glow'));
  if(filled<rows*cols) document.getElementById('cell'+filled).classList.add('glow');
}
function placeToken(){
  if(firstTap){firstTap=false;document.getElementById('tapHint').classList.add('hidden');}
  if(remaining<=0||filled>=rows*cols) return;
  const cell=document.getElementById('cell'+filled);
  cell.classList.remove('glow');cell.classList.add('filled');cell.textContent='♙';
  filled++;remaining--;
  document.getElementById('trayCount').textContent=remaining;
  glowNext();
  if(filled%cols===0){
    drum();
    for(let r=0;r<cols;r++){document.getElementById('cell'+(filled-1-r)).classList.add('rowflash');}
    const span=document.createElement('span');span.textContent=filled;
    document.getElementById('skipcount').appendChild(span);
  }
  if(filled===rows*cols){
    if(stages[stageIdx].decoy){setTimeout(showDecoy,400);}
    else{const btn=document.getElementById('sendBtn');btn.classList.remove('hidden');btn.disabled=false;}
  }
}
function showDecoy(){
  const d=stages[stageIdx].decoy;
  document.getElementById('feedback').innerHTML='<div class="decoy-box"><div class="taunt">'+d.text+'</div><div class="decoy-btns"><button class="confirmBtn" onclick="resolveDecoy(true)">✅ Confirm — Matches</button><button class="flagBtn" onclick="resolveDecoy(false)">🚩 Flag as Miscount</button></div></div>';
}
function resolveDecoy(confirmed){
  const d=stages[stageIdx].decoy;const actual=rows*cols;
  const decoyIsWrong=d.claimed!==actual;
  const playerCorrect=confirmed?!decoyIsWrong:decoyIsWrong;
  const fb=document.getElementById('feedback');const btn=document.getElementById('sendBtn');
  if(playerCorrect){
    chime();
    fb.innerHTML='<div style="color:#7CFC7C;font-weight:bold">🕵️ SPY EXPOSED! Real total is '+actual+', not '+d.claimed+'. Bonus advance!</div>';
    advanceMarker(actual+18);
  }else{
    horn();reinforcements++;document.getElementById('reinfCount').textContent=reinforcements;
    fb.innerHTML='<div class="taunt">😈 "Too slow, commander!" The fortress gains a reinforcement.</div>';
    document.getElementById('mentorHint').textContent='🎖️ Mentor: "Recount the scroll — multiply wagons × soldiers each!"';
    stepBack();advanceMarker(actual*0.4);
  }
  setTimeout(()=>{btn.classList.remove('hidden');btn.disabled=false;btn.textContent='Continue ⚔️';btn.onclick=nextStage;},1300);
}
function stepBack(){
  document.querySelectorAll('.grid .cell.filled').forEach(c=>c.classList.add('stepback'));
  setTimeout(()=>{document.querySelectorAll('.stepback').forEach(c=>{c.classList.remove('filled','stepback');c.textContent='';});},500);
}
function sendFormation(){
  const total=rows*cols;
  document.getElementById('feedback').innerHTML='<div style="color:#7CFC7C;font-weight:bold">⚔️ Formation of '+rows+'×'+cols+'='+total+' marches forward!</div>';
  advanceMarker(total*1.6);
  const btn=document.getElementById('sendBtn');btn.disabled=true;
  setTimeout(()=>{btn.textContent='Continue ⚔️';btn.disabled=false;btn.onclick=nextStage;},1000);
}
function advanceMarker(amount){
  progressPct=Math.min(92,progressPct+amount/60*20);
  document.getElementById('marker').style.left=progressPct+'%';
  if(progressPct>65) document.getElementById('gate').classList.add('cracked');
}
function nextStage(){
  stageIdx++;
  if(stageIdx<stages.length){loadStage(stageIdx);}
  else{
    document.getElementById('lockedCard').innerHTML='🔒 <b>Iron Gate — Final Siege</b><br>Chained scrolls, rapid decoys, countdown clock... (next training session)';
    document.getElementById('feedback').innerHTML='<div style="color:#f0d078;font-weight:bold">Training siege complete! Tap 👑 Preview Payoff to see how victory looks.</div>';
    document.getElementById('sendBtn').classList.add('hidden');
  }
}
function showModal(){document.getElementById('modal').classList.remove('hidden');}
function hideModal(){document.getElementById('modal').classList.add('hidden');}
loadStage(0);
