function beep(freq,dur,type){

  const ctx=new (window.AudioContext||window.webkitAudioContext)();
  const o=ctx.createOscillator(), gain=ctx.createGain();
  o.type=type||'square'; o.frequency.value=freq;

function makeDraggable(el,onDrop){
 el.addEventListener('pointerdown',e=>{
  const r=el.getBoundingClientRect();
  const offX=e.clientX-r.left, offY=e.clientY-r.top;
  const origLeft=r.left, origTop=r.top;
  el.style.position='fixed'; el.style.left=r.left+'px'; el.style.top=r.top+'px'; el.style.zIndex=999;
  el.setPointerCapture(e.pointerId);
  function move(ev){
   el.style.left=(ev.clientX-offX)+'px';
   el.style.top=(ev.clientY-offY)+'px';
  }
  function up(ev){
   el.removeEventListener('pointermove',move);
   el.removeEventListener('pointerup',up);
   const rr=el.getBoundingClientRect();
   const dropped=onDrop(rr, ev.clientX, ev.clientY);
   if(!dropped){
    el.style.transition='left .3s,top .3s';
    el.style.left=origLeft+'px'; el.style.top=origTop+'px';
    setTimeout(()=>{el.style.transition='';el.style.position='';el.style.left='';el.style.top='';el.style.zIndex='';},310);
   }
  }
  el.addEventListener('pointermove',move);
  el.addEventListener('pointerup',up);
 });
}

}
