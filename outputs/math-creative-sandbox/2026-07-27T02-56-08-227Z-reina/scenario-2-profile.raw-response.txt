function shuffle(arr){ let a=arr.slice(); for(let i=a.length-1;i>0;i--){const j=randInt(0,i);[a[i],a[j]]=[a[j],a[i]];} return a; }

function genGroupsQ(){
  const n = randInt(2,5), m = randInt(2,5);
  return {n, m, theme: pick(themes), product: n*m, built: new Array(n).fill(0)};
}
function genArrayQ(){
  const r = randInt(2,5), c = randInt(2,5);
  return {r, c, theme: pick(themes), product: r*c, curR:1, curC:1};
}
function genSkipQ(){
  const m = pick([2,3,4,5,10]);
  const terms = 5;
  const seq = [];
  for(let i=1;i<=terms;i++) seq.push(m*i);
  let blanks = shuffle([1,2,3,4]).slice(0,2).sort((a,b)=>a-b);
  return {m, seq, blanks, filled:{}};
}
function makeChoices(correct, spread){
  let set = new Set([correct]);
  while(set.size<3){
    let d = correct + pick([-spread,-spread*2,spread,spread*2, -1, 1]);
    if(d>0) set.add(d);
  }
  return shuffle([...set]);
}

const wordProblemBank = () => {

      text:`There are ${bags} baskets. Each basket holds ${each} apples 🍎. How many apples altogether?`,
      answer:bags*each
    },
    {

    }
  ];
};

  const q = state.groupsQ[state.groupsIdx];
  const allFull = q.built.every((b,i)=>b===q.m);
  let basketsHtml = '';

    </div>

  `;
}

function renderSkipIntro(){
  document.getElementById('app').innerHTML = `
    <h2>🐝 Skip-Count Trail</h2>

    <button class="bigbtn" onclick="goto(renderSkip)">Hop In! 🐾</button>
  `;

function renderSkip(){
  const q = state.skipQ[state.skipIdx];
  const nextBlankIdx = q.blanks.find(b => q.filled[b] === undefined);
  let seqHtml = q.seq.map((v,i)=>{
    if(q.blanks.includes(i)){
      if(q.filled[i] !== undefined) return `<div class="seq-box filled">${v}</div>`;
      if(i === nextBlankIdx) return `<div class="seq-box blank">?</div>`;
      return `<div class="seq-box blank">?</div>`;
    }
    return `<div class="seq-box">${v}</div>`;
  }).join('');
  let choicesHtml = '';
  if(nextBlankIdx !== undefined){
    const correct = q.seq[nextBlankIdx];
    const choices = makeChoices(correct, q.m);
    choicesHtml = `<p>Skip-counting by <b>${q.m}</b> — what goes on the next stone?</p>
      <div class="choice-row" id="choices">
        ${choices.map(c=>`<button class="choice-btn" onclick="answerSkip(${c},${correct},${nextBlankIdx})">${c}</button>`).join('')}
      </div>`;
  } else {
    choicesHtml = `<button class="bigbtn" onclick="nextSkip()">Next! 🐾</button>`;
  }
  document.getElementById('app').innerHTML = `
    <div class="progressbar"><div class="progressfill" style="width:${(state.skipIdx)/2*100}%"></div></div>
    <h2>🐝 Skip-Count Trail (by ${q.m}s)</h2>
    <div class="seq-row">${seqHtml}</div>
    ${choicesHtml}
    <div class="feedback" id="fb"></div>
  `;
}
function answerSkip(val, correct, idx){
  const q = state.skipQ[state.skipIdx];
  const ok = val === correct;
  markResult(ok);
  document.querySelectorAll('#choices .choice-btn').forEach(b=>b.disabled=true);
  if(ok){ q.filled[idx] = true; }
  document.getElementById('fb').textContent = ok ? "Right on the stone!" : `Oops! It's ${correct}.`;
  document.getElementById('fb').className = 'feedback ' + (ok?'good':'bad');
  companionSay(ok ? pick(["Hop hop hooray!","You're a skip-counting star!"])
                  : pick(["Nice try, keep hopping!","Almost there!"]), ok?'correct':'wrong');
  if(!ok) q.filled[idx] = true; // move on even if wrong, mark seen
  setTimeout(()=>goto(renderSkip), 1300);
}
function nextSkip(){
  state.skipIdx++;
  if(state.skipIdx < state.skipQ.length){ goto(renderSkip); }
  else { goto(renderWordIntro); }
}

  setTimeout(()=>{
    state.wordIdx++;
    if(state.wordIdx < state.wordQ.length){ goto(renderWord); }
    else { goto(renderBonusIntro); }
  }, 1700);
}

/* ================= BONUS ROUND ================= */
function renderBonusIntro(){
  document.getElementById('app').innerHTML = `
    <h2>✨ Bonus Round!</h2>
    <p>You unlocked Buggy's Bonus Round! Answer 3 super-fast multiplication puzzles!</p>
    <button class="bigbtn" onclick="startBonus()">Go!</button>
  `;
  companionSay("Ooh, bonus round! Let's gooo!", 'mastery');
}
function startBonus(){
  state.bonusQ = [1,2,3].map(()=>{
    const a = randInt(2,5), b = randInt(2,5);
    return {text:`${a} groups of ${b}`, answer:a*b, a, b};
  });
  state.bonusIdx = 0;
  goto(renderBonus);
}
function renderBonus(){
  const q = state.bonusQ[state.bonusIdx];
  const choices = makeChoices(q.answer, q.b);
  document.getElementById('app').innerHTML = `
    <h2>⚡ Bonus ${state.bonusIdx+1}/3</h2>
    <div class="storytext">${q.text}. How many in all?</div>
    <div class="choice-row" id="choices">
      ${choices.map(c=>`<button class="choice-btn" onclick="answerBonus(${c},${q.answer},this)">${c}</button>`).join('')}
    </div>
    <div class="feedback" id="fb"></div>
  `;
}
function answerBonus(val, correct, btn){
  const ok = val===correct;
  markResult(ok);
  document.querySelectorAll('#choices .choice-btn').forEach(b=>b.disabled=true);
  btn.classList.add(ok?'correct':'wrong');
  document.getElementById('fb').textContent = ok ? "Zoom! Correct!" : `It was ${correct}.`;
  document.getElementById('fb').className = 'feedback ' + (ok?'good':'bad');
  companionSay(ok?"Lightning fast!":"So close!", ok?'correct':'wrong');
  setTimeout(()=>{
    state.bonusIdx++;
    if(state.bonusIdx < state.bonusQ.length){ goto(renderBonus); }
    else { goto(renderEnd); }
  }, 1300);
}

function launchConfetti(){

  for(let i=0;i<26;i++){

    el.className='confetti';
    el.textContent = pick(emojis);
    el.style.left = Math.random()*100+'vw';
    el.style.animationDuration = (2+Math.random()*2)+'s';
    document.body.appendChild(el);
    setTimeout(()=>el.remove(), 4200);
  }
}

  `;
  launchConfetti();

}
