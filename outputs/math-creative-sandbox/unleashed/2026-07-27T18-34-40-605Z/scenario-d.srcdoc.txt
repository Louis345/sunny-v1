let actx;
function ac(){ if(!actx) actx = new (window.AudioContext||window.webkitAudioContext)(); return actx; }
function beep(freq, dur, type='sine', vol=0.15){
  try{
    const ctx = ac();
    const o = ctx.createOscillator();
    const g = ctx.createGain();
    o.type = type; o.frequency.value = freq;
    g.gain.value = vol;
    o.connect(g); g.connect(ctx.destination);
    o.start();
    g.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + dur);
    o.stop(ctx.currentTime + dur + 0.02);
  }catch(e){}
}
function soundClick(){ beep(500,0.08,'square',0.08); }
function soundCorrect(){ beep(600,0.12); setTimeout(()=>beep(800,0.15),110); setTimeout(()=>beep(1000,0.2),220); }

// ---------- Confetti ----------
const confettiCanvas = document.getElementById('confettiCanvas');
const cctx = confettiCanvas.getContext('2d');
let confettiParts = [];
function launchConfetti(){
  confettiParts = [];
  for(let i=0;i<150;i++){
    confettiParts.push({
      x: Math.random()*1365, y: -20-Math.random()*200,
      vx: (Math.random()-0.5)*4, vy: 2+Math.random()*4,
      size: 4+Math.random()*6, color: `hsl(${Math.random()*360},80%,60%)`,
      rot: Math.random()*360, vr: (Math.random()-0.5)*10
    });
  }
  requestAnimationFrame(confettiTick);
}
function confettiTick(){
  cctx.clearRect(0,0,1365,768);
  let alive = false;
  confettiParts.forEach(p=>{
    p.x += p.vx; p.y += p.vy; p.vy += 0.05; p.rot += p.vr;
    if(p.y < 800) alive = true;
    cctx.save();
    cctx.translate(p.x,p.y);
    cctx.rotate(p.rot*Math.PI/180);
    cctx.fillStyle = p.color;
    cctx.fillRect(-p.size/2,-p.size/2,p.size,p.size*0.6);
    cctx.restore();
  });
  if(alive) requestAnimationFrame(confettiTick);
  else cctx.clearRect(0,0,1365,768);
}

function changeHives(d){ g1_hives = Math.min(9,Math.max(1,g1_hives+d)); soundClick(); updateGroups(); }
function changeBees(d){ g1_bees = Math.min(9,Math.max(1,g1_bees+d)); soundClick(); updateGroups(); }
function updateGroups(){
  document.getElementById('hiveNum').textContent = g1_hives;
  document.getElementById('beeNum').textContent = g1_bees;
  document.getElementById('eqBox').textContent = `${g1_hives} × ${g1_bees} = ${g1_hives*g1_bees}`;
  drawHives();
  checkGroupChallenge();
}
function drawHives(){
  const field = document.getElementById('hiveField');
  field.innerHTML = '';
  for(let h=0; h<g1_hives; h++){
    const hive = document.createElement('div');
    hive.className = 'hive';
    for(let b=0; b<g1_bees; b++){
      const bee = document.createElement('div');
      bee.className = 'bee';
      hive.appendChild(bee);
    }
    field.appendChild(hive);
  }
  const total = g1_hives*g1_bees;
  const seq = [];
  for(let i=1;i<=g1_hives;i++) seq.push(i*g1_bees);
  document.getElementById('skipRow').innerHTML =
    `Skip-count by ${g1_bees}: <b>${seq.join(' , ')}</b> → Total = <b>${total}</b>`;
}
function newGroupChallenge(){
  const totals = [6,8,9,10,12,14,15,16,18,20,21,24,25,27,28,30,32,36,40,45];
  g1_target = totals[Math.floor(Math.random()*totals.length)];
  document.getElementById('challengeText').textContent = `Make a total of ${g1_target} using equal groups!`;
  say(`Can you arrange hives and bees so the total is exactly ${g1_target}?`);
  soundClick();
}
function checkGroupChallenge(){
  if(g1_target===null) return;
  if(g1_hives*g1_bees === g1_target){
    document.getElementById('challengeText').textContent = `🎉 Yes! ${g1_hives} × ${g1_bees} = ${g1_target}`;

    soundCorrect(); addStar(1);
    say("Great job! You just proved multiplication is repeated equal-group counting!");
    g1_target = null;
  }
}

// ===================== STAGE 3: SKIP-HOP =====================
let skipVal = 3, hopCount = 0;
let sh_target = null;
function renderSkipHop(){
  const div = document.createElement('div');
  div.className = 'panel card';
  div.id = 'skipPanel';
  div.innerHTML = `
    <div style="display:flex;justify-content:space-around;align-items:center;padding-top:20px;">
      <div class="stepper">
        <button class="stepbtn" onclick="changeSkip(-1)">-</button>
        <div class="steplabel">Hop size<br><span class="num" id="skipNum">${skipVal}</span></div>
        <button class="stepbtn" onclick="changeSkip(1)">+</button>
      </div>
      <button class="btn green" onclick="doHop()">🐝 Hop!</button>
      <button class="btn red" onclick="resetHop()">↺ Reset</button>
      <div class="eqbox" id="eqBoxSkip">${hopCount} × ${skipVal} = ${hopCount*skipVal}</div>
    </div>
    <div id="numberline"></div>
    <div style="text-align:center;margin-top:6px;">
      <button class="btn" onclick="newSkipChallenge()">🎯 New Challenge</button>
      <span id="skipChallengeText" style="font-size:17px;color:#a8660a;font-weight:bold;margin-left:14px;"></span>
    </div>
  `;
  stageEl.appendChild(div);
  drawNumberline();
  say("Skip-counting means hopping by the same amount each time. Every hop is one more group!");
}
function changeSkip(d){ skipVal=Math.min(9,Math.max(2,skipVal+d)); resetHop(); soundClick(); }
function doHop(){
  if(hopCount*skipVal + skipVal > 100) return;
  hopCount++;
  soundClick();
  document.getElementById('eqBoxSkip').textContent = `${hopCount} × ${skipVal} = ${hopCount*skipVal}`;
  drawNumberline(true);
  checkSkipChallenge();
}
function resetHop(){ hopCount=0; if(document.getElementById('eqBoxSkip')) document.getElementById('eqBoxSkip').textContent=`0 × ${skipVal} = 0`; drawNumberline(); }
function drawNumberline(animated){
  const nl = document.getElementById('numberline');
  nl.innerHTML='';
  const maxN = 40;
  const width = 1180;
  for(let i=0;i<=maxN;i+=2){
    const x = i/maxN*width;
    const tick = document.createElement('div');
    tick.className='tick'; tick.style.left=x+'px';
    nl.appendChild(tick);
    if(i%10===0){
      const lbl=document.createElement('div');
      lbl.className='ticklabel'; lbl.style.left=x+'px'; lbl.textContent=i;
      nl.appendChild(lbl);
    }
  }
  for(let i=1;i<=hopCount;i++){
    const val = i*skipVal;
    if(val>maxN) break;
    const x = val/maxN*width;
    const mark=document.createElement('div');
    mark.className='hopmark'; mark.style.left=x+'px'; mark.textContent=val;
    nl.appendChild(mark);
  }
  const bee = document.createElement('div');
  bee.id='beebug'; bee.innerHTML='🐝';
  bee.style.fontSize='30px';
  const curVal = Math.min(hopCount*skipVal, maxN);
  bee.style.left = (curVal/maxN*width)+'px';
  nl.appendChild(bee);
}
function newSkipChallenge(){
  const skip = Math.floor(Math.random()*7)+2;
  const hops = Math.floor(Math.random()*6)+2;
  sh_target = {skip, hops, val: skip*hops};
  document.getElementById('skipChallengeText').textContent =
    `Set hop size ${skip} and hop ${hops} times to land on ${skip*hops}!`;
  say(`Try it: hop by ${skip}, ${hops} times. Where do you land?`);
  soundClick();
}
function checkSkipChallenge(){
  if(!sh_target) return;
  if(skipVal===sh_target.skip && hopCount===sh_target.hops){
    document.getElementById('skipChallengeText').textContent = `🎉 Landed on ${sh_target.val}!`;
    document.getElementById('skipPanel').classList.add('flash-correct');
    setTimeout(()=>document.getElementById('skipPanel').classList.remove('flash-correct'),600);
    soundCorrect(); addStar(1);
    say("Awesome hopping! Skip-counting IS multiplication, one jump at a time.");
    sh_target=null;
  }
}

// ===================== STAGE 4: WORD PROBLEMS =====================
const wpTemplates = [
  {tpl:(g,p)=>`Buzz packs ${g} baskets with ${p} honey jars in each basket. How many jars in total?`, unit:'jars'},
  {tpl:(g,p)=>`There are ${g} flower beds with ${p} flowers growing equally in each bed. How many flowers altogether?`, unit:'flowers'},
  {tpl:(g,p)=>`The market has ${g} shelves, and each shelf holds ${p} honey pots. How many honey pots in all?`, unit:'pots'},
  {tpl:(g,p)=>`${g} bees each collect ${p} drops of nectar. How many drops did they collect together?`, unit:'drops'},
  {tpl:(g,p)=>`A crate has ${g} rows of apples with ${p} apples in every row. How many apples in the crate?`, unit:'apples'},
  {tpl:(g,p)=>`Buzz arranges ${g} trays with ${p} muffins on each tray. How many muffins total?`, unit:'muffins'},
];
let wp_current = null;
let wp_g = 2, wp_p = 2;
let wp_solved = 0;

function genWordProblem(){
  const t = wpTemplates[Math.floor(Math.random()*wpTemplates.length)];
  const g = Math.floor(Math.random()*7)+2;
  const p = Math.floor(Math.random()*7)+2;
  return {text: t.tpl(g,p), g, p, unit:t.unit, product:g*p};
}

function renderWordProblems(){
  wp_current = genWordProblem();
  wp_g = 2; wp_p = 2;
  const div = document.createElement('div');
  div.className='panel card';
  div.id='wpPanel';
  div.innerHTML = `
    <div class="wordbox" id="wpText">${wp_current.text}</div>
    <div style="display:flex;justify-content:center;gap:40px;align-items:center;">
      <div class="stepper">
        <button class="stepbtn" onclick="wpChange('g',-1)">-</button>
        <div class="steplabel">Groups<br><span class="num" id="wpGnum">${wp_g}</span></div>
        <button class="stepbtn" onclick="wpChange('g',1)">+</button>
      </div>
      <div class="stepper">
        <button class="stepbtn" onclick="wpChange('p',-1)">-</button>
        <div class="steplabel">Each Group<br><span class="num" id="wpPnum">${wp_p}</span></div>
        <button class="stepbtn" onclick="wpChange('p',1)">+</button>
      </div>
      <div class="eqbox" id="wpEq">${wp_g} × ${wp_p} = ${wp_g*wp_p}</div>
    </div>
    <div id="wpField" style="text-align:center;margin-top:6px;height:220px;overflow:auto;"></div>
    <div style="text-align:center;">
      <button class="btn green" onclick="checkWordProblem()">✅ Check Answer</button>
      <button class="btn" onclick="renderWordProblems()">🔄 New Problem</button>
      <span style="margin-left:20px;font-size:16px;color:#a8660a;font-weight:bold;">Solved: <span id="wpSolved">${wp_solved}</span></span>
    </div>
  `;
  stageEl.appendChild(div);
  drawWpField();
  say("Read carefully! Set the groups and amount-per-group to match the story, then check your equation.");
}
function wpChange(which, d){
  if(which==='g') wp_g=Math.min(9,Math.max(1,wp_g+d));
  else wp_p=Math.min(9,Math.max(1,wp_p+d));
  document.getElementById('wpGnum').textContent=wp_g;
  document.getElementById('wpPnum').textContent=wp_p;
  document.getElementById('wpEq').textContent = `${wp_g} × ${wp_p} = ${wp_g*wp_p}`;
  soundClick();
  drawWpField();
}
function drawWpField(){
  const field = document.getElementById('wpField');
  field.innerHTML='';
  for(let h=0;h<wp_g;h++){
    const hive=document.createElement('div');
    hive.className='hive';
    hive.style.background='radial-gradient(circle,#cdeeff,#8fd0f2)';
    for(let b=0;b<wp_p;b++){
      const bee=document.createElement('div');
      bee.className='bee';
      bee.style.background='#2b6a8f';
      hive.appendChild(bee);
    }
    field.appendChild(hive);
  }
}
function checkWordProblem(){
  const panel = document.getElementById('wpPanel');
  if(wp_g === wp_current.g && wp_p === wp_current.p){
    soundCorrect(); addStar(2); wp_solved++; streak++;
    document.getElementById('wpSolved').textContent = wp_solved;
    panel.classList.add('flash-correct');
    setTimeout(()=>panel.classList.remove('flash-correct'),600);
    say(`Correct! ${wp_current.g} × ${wp_current.p} = ${wp_current.product} ${wp_current.unit}. You matched the story perfectly!`);
    if(streak>0 && streak%5===0){
      launchConfetti();
      say(`🏆 Amazing! ${streak} in a row! You're a true Multiplication Master, just like Buzz!`);
    }
    setTimeout(renderWordProblems, 1600);
  } else if(wp_g*wp_p === wp_current.product){
    soundWrong();
    panel.classList.add('flash-wrong');
    setTimeout(()=>panel.classList.remove('flash-wrong'),600);
    say(`Close! Your total is right, but check the story again — is it ${wp_current.g} groups of ${wp_current.p}, or did you mix up groups and size?`);
  } else {
    streak = 0;
    soundWrong();
    panel.classList.add('flash-wrong');
    setTimeout(()=>panel.classList.remove('flash-wrong'),600);
    say(`Not quite. Re-read the problem: how many equal groups are there, and how many are in each group?`);
  }
}

// init
render();
</script>
