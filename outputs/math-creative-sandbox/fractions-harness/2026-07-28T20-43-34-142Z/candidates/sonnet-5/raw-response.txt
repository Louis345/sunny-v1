const AudioCtx = window.AudioContext || window.webkitAudioContext;
const actx = new AudioCtx();
function beep(freq, dur, type='sine', vol=0.2, delay=0){
  const t0 = actx.currentTime + delay;
  const osc = actx.createOscillator();
  const gain = actx.createGain();
  osc.type = type; osc.frequency.setValueAtTime(freq, t0);
  gain.gain.setValueAtTime(0, t0);
  gain.gain.linearRampToValueAtTime(vol, t0+0.02);
  gain.gain.exponentialRampToValueAtTime(0.001, t0+dur);
  osc.connect(gain).connect(actx.destination);
  osc.start(t0); osc.stop(t0+dur+0.05);
}

function animateVal(from, to, duration, onUpdate, onComplete, ease){
  ease = ease || (t=>1-Math.pow(1-t,3));
  const t0 = performance.now();
  function step(now){
    let t = Math.min(1,(now-t0)/duration);

// Generic function: draws shape with N equal wedge/segments, returns {svg, segEls(array), shadeCount(fn), isCorrectShape(fn)}
function buildShape(shapeType, n, opts={}){
  const w=460, h=380;
  const ns = "http://www.w3.org/2000/svg";
  const svg = document.createElementNS(ns,'svg');
  svg.setAttribute('width', w);
  svg.setAttribute('height', h);
  svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
  svg.style.filter = 'drop-shadow(0 6px 8px rgba(0,0,0,0.18))';

  const cx=w/2, cy=h/2;
  const segs = [];

  function addSeg(el, index){
    el.setAttribute('data-index', index);
    el.setAttribute('fill', '#fff2df');
    el.setAttribute('stroke', '#8a5a28');
    el.setAttribute('stroke-width', '3');
    el.style.cursor='pointer';
    svg.appendChild(el);
    segs.push(el);
  }

  if(shapeType==='circle'){
    const r = 160;
    for(let i=0;i<n;i++){
      const a0 = -Math.PI/2 + i*(2*Math.PI/n);
      const a1 = -Math.PI/2 + (i+1)*(2*Math.PI/n);
      const x0 = cx + r*Math.cos(a0), y0=cy+r*Math.sin(a0);
      const x1 = cx + r*Math.cos(a1), y1=cy+r*Math.sin(a1);
      const large = (a1-a0)>Math.PI?1:0;
      const path = document.createElementNS(ns,'path');
      path.setAttribute('d', `M${cx},${cy} L${x0},${y0} A${r},${r} 0 ${large} 1 ${x1},${y1} Z`);
      addSeg(path,i);
    }
  } else if(shapeType==='rect'){
    const rw=380, rh=220, x0=cx-rw/2, y0=cy-rh/2;
    for(let i=0;i<n;i++){
      const seg = document.createElementNS(ns,'rect');
      seg.setAttribute('x', x0 + i*(rw/n));
      seg.setAttribute('y', y0);
      seg.setAttribute('width', rw/n);
      seg.setAttribute('height', rh);
      addSeg(seg,i);
    }
  } else if(shapeType==='triangle'){
    // divide triangle into n equal-area vertical-ish strips using apex + base subdivisions
    const apex = {x:cx, y:cy-170};
    const baseL = {x:cx-190, y:cy+150};
    const baseR = {x:cx+190, y:cy+150};
    for(let i=0;i<n;i++){
      const t0=i/n, t1=(i+1)/n;
      const p0 = {x: baseL.x + (baseR.x-baseL.x)*t0, y: baseL.y};
      const p1 = {x: baseL.x + (baseR.x-baseL.x)*t1, y: baseL.y};
      const path = document.createElementNS(ns,'path');
      path.setAttribute('d', `M${apex.x},${apex.y} L${p0.x},${p0.y} L${p1.x},${p1.y} Z`);
      addSeg(path,i);
    }
  } else if(shapeType==='hexagon'){
    const r=170;
    const pts=[];
    for(let i=0;i<6;i++){
      const a = -Math.PI/2 + i*(Math.PI/3);
      pts.push({x:cx+r*Math.cos(a), y:cy+r*Math.sin(a)});
    }
    for(let i=0;i<n;i++){
      const a0 = -Math.PI/2 + i*(2*Math.PI/n);
      const a1 = -Math.PI/2 + (i+1)*(2*Math.PI/n);
      const x0 = cx + r*Math.cos(a0), y0=cy+r*Math.sin(a0);
      const x1 = cx + r*Math.cos(a1), y1=cy+r*Math.sin(a1);
      const path = document.createElementNS(ns,'path');
      path.setAttribute('d', `M${cx},${cy} L${x0},${y0} L${x1},${y1} Z`);
      addSeg(path,i);
    }
    // outer hex outline for style
    const outline = document.createElementNS(ns,'polygon');
    outline.setAttribute('points', pts.map(p=>`${p.x},${p.y}`).join(' '));
    outline.setAttribute('fill','none');
    outline.setAttribute('stroke','#8a5a28');
    outline.setAttribute('stroke-width','4');
    svg.appendChild(outline);
  }

  return {svg, segs};
}

    dot.setAttribute('cx', bbox.x+bbox.width/2);
    dot.setAttribute('cy', bbox.y+bbox.height/2);
    dot.setAttribute('r', 4);
    dot.setAttribute('fill', '#c98f52');
    dot.setAttribute('opacity', 0.55);
    dot.classList.add('guideDot');
    svg.appendChild(dot);
  });
}

function onSegTap(e){
  const seg = e.currentTarget;
  const idx = +seg.getAttribute('data-index');
  if(!activeShapeData) return;
  const data = activeShapeData;

  if(data.shaded.has(idx)){
    data.shaded.delete(idx);
    animateColor(seg, '#fff2df');
    return;
  }
  data.shaded.add(idx);
  const colorIdx = data.secondPhase ? 1 : 0;
  animateColor(seg, SHADE_COLORS[colorIdx % SHADE_COLORS.length]);
  beep(700,0.08,'sine',0.05);

  const wantCount = data.secondPhase ? 1 : 1; // unit fraction: always shade exactly 1 part
  if(data.shaded.size >= wantCount){
    setTimeout(()=> evaluateOrder(), 350);
  }
}

function animateColor(el, hexTo){
  // simple ease into fill using rAF interpolation of rgb
  const fromHex = el.getAttribute('fill');
  const from = hexToRgb(fromHex) || [255,242,223];
  const to = hexToRgb(hexTo);
  animateVal(0,1,300, t=>{
    const r = Math.round(from[0]+(to[0]-from[0])*t);
    const g = Math.round(from[1]+(to[1]-from[1])*t);
    const b = Math.round(from[2]+(to[2]-from[2])*t);
    el.setAttribute('fill', `rgb(${r},${g},${b})`);
  });
}
function hexToRgb(hex){
  if(!hex || hex[0]!=='#') return [255,255,255];
  const v = hex.substring(1);
  const bigint = parseInt(v.length===3 ? v.split('').map(c=>c+c).join('') : v, 16);
  return [(bigint>>16)&255, (bigint>>8)&255, bigint&255];
}

function wobbleShape(){
  const svg = shapeHost.querySelector('svg');
  if(!svg) return;
  let t=0;
  const dur=500;
  const t0=performance.now();
  function step(now){
    const p=(now-t0)/dur;
    if(p>=1){ svg.style.transform=''; return; }
    const x = Math.sin(p*Math.PI*6)*10*(1-p);
    svg.style.transform = `translateX(${x}px)`;
    requestAnimationFrame(step);
  }
  requestAnimationFrame(step);
}

function bounceShape(cb){
  const svg = shapeHost.querySelector('svg');
  if(!svg){ if(cb) cb(); return; }
  const dur=450; const t0=performance.now();
  function step(now){
    const p=Math.min(1,(now-t0)/dur);
    const s = 1 + Math.sin(p*Math.PI)*0.12;
    svg.style.transform = `scale(${s})`;
    if(p<1) requestAnimationFrame(step); else { svg.style.transform=''; if(cb) cb(); }
  }
  requestAnimationFrame(step);
}

function evaluateOrder(){
  const data = activeShapeData;
  if(!data) return;
  const correct = data.shaded.size === 1; // exactly one unit shaded, no more no less
  if(correct){
    onOrderCorrect();
  } else {
    onOrderWrong();
  }
}

function onOrderCorrect(){
  const data = activeShapeData;
  chimeCorrect();
  bounceShape(()=>{
    if(data.isPractice){
      say("Great practice! Want to try a real order now?");
      flashText('✓', '#4caf7a');
      setTimeout(()=>newOrder(false), 900);
      return;
    }
    if(data.doubleOrder && !data.secondPhase){
      // move to second fraction on same shape (fresh shape though, re-divided by secondDenom)
      flashText('✓','#4caf7a');
      data.secondPhase = true;
      const shapeType = data.shapeType;
      const denom2 = data.secondDenom;
      const {svg, segs} = buildShape(shapeType, denom2);
      shapeHost.innerHTML='';
      shapeHost.appendChild(svg);
      segs.forEach(seg=> seg.addEventListener('pointerdown', onSegTap));
      activeShapeData = {segs, n:denom2, shaded:new Set(), targetCount:1, shapeType, isDemo:false, isPractice:false, doubleOrder:true, secondPhase:true, denom:denom2};
      promptTarget.textContent = `1/${denom2}`;
      promptWords.textContent = `(${WORDS[1]} ${ORDINAL_WORDS[denom2]||'half'})`;
      say(`Nice! Now give ${state.belt[0]?.name||'the customer'} the second slice: one ${ORDINAL_WORDS[denom2]||'half'}!`);
      if(currentStage().guidesAlways || state.gridForced) showGuideDots();
      return;
    }

    flashText('✓ Yum!', '#4caf7a');
    awardCrumb();
    advanceBelt(true);
    state.correctInARow++;
    if(currentStage().judge && state.correctInARow>0 && state.correctInARow%3===0){
      say("Three in a row! You've earned a Fraction Judge ceremony! 👩‍⚖️✨");
      celebrateJudge();
    } else {
      say(pickPraise());
    }
    checkStageAdvance();
    setTimeout(()=>newOrder(false), 900);
  });
}

function spawnConfetti(n=80){
  for(let i=0;i<n;i++){
    confettiParticles.push({
      x: Math.random()*1365, y: -20-Math.random()*200,
      vx: (Math.random()-0.5)*2, vy: 2+Math.random()*3,
      size: 4+Math.random()*6,
      color: SHADE_COLORS[Math.floor(Math.random()*SHADE_COLORS.length)],
      rot: Math.random()*360, vr: (Math.random()-0.5)*10
    });
  }
  if(!confettiRunning) runConfetti();
}
let confettiRunning=false;
function runConfetti(){
  confettiRunning=true;
  cctx.clearRect(0,0,1365,768);
  confettiParticles.forEach(p=>{
    p.x+=p.vx; p.y+=p.vy; p.rot+=p.vr;
    cctx.save();
    cctx.translate(p.x,p.y);
    cctx.rotate(p.rot*Math.PI/180);
    cctx.fillStyle=p.color;
    cctx.fillRect(-p.size/2,-p.size/2,p.size,p.size);
    cctx.restore();
  });
  confettiParticles = confettiParticles.filter(p=>p.y<820);
  if(confettiParticles.length>0){
    requestAnimationFrame(runConfetti);
  } else {
    confettiRunning=false;
    cctx.clearRect(0,0,1365,768);
  }
}

function runComparisonChallenge(){
  document.getElementById('compareArea').style.display='flex';
  document.getElementById('stageArea').style.display='none';

  let dA, dB;
  do {
    dA = pickDenomForStage(); dB = pickDenomForStage();
  } while(dA===dB);

  const shapeTypeA = pickShapeForStage();
  const shapeTypeB = shapeTypeA; // keep same shape type for fair visual comparison

  const cardA = document.getElementById('cardA');
  const cardB = document.getElementById('cardB');
  cardA.innerHTML = `<div>1/${dA}</div>`;
  cardB.innerHTML = `<div>1/${dB}</div>`;

  const {svg:svgA} = buildShape(shapeTypeA, dA);
  const {svg:svgB} = buildShape(shapeTypeB, dB);
  svgA.setAttribute('width', 260); svgA.setAttribute('height',220);
  svgB.setAttribute('width', 260); svgB.setAttribute('height',220);
  // pre-shade one segment each to visualize
  svgA.querySelector('[data-index="0"]').setAttribute('fill', SHADE_COLORS[0]);
  svgB.querySelector('[data-index="0"]').setAttribute('fill', SHADE_COLORS[2]);

  cardA.appendChild(svgA);
  cardB.appendChild(svgB);

  const btnA = document.createElement('button');
  btnA.textContent = 'This one is BIGGER';
  const btnB = document.createElement('button');
  btnB.textContent = 'This one is BIGGER';
  cardA.appendChild(btnA);
  cardB.appendChild(btnB);

  say(`Fraction Judge time! Which slice is bigger: 1/${dA} or 1/${dB}?`);

  function resolve(pickedIsA){
    const correctIsA = dA < dB; // smaller denominator = bigger unit fraction piece
    const correct = pickedIsA === correctIsA;
    if(correct){
      chimeCorrect();
      flashText('✓ Correct!', '#4caf7a');
      awardCrumb();
      state.correctInARow++;
      if(currentStage().judge && state.correctInARow>0 && state.correctInARow%3===0){
        celebrateJudge();
        say("Three correct judgments in a row — amazing Fraction Judge work!");
      } else {
        say("Yes! Fewer, bigger pieces beat more, smaller pieces sharing the same whole.");
      }
    } else {
      boopWrong();
      flashText('Hmm, try again!', '#e07b39');
      say(`Actually, 1/${Math.min(dA,dB)} is bigger — fewer slices means each one is chunkier!`);
      state.correctInARow = 0;
    }
    setTimeout(()=>{
      document.getElementById('compareArea').style.display='none';
      document.getElementById('stageArea').style.display='flex';
      checkStageAdvance();
      newOrder(false);
    }, 1600);
  }
  btnA.onclick = ()=>resolve(true);
  btnB.onclick = ()=>resolve(false);
}
