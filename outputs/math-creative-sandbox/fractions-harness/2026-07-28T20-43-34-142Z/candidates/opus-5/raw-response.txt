const Audio_ = (()=>{
  let ctx=null, master=null, ambGain=null, on=true, started=false;
  function init(){
    if(ctx) return;
    ctx = new (window.AudioContext||window.webkitAudioContext)();
    master = ctx.createGain(); master.gain.value=0.75; master.connect(ctx.destination);
    ambGain = ctx.createGain(); ambGain.gain.value=0.0; ambGain.connect(master);
    // oven hum
    const hum = ctx.createOscillator(); hum.type='sine'; hum.frequency.value=58;
    const humG = ctx.createGain(); humG.gain.value=0.055;
    const lfo = ctx.createOscillator(); lfo.frequency.value=0.16;
    const lfoG = ctx.createGain(); lfoG.gain.value=0.02;
    lfo.connect(lfoG); lfoG.connect(humG.gain);
    hum.connect(humG); humG.connect(ambGain); hum.start(); lfo.start();
    // soft noise "kitchen air"
    const bufSize = ctx.sampleRate*2;
    const buf = ctx.createBuffer(1,bufSize,ctx.sampleRate);
    const d = buf.getChannelData(0);
    for(let i=0;i<bufSize;i++) d[i]=(Math.random()*2-1)*0.35;
    const noise = ctx.createBufferSource(); noise.buffer=buf; noise.loop=true;
    const nf = ctx.createBiquadFilter(); nf.type='lowpass'; nf.frequency.value=420;
    const ng = ctx.createGain(); ng.gain.value=0.035;
    noise.connect(nf); nf.connect(ng); ng.connect(ambGain); noise.start();
    gsapSafeRamp();
    // occasional clatter
    setInterval(()=>{ if(on && Math.random()<0.35) clatter(); }, 5200);
    started=true;
  }
  function gsapSafeRamp(){ ambGain.gain.setTargetAtTime(on?0.9:0, ctx.currentTime, 1.2); }
  function tone(freq, t0, dur, type='sine', vol=0.25, glide=null){
    if(!ctx||!on) return;
    const o=ctx.createOscillator(), g=ctx.createGain();
    o.type=type; o.frequency.setValueAtTime(freq, ctx.currentTime+t0);
    if(glide) o.frequency.exponentialRampToValueAtTime(glide, ctx.currentTime+t0+dur);
    g.gain.setValueAtTime(0.0001, ctx.currentTime+t0);
    g.gain.exponentialRampToValueAtTime(vol, ctx.currentTime+t0+0.015);
    g.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime+t0+dur);
    o.connect(g); g.connect(master); o.start(ctx.currentTime+t0); o.stop(ctx.currentTime+t0+dur+0.05);
  }
  function clatter(){
    if(!ctx) return;
    tone(300+Math.random()*400, 0, 0.07, 'triangle', 0.03);
    tone(900+Math.random()*500, 0.05, 0.05, 'square', 0.012);
  }
  return {
    unlock(){ init(); if(ctx.state==='suspended') ctx.resume(); },
    toggle(){ on=!on; if(ctx) gsapSafeRamp(); return on; },
    isOn(){return on;},
    chime(){ // warm marimba
      tone(523.25,0,0.5,'sine',0.30); tone(659.25,0.06,0.5,'sine',0.24); tone(783.99,0.12,0.6,'sine',0.20);
      tone(1046.5,0.18,0.4,'triangle',0.08);
    },
    tap(){ tone(660,0,0.12,'sine',0.16); },
    untap(){ tone(440,0,0.10,'sine',0.10); },
    boop(){ tone(200,0,0.22,'sine',0.22,140); },
    cheer(){
      [0,.09,.18,.28,.40].forEach((t,i)=>tone([523,659,784,1046,1318][i],t,0.55,'sine',0.22));
      // applause-ish noise burst
      if(!ctx||!on) return;
      const b=ctx.createBuffer(1,ctx.sampleRate*1.2,ctx.sampleRate); const d=b.getChannelData(0);
      for(let i=0;i<d.length;i++){ d[i]=(Math.random()*2-1)*Math.pow(1-i/d.length,1.6)*0.5;}
      const s=ctx.createBufferSource(); s.buffer=b;
      const f=ctx.createBiquadFilter(); f.type='bandpass'; f.frequency.value=1800; f.Q.value=0.7;
      const g=ctx.createGain(); g.gain.value=0.18;
      s.connect(f); f.connect(g); g.connect(master); s.start();
    },
    jingle(){ [523,587,659,784,880,1046].forEach((f,i)=>tone(f,i*0.085,0.35,'triangle',0.18)); }
  };
})();

/* =========================================================
   SHAPE GEOMETRY — returns array of SVG path strings for n equal parts
   ========================================================= */
const SVGNS='http://www.w3.org/2000/svg';
function poly(pts){ return 'M'+pts.map(p=>p[0].toFixed(2)+','+p[1].toFixed(2)).join(' L')+' Z'; }

function shapeParts(kind, n, cxp, cyp, R){
  const parts=[];
  if(kind==='circle' || kind==='pie'){
    for(let i=0;i<n;i++){
      const a0=-Math.PI/2 + i*2*Math.PI/n, a1=-Math.PI/2+(i+1)*2*Math.PI/n;
      const x0=cxp+R*Math.cos(a0), y0=cyp+R*Math.sin(a0);
      const x1=cxp+R*Math.cos(a1), y1=cyp+R*Math.sin(a1);
      const large = (2*Math.PI/n)>Math.PI?1:0;
      if(n===1) parts.push(`M ${cxp-R},${cyp} a ${R},${R} 0 1,0 ${2*R},0 a ${R},${R} 0 1,0 ${-2*R},0 Z`);
      else parts.push(`M ${cxp},${cyp} L ${x0.toFixed(2)},${y0.toFixed(2)} A ${R},${R} 0 ${large},1 ${x1.toFixed(2)},${y1.toFixed(2)} Z`);
    }
  } else if(kind==='rect'){
    const W=R*2.15, H=R*1.35;
    const x=cxp-W/2, y=cyp-H/2;
    // vertical strips (or 2x2 grid for 4, 2x3 for 6, 2x4 for 8)
    let cols=n, rows=1;
    if(n===4){cols=2;rows=2;} if(n===6){cols=3;rows=2;} if(n===8){cols=4;rows=2;}
    const w=W/cols, h=H/rows;
    for(let r=0;r<rows;r++) for(let c=0;c<cols;c++)
      parts.push(poly([[x+c*w,y+r*h],[x+(c+1)*w,y+r*h],[x+(c+1)*w,y+(r+1)*h],[x+c*w,y+(r+1)*h]]));
  } else if(kind==='tri'){
    // equilateral triangle sliced from apex to base -> equal-area wedges (equal base segments)
    const h=R*1.75, w=R*2.0;
    const apex=[cxp, cyp-h/2], bl=[cxp-w/2, cyp+h/2], br=[cxp+w/2, cyp+h/2];
    for(let i=0;i<n;i++){
      const t0=i/n, t1=(i+1)/n;
      const p0=[bl[0]+(br[0]-bl[0])*t0, bl[1]];
      const p1=[bl[0]+(br[0]-bl[0])*t1, bl[1]];
      parts.push(poly([apex,p0,p1]));
    }
  } else if(kind==='hex'){
    // regular hexagon: wedge from center to each equal arc of perimeter -> equal areas
    const verts=[];
    for(let i=0;i<6;i++){ const a=-Math.PI/2+i*Math.PI/3; verts.push([cxp+R*Math.cos(a),cyp+R*Math.sin(a)]); }
    // perimeter param
    function ptAt(t){ // t in [0,6)
      const i=Math.floor(t)%6, f=t-Math.floor(t);
      const a=verts[i], b=verts[(i+1)%6];
      return [a[0]+(b[0]-a[0])*f, a[1]+(b[1]-a[1])*f];
    }
    for(let i=0;i<n;i++){
      const t0=i*6/n, t1=(i+1)*6/n;
      const pts=[[cxp,cyp], ptAt(t0)];
      for(let k=Math.ceil(t0+1e-9);k<t1-1e-9;k++) pts.push(verts[k%6]);
      pts.push(ptAt(t1));
      parts.push(poly(pts));
    }
  }
  return parts;
}

/* =========================================================
   TREAT WIDGET — interactive sliceable shape
   ========================================================= */
function makeTreat(opts){
  // opts: kind, n, cx, cy, R, showGuides, interactive, frostColor
  const svg=document.createElementNS(SVGNS,'svg');
  const W=opts.W||520, H=opts.H||360;
  svg.setAttribute('width',W); svg.setAttribute('height',H);
  svg.style.position='absolute';
  svg.style.left=(opts.left!==undefined?opts.left:0)+'px';
  svg.style.top =(opts.top!==undefined?opts.top:0)+'px';

  const g=document.createElementNS(SVGNS,'g');
  svg.appendChild(g);

  // shadow blob
  const sh=document.createElementNS(SVGNS,'ellipse');
  sh.setAttribute('cx',opts.cx); sh.setAttribute('cy',opts.cy+opts.R*1.05);
  sh.setAttribute('rx',opts.R*0.95); sh.setAttribute('ry',opts.R*0.16);
  sh.setAttribute('fill','rgba(90,50,20,.18)');
  g.appendChild(sh);

  const paths=shapeParts(opts.kind,opts.n,opts.cx,opts.cy,opts.R);
  const outlinePaths=shapeParts(opts.kind,1,opts.cx,opts.cy,opts.R);
  const frost = opts.frostColor||FROST[Math.floor(Math.random()*FROST.length)];

  const slices=[];
  paths.forEach((d,i)=>{
    const p=document.createElementNS(SVGNS,'path');
    p.setAttribute('d',d);
    p.setAttribute('fill',DOUGH[opts.kind]);
    p.setAttribute('stroke', opts.showGuides? '#c9945c' : 'rgba(201,148,92,0)');
    p.setAttribute('stroke-width', 3);
    p.setAttribute('stroke-dasharray', opts.showGuides? '9 7':'0');
    p.setAttribute('stroke-linejoin','round');
    if(opts.interactive) p.classList.add('slice');
    p.dataset.i=i; p.dataset.on='0';
    g.appendChild(p); slices.push(p);
  });

  // thick outer outline
  const out=document.createElementNS(SVGNS,'path');
  out.setAttribute('d',outlinePaths[0]);
  out.setAttribute('fill','none'); out.setAttribute('stroke','#a5643a');
  out.setAttribute('stroke-width',7); out.setAttribute('stroke-linejoin','round');
  out.style.pointerEvents='none';
  g.appendChild(out);

  const api={
    svg, g, slices, frost, kind:opts.kind, n:opts.n,
    shaded(){ return slices.filter(s=>s.dataset.on==='1').length; },
    setGuides(on){
      slices.forEach(s=>{
        gsap.to(s,{attr:{'stroke-opacity':on?1:0},duration:.35});
        s.setAttribute('stroke', '#c9945c');
        s.setAttribute('stroke-dasharray', on?'9 7':'9 7');
        s.setAttribute('stroke-opacity', on?1:0);
      });
    },
    shade(i,on,silent){
      const p=slices[i]; if(!p) return;
      p.dataset.on=on?'1':'0';
      gsap.to(p,{attr:{fill: on? frost : DOUGH[opts.kind]},duration:.30,ease:'sine.out'});
      if(on && !silent){
        // sparkle dots
        gsap.fromTo(p,{attr:{'fill-opacity':.4}},{attr:{'fill-opacity':1},duration:.3});
      }
    },
    clearAll(){ slices.forEach((s,i)=>api.shade(i,false,true)); },
    wobble(){
      gsap.fromTo(g,{rotation:0,transformOrigin:`${opts.cx}px ${opts.cy}px`},
        {keyframes:[{rotation:-5,duration:.09},{rotation:5,duration:.13},{rotation:-3.5,duration:.11},{rotation:2,duration:.09},{rotation:0,duration:.09}],ease:'sine.inOut'});
    },
    bounce(){
      gsap.fromTo(g,{y:0},{keyframes:[{y:-26,duration:.22,ease:'power2.out'},{y:0,duration:.34,ease:'bounce.out'}]});
    },
    sparkle(){
      for(let k=0;k<10;k++){
        const st=document.createElementNS(SVGNS,'text');
        st.textContent='✦';
        st.setAttribute('x',opts.cx+(Math.random()-.5)*opts.R*2);
        st.setAttribute('y',opts.cy+(Math.random()-.5)*opts.R*2);
        st.setAttribute('fill','#fff3b0'); st.setAttribute('font-size',10+Math.random()*16);
        g.appendChild(st);
        gsap.fromTo(st,{opacity:0,scale:0},{opacity:1,scale:1,duration:.25,onComplete(){
          gsap.to(st,{opacity:0,y:-30,duration:.6,onComplete(){st.remove();}});
        }});
      }
    },
    slideOut(dir=1,cb){
      gsap.to(svg,{x:dir*760, y:60, rotation:dir*14, duration:.9, ease:'power2.in',onComplete:cb});
    },
    slideIn(from=-700){
      gsap.fromTo(svg,{x:from,y:40,rotation:-8,opacity:0},{x:0,y:0,rotation:0,opacity:1,duration:.75/G.beltSpeed,ease:'back.out(1.1)'});
    },
    driftBack(cb){
      gsap.to(svg,{x:-90,rotation:-7,duration:.4,yoyo:true,repeat:1,ease:'sine.inOut',onComplete:cb});
    },
    steam(){
      for(let k=0;k<3;k++){
        const c=document.createElementNS(SVGNS,'path');
        const x=opts.cx-30+k*30;
        c.setAttribute('d',`M${x},${opts.cy-opts.R-6} q10,-16 0,-32 q-10,-16 0,-30`);
        c.setAttribute('stroke','rgba(255,255,255,.85)'); c.setAttribute('stroke-width',7);
        c.setAttribute('fill','none'); c.setAttribute('stroke-linecap','round');
        g.appendChild(c);
        gsap.fromTo(c,{opacity:0,y:14},{opacity:.9,y:-6,duration:.45,delay:k*.09,onComplete(){
          gsap.to(c,{opacity:0,y:-34,duration:.8,onComplete:()=>c.remove()});
        }});
      }
    }
  };
  if(opts.showGuides===false) api.setGuides(false);
  return api;
}

function shuffle(a){for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];}return a;}
function pick(a){return a[Math.floor(Math.random()*a.length)];}

/* =========================================================
   TASK GENERATION
   ========================================================= */
function genTask(){
  const st=STAGES[G.stage].key;
  const i=G.taskInStage;
  if(st==='warm'){
    const kinds=['circle','rect','circle','rect','circle'];
    const dens =[3,2,3,4,2];
    return {type:'shade', kind:kinds[i%5], n:dens[i%5], target:1, guides:true, phrase: Math.random()<0.5?'word':'num'};
  }
  if(st==='rising'){
    if(i===2 || i===5) return genCompare([2,3,4]);
    const kind=pick(['rect','circle']); const n=pick([2,3,4]);
    return {type:'shade',kind,n,target:1,guides:false,phrase:Math.random()<.5?'word':'num'};
  }
  if(st==='master'){
    if(i===2||i===5) return genCompare([2,3,4,5,6,8]);
    if(i===6) return genCompare([3,4,6,8]);
    const kind=pick(['rect','circle','tri','hex']);
    let n = kind==='hex'? pick([2,3,6]) : pick([2,3,4,5,6,8]);
    if(kind==='tri') n=pick([2,3,4,6]);
    return {type:'shade',kind,n,target:1,guides:false,phrase:Math.random()<.5?'word':'num'};
  }
  // grand
  const roll=Math.random();
  if(i===3 || i===7) return genBetween();
  if(roll<0.28) return genCompare([2,3,4,5,6,8]);
  if(roll<0.48) return genDouble();
  const kind=pick(['rect','circle','tri','hex']);
  let n = kind==='hex'? pick([2,3,6]) : pick([2,3,4,5,6,8]);
  if(kind==='tri') n=pick([2,3,4,6]);
  return {type:'shade',kind,n,target:1,guides:false,phrase:Math.random()<.5?'word':'num',timed:true};
}
function genCompare(pool){
  let a=pick(pool), b=pick(pool); let guard=0;
  while(b===a && guard++<20) b=pick(pool);
  if(b===a) b = a===2?4:2;
  return {type:'compare', a, b,
    kindA: pick(['circle','rect','hex','tri']), kindB: pick(['circle','rect','hex','tri']),
    ask: Math.random()<0.75?'bigger':'smaller'};
}
function genDouble(){
  const kA=pick(['rect','circle']), kB=pick(['circle','tri','hex']);
  let nA=pick([2,3,4]), nB=pick([4,5,6,8]);
  if(kB==='hex') nB=pick([3,6]);
  if(kB==='tri') nB=pick([4,6]);
  return {type:'double', a:{kind:kA,n:nA}, b:{kind:kB,n:nB}};
}
function genBetween(){
  // find lo>hi denominators: piece bigger than 1/lo but smaller than 1/hi  (lo>hi)
  const sets=[[5,3,4],[6,3,4],[8,4,5],[8,3,[4,5,6]],[6,2,[3,4,5]],[5,2,[3,4]]];
  const s=pick(sets);
  const lo=s[0], hi=s[1];
  const answers = Array.isArray(s[2])? s[2] : [s[2]];
  const kind=pick(['circle','rect']);
  // options
  const opts=new Set(answers);
  while(opts.size<4){ const c=2+Math.floor(Math.random()*7); if(!(c>hi&&c<lo)) opts.add(c); if(opts.size>=4)break; }
  return {type:'between', lo, hi, answers, kind, options: shuffle([...opts]).slice(0,4)};
}

function checkShade(){

/* =========================================================
   TIMER
   ========================================================= */
function startTimer(sec){
  timerBar.style.display='block';
  let t0=performance.now();
  timerFill.style.width='100%';
  function step(now){
    const el=(now-t0)/1000;
    const p=Math.max(0,1-el/sec);
    timerFill.style.width=(p*100)+'%';
    timerFill.style.background = p>0.4? 'linear-gradient(90deg,#8fd06a,#e8c04a)':'linear-gradient(90deg,#e8a04a,#e07a5a)';
    if(p<=0){ G.timerId=null; timeUp(); return; }
    G.timerId=requestAnimationFrame(step);
  }
  G.timerId=requestAnimationFrame(step);
}
function timeUp(){
  Audio_.boop();
  toast('Belt moved on! Try the next one 🤍',false);
  customerSad();
  const tr=G.current&&G.current.treat; if(tr) tr.wobble();
  setTimeout(nextTask,900);
}

function success(treat){

  awardCrumb();

  setTimeout(()=>{ treat.slideOut(1, nextTask); }, 750);
}
function successGeneric(msg, extra, treats){
  if(G.timerId){cancelAnimationFrame(G.timerId);G.timerId=null;}
  Audio_.chime(); toast(msg);
  awardCrumb(); customerHappy();
  actionRow.querySelectorAll('button').forEach(b=>b.disabled=true);

  setTimeout(()=>{ if(treats) treats.forEach((t,i)=>t.slideOut(i?1:-1)); nextTask(); }, 1250);

}
function failGeneric(why, treats){

const FP={kind:'circle',n:4};
document.getElementById('btnFreeplay').onclick=()=>{ hide('ovFinal'); openFree(); };
document.getElementById('btnCloseFree').onclick=()=>{ hide('ovFree'); show('ovFinal'); };

function openFree(){
  const sh=document.getElementById('fpShapes'); sh.innerHTML='';
  ['rect','circle','tri','hex'].forEach(k=>{
    const c=document.createElement('div'); c.className='chip'+(FP.kind===k?' sel':'');
    c.innerHTML=`${SHAPE_EMOJI[k]} ${SHAPE_NAME[k]}`;
    c.onclick=()=>{FP.kind=k; if(k==='hex'&&![2,3,6].includes(FP.n))FP.n=6; if(k==='tri'&&![2,3,4,6].includes(FP.n))FP.n=4; openFree();};
    sh.appendChild(c);
  });
  const dn=document.getElementById('fpDenoms'); dn.innerHTML='';
  let allowed=[2,3,4,5,6,8];
  if(FP.kind==='hex') allowed=[2,3,6];
  if(FP.kind==='tri') allowed=[2,3,4,6];
  allowed.forEach(n=>{
    const c=document.createElement('div'); c.className='chip'+(FP.n===n?' sel':'');
    c.textContent='1/'+n;
    c.onclick=()=>{FP.n=n; openFree();};
    dn.appendChild(c);
  });
  const pv=document.getElementById('fpPreview'); pv.innerHTML='';
  pv.style.cssText='position:relative;width:300px;height:220px';
  const tr=makeTreat({kind:FP.kind,n:FP.n,cx:150,cy:105,R:88,W:300,H:220,left:0,top:0,showGuides:true,interactive:false});
  pv.appendChild(tr.svg);
  document.getElementById('fpMsg').textContent='';
  document.getElementById('btnAskRoly').onclick=()=>{
    Audio_.unlock();
    // Roly attempts: 70% correct, else shades wrong count -> she checks
    const correct = Math.random()<0.65;
    const wrongCount = correct?1:(Math.random()<0.5? Math.min(FP.n,2) : (FP.n>2?FP.n-1:2));
    tr.clearAll();
    let idx=0;
    const count = correct?1:wrongCount;
    const iv=setInterval(()=>{ tr.shade(idx,true); Audio_.tap(); idx++; if(idx>=count) clearInterval(iv); },260);
    setTimeout(()=>{
      const msg=document.getElementById('fpMsg');
      msg.innerHTML=`Roly shaded <b>${count}</b> piece${count>1?'s':''}. Is that 1/${FP.n}?
        <button class="chip" id="fpYes">✅ Correct</button> <button class="chip" id="fpNo">❌ Wrong</button>`;
      document.getElementById('fpYes').onclick=()=>judgeRoly(true,correct,tr,count);
      document.getElementById('fpNo').onclick=()=>judgeRoly(false,correct,tr,count);
    }, 260*count+300);
  };
  show('ovFree');
}
function judgeRoly(said, actual, tr, count){
  const msg=document.getElementById('fpMsg');
  if(said===actual){
    Audio_.chime(); tr.sparkle(); confettiBurst(50,682,300);
    msg.innerHTML = actual
      ? `🎉 Right! One shaded piece of ${FP.n} equal parts <b>is</b> 1/${FP.n}. Great teaching, Baker!`
      : `🎉 Right! ${count} pieces is <b>${count}/${FP.n}</b>, not 1/${FP.n}. You caught Roly's mistake!`;
    setTimeout(()=>{ tr.clearAll(); document.getElementById('fpMsg').innerHTML='Design another order!'; },2400);
  } else {
    Audio_.boop(); tr.wobble();
    msg.innerHTML = actual
      ? `Hmm — Roly shaded exactly 1 of ${FP.n} equal parts. That <b>is</b> 1/${FP.n}!`
      : `Look again — he shaded ${count} pieces. 1/${FP.n} means only <b>ONE</b> piece.`;
  }
}
