const STAGE_TITLES = [

function buildProgressDots(){
  progressEl.innerHTML = '';
  for(let i=0;i<6;i++){
    const d = document.createElement('div');
    d.className = 'dot' + (i===stage?' active':'') + (achieved[i] && i!==stage ? ' done':'');
    d.addEventListener('click', ()=>{ if(i<=stage || achieved[i-1]) goTo(i); });
    progressEl.appendChild(d);
  }
}

function updateNav(){
  backBtn.disabled = stage===0;
  nextBtn.disabled = !achieved[stage] || stage===5;
  nextBtn.textContent = stage===5 ? "🎉 Done" : "Next ➡";
  stageLabel.textContent = "Step " + (stage+1) + " of 6";
}

function goTo(i){
  stage = i;
  render();
}

backBtn.addEventListener('click', ()=>{ if(stage>0) goTo(stage-1); });
nextBtn.addEventListener('click', ()=>{ if(stage<5 && achieved[stage]) goTo(stage+1); });

function render(){
  buildProgressDots();
  updateNav();
  main.innerHTML = '';
  const div = document.createElement('div');
  div.className = 'stage visible fadeGrow';
  main.appendChild(div);
  RENDERERS[stage](div);
}

function renderWelcome(container){

    groups++;
    if(groups>5){ groups=1; }
    draw();
  }, 1400);
  // stop timer if we leave stage (cleanup on next render via garbage collection is fine since element removed)
}

/* ===================== STAGE 2: SKIP COUNTING ===================== */
function renderSkip(container){
  const step = 4, targetJumps = 5, targetTotal = step*targetJumps;
  container.innerHTML = `
    <h2 class="stageTitle">${STAGE_TITLES[2]}</h2>
    <p class="instructions">Each jump on the number line is worth <b>${step}</b>. Keep jumping until you reach <b>${targetTotal}</b>,
    then tell us how many jumps it took — that's the same as ${step} × <i>jumps</i>!</p>
    <div id="numberLine"></div>
    <div class="equationBox" id="skipEq">Jumps: 0 &nbsp;|&nbsp; Total: 0</div>
    <button class="btn" id="jumpBtn">🐇 Jump +${step}</button>
    <button class="btn secondary small" id="resetJump" style="margin-top:10px;">↺ Reset</button>
    <div id="skipQuestion" style="margin-top:14px;"></div>
    <div id="feedback"></div>
  `;
  const lineDiv = container.querySelector('#numberLine');
  const eq = container.querySelector('#skipEq');
  const feedback = container.querySelector('#feedback');
  const qDiv = container.querySelector('#skipQuestion');
  let jumps = 0;
  const maxVal = 50;
  const width = 1050;

  function posFor(val){ return (val/maxVal)*width; }

  function buildLine(){
    lineDiv.innerHTML = '<div class="lineTrack"></div>';
    for(let v=0; v<=maxVal; v+=5){
      const t = document.createElement('div');
      t.className='tick'; t.style.left = posFor(v)+'px';
      lineDiv.appendChild(t);
      const l = document.createElement('div');
      l.className='tickLabel'; l.style.left = posFor(v)+'px'; l.textContent = v;
      lineDiv.appendChild(l);
    }
    const jp = document.createElement('div');
    jp.className='jumper'; jp.id='jumperIcon'; jp.style.left = posFor(0)+'px';
    jp.textContent='🐇';
    lineDiv.appendChild(jp);
  }
  buildLine();

  function updateEq(){
    const total = jumps*step;
    eq.innerHTML = `Jumps: <span class="hl">${jumps}</span> &nbsp;|&nbsp; Total: <span class="hl">${total}</span>` +
      (jumps>0 ? `<br><span style="font-size:18px;color:#555;">${step} × ${jumps} = ${total}</span>` : '');
    document.getElementById('jumperIcon').style.left = posFor(Math.min(total,maxVal))+'px';
  }

  container.querySelector('#jumpBtn').onclick = ()=>{
    const total = jumps*step;
    if(total + step <= maxVal){
      jumps++;
      const landing = document.createElement('div');
      landing.className='landing';
      landing.style.left = posFor(jumps*step)+'px';
      lineDiv.appendChild(landing);
      updateEq();
      if(jumps*step === targetTotal){
        askQuestion();
      }
    }
  };
  container.querySelector('#resetJump').onclick = ()=>{
    jumps = 0;
    buildLine();
    updateEq();
    qDiv.innerHTML='';
    feedback.textContent='';
  };

  function askQuestion(){
    qDiv.innerHTML = `<p style="font-size:19px;">You reached ${targetTotal}! How many jumps of ${step} did that take?</p>
      <div class="choiceRow" id="skipChoices"></div>`;
    const options = [3,4,5,6].sort(()=>Math.random()-0.5);
    const row = qDiv.querySelector('#skipChoices');
    options.forEach(opt=>{
      const b = document.createElement('button');
      b.className='choiceBtn';
      b.textContent = opt;
      b.onclick = ()=>{
        Array.from(row.children).forEach(c=>c.disabled=true);
        if(opt===targetJumps){
          b.classList.add('correct');
          feedback.textContent = `🎉 Yes! ${step} × ${targetJumps} = ${targetTotal}. Skip-counting IS multiplication!`;
          feedback.className='good';
          achieved[2]=true;
          updateNav(); buildProgressDots();
        } else {
          b.classList.add('wrong');
          feedback.textContent = "Try counting the landing dots again!";
          feedback.className='bad';
        }
      };
      row.appendChild(b);
    });
  }
  updateEq();
}

  let rows=2, cols=2;
  const rVal = container.querySelector('#rVal');
  const cVal = container.querySelector('#cVal');
  const visual = container.querySelector('#arrayVisual');
  const eq = container.querySelector('#arrayEq');
  const feedback = container.querySelector('#feedback');

  function draw(){
    rVal.textContent = rows; cVal.textContent = cols;
    visual.innerHTML = '';
    const grid = document.createElement('div');
    grid.className = 'arrayGrid';
    grid.style.gridTemplateColumns = `repeat(${cols}, 42px)`;
    for(let r=0;r<rows;r++){
      for(let c=0;c<cols;c++){
        const cell = document.createElement('div');
        cell.className='arrayCell';
        cell.textContent='🔵';
        grid.appendChild(cell);
      }
    }
    visual.appendChild(grid);
    let rowTotals = [];
    for(let r=1;r<=rows;r++) rowTotals.push(r*cols);
    eq.innerHTML = `${rows} × ${cols} = <span class="hl">${rows*cols}</span><br>
      <span style="font-size:18px;color:#555;">Row totals (skip-count by ${cols}): ${rowTotals.join(', ')}</span>`;
  }
  draw();
  container.querySelector('#rPlus').onclick=()=>{ if(rows<6) rows++; draw(); };
  container.querySelector('#rMinus').onclick=()=>{ if(rows>1) rows--; draw(); };
  container.querySelector('#cPlus').onclick=()=>{ if(cols<8) cols++; draw(); };
  container.querySelector('#cMinus').onclick=()=>{ if(cols>1) cols--; draw(); };

  container.querySelector('#checkArray').onclick = ()=>{
    if(rows===target.rows && cols===target.cols){
      feedback.textContent = `🎉 Great array! ${target.rows} rows × ${target.cols} columns = ${target.rows*target.cols}.`;
      feedback.className='good';
      achieved[3]=true;
      updateNav(); buildProgressDots();
    } else {
      feedback.textContent = `Keep adjusting — you need ${target.rows} rows and ${target.cols} columns.`;
      feedback.className='bad';
    }
  };
}

  }

let wpIndex = 0;
let wpSolvedCount = 0;

function renderWord(container){
  wpIndex = 0;
  wpSolvedCount = 0;
  drawWordProblem(container);
}

function drawWordProblem(container){
  const p = wordProblems[wpIndex];
  container.innerHTML = `
    <h2 class="stageTitle">${STAGE_TITLES[4]}</h2>
    <p class="instructions">Read the story. Pick the multiplication expression that matches it.</p>
    <div class="wordCard">
      <div class="emoji-story">${p.emoji}</div>
      <div class="story">${p.story}</div>
      <div class="choiceRow" id="wordChoices"></div>
      <div id="wordVisual" style="margin-top:14px;"></div>
    </div>
    <div id="feedback"></div>
    <div style="margin-top:10px; color:#888;">Problem ${wpIndex+1} of ${wordProblems.length}</div>
  `;
  const row = container.querySelector('#wordChoices');
  const feedback = container.querySelector('#feedback');
  const visualDiv = container.querySelector('#wordVisual');

  p.options.forEach(opt=>{
    const b = document.createElement('button');
    b.className='choiceBtn';
    b.textContent = opt;
    b.onclick = ()=>{
      Array.from(row.children).forEach(c=>c.disabled=true);
      if(opt===p.correct){
        b.classList.add('correct');
        feedback.textContent = "🎉 Correct! Let's see the groups:";
        feedback.className='good';
        showGroupsVisual(visualDiv, p.groups, p.items);
        wpSolvedCount++;
        setTimeout(()=>{
          if(wpIndex < wordProblems.length-1){
            wpIndex++;
            drawWordProblem(container);
          } else {
            achieved[4] = true;
            updateNav(); buildProgressDots();
            feedback.textContent = "🌟 You solved all the word problems!";
          }
        }, 1800);
      } else {
        b.classList.add('wrong');
        feedback.textContent = "Not quite — think about equal groups and how many groups there are.";
        feedback.className='bad';
        setTimeout(()=>{ Array.from(row.children).forEach(c=>c.disabled=false); b.classList.remove('wrong'); }, 900);
      }
    };
    row.appendChild(b);
  });
}

function showGroupsVisual(container, groups, items){
  container.innerHTML='';
