<script>
const STAGE_TITLES = [
  "Welcome to Multiplication!",
  "Equal Groups: Match the Picture",
  "Skip Counting on the Number Line",
  "Arrays: Match the Picture",
  "Word Problems",
  "You're a Multiplication Star!"

let stage = 0;
let achieved = [true,false,false,false,false,true];
let welcomeTimer = null;

const main = document.getElementById('main');
const progressEl = document.getElementById('progress');
const backBtn = document.getElementById('backBtn');
const nextBtn = document.getElementById('nextBtn');
const stageLabel = document.getElementById('stageLabel');

function buildProgressDots(){
  progressEl.innerHTML = '';
  for(let i=0;i<6;i++){
    const d = document.createElement('div');
    d.className = 'dot' + (i===stage?' active':'') + (achieved[i] && i!==stage ? ' done':'');
    d.addEventListener('click', ()=>{ if(i<=stage || achieved[i-1]) goTo(i); });
    progressEl.appendChild(d);
  }
}

function updateNav(){
  backBtn.disabled = stage===0;
  nextBtn.disabled = !achieved[stage] || stage===5;
  nextBtn.textContent = stage===5 ? "🎉 Done" : "Next ➡";
  stageLabel.textContent = "Step " + (stage+1) + " of 6";
}

function goTo(i){
  if(welcomeTimer){ clearInterval(welcomeTimer); welcomeTimer=null; }
  stage = i;
  render();
}

backBtn.addEventListener('click', ()=>{ if(stage>0) goTo(stage-1); });
nextBtn.addEventListener('click', ()=>{ if(stage<5 && achieved[stage]) goTo(stage+1); });

function render(){
  buildProgressDots();
  updateNav();
  main.innerHTML = '';
  const div = document.createElement('div');
  div.className = 'stage visible fadeGrow';
  main.appendChild(div);
  RENDERERS[stage](div);
}

/* ============ SHARED HELPERS ============ */

function randBetween(a,b){ return a + Math.floor(Math.random()*(b-a+1)); }

// draw a number-line with jump footprints and a flag at target position (flag has no visible number)
function drawJumpLine(el, {maxVal, step, jumps, targetTotal, ghostStep}){
  el.innerHTML = '<div class="lineTrack"></div>';
  const width = 900;
  const posFor = v => Math.min(v,maxVal)/maxVal*width;
  for(let v=0; v<=maxVal; v+=Math.max(1,Math.round(maxVal/10))){
    const t = document.createElement('div');
    t.className='tick'; t.style.left = posFor(v)+'px';
    el.appendChild(t);
  }
  // ghost footprints hinting spacing (no numbers)
  if(ghostStep){
    for(let v=ghostStep; v<maxVal; v+=ghostStep){
      const g = document.createElement('div');
      g.className = 'ghostDot';
      g.style.left = posFor(v)+'px';
      el.appendChild(g);
    }
  }
  // landing dots for jumps made so far
  const total = jumps*step;
  for(let j=1;j<=jumps;j++){
    const landing = document.createElement('div');
    landing.className='landing';
    landing.style.left = posFor(j*step)+'px';
    el.appendChild(landing);
  }
  // target flag (position only, no digit)
  if(targetTotal!==undefined){
    const flag = document.createElement('div');
    flag.className='flag';
    flag.style.left = posFor(targetTotal)+'px';
    flag.textContent = '🚩';
    el.appendChild(flag);
  }
  const jp = document.createElement('div');
  jp.className='jumper';
  jp.style.left = posFor(total)+'px';
  jp.textContent='🐇';
  el.appendChild(jp);
}

// render ghost (ideal target) equal-groups picture — child must count it themselves
function renderGhostGroups(el, groups, items, emoji){
  el.innerHTML = '';
  for(let g=0; g<groups; g++){
    const box = document.createElement('div');
    box.className = 'groupBox ghost';
    for(let i=0;i<items;i++){
      const it = document.createElement('span');
      it.className = 'item ghostItem';
      it.textContent = emoji;
      box.appendChild(it);
    }
    el.appendChild(box);
  }
}

// render ghost array picture
function renderGhostArray(el, rows, cols, emoji){
  el.innerHTML = '';
  const grid = document.createElement('div');
  grid.className = 'arrayGrid';
  grid.style.gridTemplateColumns = `repeat(${cols}, 34px)`;
  for(let r=0;r<rows;r++){
    for(let c=0;c<cols;c++){
      const cell = document.createElement('div');
      cell.className='arrayCell ghost';
      cell.textContent = emoji;
      grid.appendChild(cell);
    }
  }
  el.appendChild(grid);
}

// visual comparison of child's build vs target — reasoning-dependent feedback
function renderCompare(container, yourTotal, targetTotal, yourGroups, targetGroups, yourItems, targetItems, unitLabel1, unitLabel2){
  container.innerHTML = '';
  const wrap = document.createElement('div');
  wrap.className = 'compareWrap';

  function dotsRow(label, count, isYours){
    const row = document.createElement('div');
    row.className = 'compareRow';
    const lab = document.createElement('div');
    lab.className='rowLabel';
    lab.textContent = label;
    row.appendChild(lab);
    const maxShow = 24;
    const showCount = Math.min(count, maxShow);
    for(let i=0;i<showCount;i++){
      const d = document.createElement('div');
      d.className = 'compareDot' + (isYours && count>targetTotal && i>=targetTotal ? ' extra':'');
      row.appendChild(d);
    }
    if(count>maxShow){
      const more = document.createElement('span');
      more.style.fontWeight='bold'; more.style.color='#4a3f8c';
      more.textContent = ' +' + (count-maxShow);
      row.appendChild(more);
    }
    return row;
  }

  wrap.appendChild(dotsRow('Your total: ' + yourTotal, yourTotal, true));
  wrap.appendChild(dotsRow('Target total: ?', targetTotal, false));

  const diff = document.createElement('div');
  diff.className = 'diffMsg';
  if(yourTotal < targetTotal){
    diff.textContent = `You need ${targetTotal - yourTotal} more to match the picture!`;
  } else if(yourTotal > targetTotal){
    diff.textContent = `That's ${yourTotal - targetTotal} too many — take some away!`;
  } else {
    diff.textContent = `Totals match! ✅`;
  }
  wrap.appendChild(diff);

  const shapeMsg = document.createElement('div');
  shapeMsg.className = 'shapeMsg';
  if(yourTotal === targetTotal && (yourGroups!==targetGroups || yourItems!==targetItems)){
    let hint = [];
    if(yourGroups > targetGroups) hint.push(`fewer ${unitLabel1}`);
    if(yourGroups < targetGroups) hint.push(`more ${unitLabel1}`);
    if(yourItems > targetItems) hint.push(`fewer ${unitLabel2} in each`);
    if(yourItems < targetItems) hint.push(`more ${unitLabel2} in each`);
    shapeMsg.textContent = `Same total, but the picture is shaped differently — try ${hint.join(' and ')}.`;
  } else if(yourTotal===targetTotal){
    shapeMsg.textContent = '';
  } else {
    if(yourGroups > targetGroups) shapeMsg.textContent = `Hint: the picture has fewer ${unitLabel1} than yours.`;
    else if(yourGroups < targetGroups) shapeMsg.textContent = `Hint: the picture has more ${unitLabel1} than yours.`;
    else if(yourItems > targetItems) shapeMsg.textContent = `Hint: each ${unitLabel1.slice(0,-1)} in the picture has fewer ${unitLabel2}.`;
    else if(yourItems < targetItems) shapeMsg.textContent = `Hint: each ${unitLabel1.slice(0,-1)} in the picture has more ${unitLabel2}.`;
  }
  wrap.appendChild(shapeMsg);
  container.appendChild(wrap);
}

  container.innerHTML = `
    <h2 class="stageTitle">${STAGE_TITLES[0]}</h2>
    <p class="instructions">Multiplication is a fast way to add the <b>same number</b> over and over.<br>

  let groups = 1;
  const perGroup = 3;
  function draw(){
    gWrap.innerHTML = '';
    for(let g=0; g<groups; g++){
      const box = document.createElement('div');
      box.className = 'groupBox';
      for(let i=0;i<perGroup;i++){
        const it = document.createElement('span');
        it.className = 'item';
        it.textContent = '🍎';
        box.appendChild(it);
      }
      gWrap.appendChild(box);

    eq.innerHTML = `<span class="hl">${groups}</span> group${groups>1?'s':''} of <span class="hl">${perGroup}</span> = <span class="hl">${groups*perGroup}</span> apples`;
  }
  draw();
  welcomeTimer = setInterval(()=>{
    groups++;
    if(groups>5){ groups=1; }
    draw();
  }, 1400);
}

    <div class="jumpLine" id="groupsJumpLine"></div>

    <div id="feedback"></div>
    <div id="compareArea"></div>
    <button class="btn" id="checkGroups">✅ Check My Groups</button>
  `;
  renderGhostGroups(container.querySelector('#ghostGroups'), target.groups, target.items, '⭐');

  let groups = 2, items = 2;
  const gVal = container.querySelector('#gVal');
  const iVal = container.querySelector('#iVal');
  const visual = container.querySelector('#groupsVisual');
  const eq = container.querySelector('#groupsEq');
  const feedback = container.querySelector('#feedback');
  const jumpLineEl = container.querySelector('#groupsJumpLine');
  const compareArea = container.querySelector('#compareArea');
  const maxVal = target.groups*target.items + Math.max(target.items*2, 6);

  function draw(){
    gVal.textContent = groups;
    iVal.textContent = items;
    visual.innerHTML = '';
    for(let g=0; g<groups; g++){
      const box = document.createElement('div');
      box.className = 'groupBox';
      for(let i=0;i<items;i++){
        const it = document.createElement('span');
        it.className = 'item';
        it.textContent = '⭐';
        box.appendChild(it);
      }
      visual.appendChild(box);
    }
    eq.innerHTML = `${groups} × ${items} = <span class="hl">${groups*items}</span>`;
    drawJumpLine(jumpLineEl, {maxVal, step: items, jumps: groups, targetTotal: target.groups*target.items});
    compareArea.innerHTML = '';
  }
  draw();

  container.querySelector('#gPlus').onclick = ()=>{ if(groups<7) groups++; draw(); };
  container.querySelector('#gMinus').onclick = ()=>{ if(groups>1) groups--; draw(); };
  container.querySelector('#iPlus').onclick = ()=>{ if(items<7) items++; draw(); };
  container.querySelector('#iMinus').onclick = ()=>{ if(items>1) items--; draw(); };

  container.querySelector('#checkGroups').onclick = ()=>{
    if(groups===target.groups && items===target.items){
      feedback.textContent = "🎉 Perfect match! " + target.groups + " × " + target.items + " = " + (target.groups*target.items) + ".";
      feedback.className = 'good';
      compareArea.innerHTML = '';
      achieved[1] = true;
      updateNav();
      buildProgressDots();
    } else {
      feedback.textContent = "";
      renderCompare(compareArea, groups*items, target.groups*target.items, groups, target.groups, items, target.items, 'groups', 'stars');
    }
  };
}

/* ===================== STAGE 2: SKIP COUNTING ===================== */
function renderSkip(container){
  const secretStep = randBetween(2,5);
  const secretJumps = randBetween(4,6);
  const targetTotal = secretStep*secretJumps;
  const maxVal = targetTotal + secretStep*2;
  container.innerHTML = `
    <h2 class="stageTitle">${STAGE_TITLES[2]}</h2>
    <p class="instructions">See the dashed footprints and the flag 🚩 on the number line? They mark a hidden skip-counting pattern.
    Set your jump size to match the spacing between footprints, then jump until you land exactly on the flag!</p>
    <div class="stepper">
      <div>My Jump Size</div>
      <div class="row">
        <button class="stepBtn" id="sMinus">−</button>
        <div class="value" id="sVal">1</div>
        <button class="stepBtn" id="sPlus">+</button>
      </div>
    </div>
    <div class="jumpLine" id="numberLine"></div>
    <div class="equationBox" id="skipEq">Jumps: 0 &nbsp;|&nbsp; Total: 0</div>
    <button class="btn" id="jumpBtn">🐇 Jump!</button>
    <button class="btn secondary small" id="resetJump" style="margin-top:8px;">↺ Reset</button>
    <div id="skipQuestion" style="margin-top:10px;"></div>
    <div id="feedback"></div>
  `;
  const lineDiv = container.querySelector('#numberLine');
  const eq = container.querySelector('#skipEq');
  const feedback = container.querySelector('#feedback');
  const qDiv = container.querySelector('#skipQuestion');
  const sVal = container.querySelector('#sVal');
  let jumps = 0;
  let myStep = 1;

  function updateEq(){
    const total = jumps*myStep;
    eq.innerHTML = `Jumps: <span class="hl">${jumps}</span> &nbsp;|&nbsp; Total: <span class="hl">${total}</span>` +
      (jumps>0 ? `<br><span style="font-size:16px;color:#555;">${myStep} × ${jumps} = ${total}</span>` : '');
    drawJumpLine(lineDiv, {maxVal, step: myStep, jumps, targetTotal, ghostStep: secretStep});
  }

  container.querySelector('#sPlus').onclick = ()=>{ if(myStep<8){ myStep++; sVal.textContent=myStep; jumps=0; updateEq(); qDiv.innerHTML=''; feedback.textContent=''; } };
  container.querySelector('#sMinus').onclick = ()=>{ if(myStep>1){ myStep--; sVal.textContent=myStep; jumps=0; updateEq(); qDiv.innerHTML=''; feedback.textContent=''; } };

  container.querySelector('#jumpBtn').onclick = ()=>{
    const total = jumps*myStep;
    if(total + myStep <= maxVal){
      jumps++;
      updateEq();
      const newTotal = jumps*myStep;
      if(newTotal === targetTotal && myStep === secretStep){
        feedback.textContent = "🎯 Bullseye! You landed right on the flag!";
        feedback.className = 'good';
        askQuestion();
      } else if(newTotal === targetTotal && myStep !== secretStep){
        feedback.textContent = "You reached the flag's spot, but with a different jump size — check the footprint spacing!";
        feedback.className = 'bad';
      } else if(newTotal > targetTotal){
        feedback.textContent = `Overshot by ${newTotal-targetTotal}! Reset and try a different jump size or fewer jumps.`;
        feedback.className = 'bad';
      } else {
        feedback.textContent = `Total is ${targetTotal-newTotal} away from the flag — keep jumping or adjust your size.`;
        feedback.className = '';
      }
    }
  };
  container.querySelector('#resetJump').onclick = ()=>{
    jumps = 0;
    updateEq();
    qDiv.innerHTML='';
    feedback.textContent='';
  };

  function askQuestion(){
    qDiv.innerHTML = `<p style="font-size:17px;">Great! How many jumps of ${myStep} did that take to reach the flag?</p>
      <div class="choiceRow" id="skipChoices"></div>`;
    const wrong = new Set();
    while(wrong.size<3){
      const w = secretJumps + randBetween(-2,2);
      if(w>0 && w!==secretJumps) wrong.add(w);
    }
    const options = [secretJumps, ...wrong].sort(()=>Math.random()-0.5);
    const row = qDiv.querySelector('#skipChoices');
    options.forEach(opt=>{
      const b = document.createElement('button');
      b.className='choiceBtn';
      b.textContent = opt;
      b.onclick = ()=>{
        Array.from(row.children).forEach(c=>c.disabled=true);
        if(opt===secretJumps){
          b.classList.add('correct');
          feedback.textContent = `🎉 Yes! ${myStep} × ${secretJumps} = ${targetTotal}. Skip-counting IS multiplication!`;
          feedback.className='good';
          achieved[2]=true;
          updateNav(); buildProgressDots();
        } else {
          b.classList.add('wrong');
          feedback.textContent = "Count the red landing dots on the line again!";
          feedback.className='bad';
        }
      };
      row.appendChild(b);
    });
  }
  updateEq();
}
