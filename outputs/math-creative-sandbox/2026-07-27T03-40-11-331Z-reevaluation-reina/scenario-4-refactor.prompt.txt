function goTo(id){

function renderProgress(currentId){
  const bar = document.getElementById('progressBar');
  bar.innerHTML = '';
  const idx = SCREEN_ORDER.indexOf(currentId);
  SCREEN_ORDER.forEach((id,i)=>{
    const dot = document.createElement('div');
    dot.className = 'progress-dot' + (i===idx ? ' active' : (i<idx ? ' done' : ''));
    bar.appendChild(dot);
  });
}

/* ---------------- Helpers ---------------- */
function makeStepper(label, min, max, value, onChange){
  const wrap = document.createElement('div');
  wrap.className = 'stepper';
  const lbl = document.createElement('div');
  lbl.className = 'stepper-label';
  lbl.textContent = label;
  const row = document.createElement('div');
  row.className = 'stepper-row';
  const minus = document.createElement('button');
  minus.className = 'step-btn'; minus.textContent = '−';
  const val = document.createElement('div');
  val.className = 'stepper-value'; val.textContent = value;
  const plus = document.createElement('button');
  plus.className = 'step-btn'; plus.textContent = '+';
  minus.onclick = () => { if(value > min){ value--; val.textContent = value; onChange(value); } };
  plus.onclick = () => { if(value < max){ value++; val.textContent = value; onChange(value); } };
  row.append(minus, val, plus);
  wrap.append(lbl, row);
  return { el: wrap, get: () => value, set: (v) => { value = v; val.textContent = v; } };
}

function renderGroupsVisual(container, groups, size, emoji){
  container.innerHTML = '';
  let cum = 0;
  for(let g=0; g<groups; g++){
    const box = document.createElement('div');
    box.className = 'group-box';
    const items = document.createElement('div');
    items.className = 'group-items';
    for(let i=0;i<size;i++){
      const span = document.createElement('span');
      span.textContent = emoji;
      items.appendChild(span);
    }
    box.appendChild(items);
    cum += size;
    const cLabel = document.createElement('div');
    cLabel.className = 'cum-label';
    cLabel.textContent = cum;
    box.appendChild(cLabel);
    container.appendChild(box);
  }
}

function renderArrayRows(container, rows, cols, emoji){
  container.innerHTML = '';
  const rowEls = [];
  for(let r=0; r<rows; r++){
    const rowDiv = document.createElement('div');
    rowDiv.className = 'array-row';
    for(let c=0;c<cols;c++){
      const cell = document.createElement('div');
      cell.className = 'array-cell2';
      cell.textContent = emoji;
      rowDiv.appendChild(cell);
    }
    container.appendChild(rowDiv);
    rowEls.push(rowDiv);
  }
  return rowEls;
}

function skipList(n1, n2){
  let arr = [];
  for(let i=1;i<=n1;i++) arr.push(i*n2);
  return arr;
}

function randInt(min,max){ return Math.floor(Math.random()*(max-min+1))+min; }

function starsDisplay(count, max){
  let s = '';
  for(let i=0;i<max;i++) s += (i<count ? '⭐' : '✩');
  return s;
}

/* ================= TEACH 1: Equal Groups (fixed 3 groups of 4) ================= */
function setupTeach1(){
  const container = document.getElementById('t1-groups');
  container.innerHTML = '';
  const groups = 3, size = 4;
  let currentIndex = 0;
  const revealedSeq = [];
  document.getElementById('t1-seq').textContent = '';
  document.getElementById('t1-eq').textContent = '';
  document.getElementById('t1-next').disabled = true;

  for(let g=0; g<groups; g++){
    const box = document.createElement('div');
    box.className = 'group-box dim';
    const items = document.createElement('div');
    items.className = 'group-items';
    for(let i=0;i<size;i++){
      const span = document.createElement('span');
      span.textContent = '🌼';
      items.appendChild(span);
    }
    box.appendChild(items);
    const cLabel = document.createElement('div');
    cLabel.className = 'cum-label';
    cLabel.textContent = '?';
    box.appendChild(cLabel);
    box.dataset.index = g;

    box.onclick = () => {
      const gi = parseInt(box.dataset.index);
      if(gi !== currentIndex) return;
      box.classList.remove('dim');
      box.classList.add('active-tap');
      const cum = (currentIndex+1) * size;
      cLabel.textContent = cum;
      revealedSeq.push(cum);
      document.getElementById('t1-seq').textContent = revealedSeq.join(', ');
      currentIndex++;
      if(currentIndex === groups){
        document.getElementById('t1-eq').textContent = `${groups} groups of ${size} = ${groups} × ${size} = ${groups*size}`;
        document.getElementById('t1-next').disabled = false;
      }
    };
    container.appendChild(box);
  }
}
setupTeach1();

  });

    totalStars++;
    let extra = '';
    if((b2r===target.rows && b2c===target.cols) || (b2r===target.cols && b2c===target.rows)){
      extra = ' You matched the exact array shape too! ⭐';
    }
    fb.textContent = `🎉 Yes! ${b2r} × ${b2c} = ${gotTotal}.${extra}`;
    fb.className = 'feedback good';
    document.getElementById('b2-check').style.display = 'none';
    document.getElementById('b2-continue').style.display = 'inline-block';
  } else {
    fb.textContent = `Not quite — your array shows ${gotTotal}. Try adjusting rows or columns to reach ${wantTotal}.`;
    fb.className = 'feedback bad';
  }
}

function nextArrayRound(){
  arrayRoundIndex++;
  if(arrayRoundIndex < arrayTargets.length){
    setupArrayRound();
  } else {
    startWordProblems();
  }
}

const problemTemplates = [
  { text:(a,b)=>`Pip lines up ${a} baskets in the shed. Each basket holds ${b} shiny apples. How many apples in all?`, emoji:'🍎', hintMode:'groups' },
  { text:(a,b)=>`Pip sets out ${a} flower pots. Each pot has ${b} tulips. How many tulips in total?`, emoji:'🌷', hintMode:'groups' },
  { text:(a,b)=>`Pip plants seeds in ${a} neat rows, with ${b} seeds in each row. How many seeds did Pip plant?`, emoji:'🌱', hintMode:'array' },
  { text:(a,b)=>`There are ${a} spiderwebs in the garden. Each web has ${b} sparkly dew drops. How many dew drops in all?`, emoji:'💧', hintMode:'groups' },
  { text:(a,b)=>`The garden has ${a} rows of pumpkins with ${b} pumpkins in each row. How many pumpkins are there?`, emoji:'🎃', hintMode:'array' },
  { text:(a,b)=>`Pip bakes ${a} trays of cookies. Each tray has ${b} cookies. How many cookies did Pip bake?`, emoji:'🍪', hintMode:'groups' },
  { text:(a,b)=>`Ladybugs line up in ${a} rows and ${b} columns on a big leaf. How many ladybugs are there?`, emoji:'🐞', hintMode:'array' },
  { text:(a,b)=>`Pip has ${a} jars on the shelf. Each jar has ${b} fireflies. How many fireflies in all?`, emoji:'✨', hintMode:'groups' }
];

let wordProblems = [];
let wordIndex = 0;
let wordStars = 0;
let wMode = 'groups';
let wN1 = 2, wN2 = 2;

function pickWordProblems(){
  const shuffled = [...problemTemplates].sort(()=>Math.random()-0.5);
  const chosen = shuffled.slice(0,5);
  return chosen.map(t => {
    const a = randInt(2,5);
    const b = randInt(2,6);
    return { text: t.text(a,b), emoji: t.emoji, a, b, hintMode: t.hintMode };
  });
}

function startWordProblems(){
  wordProblems = pickWordProblems();
  wordIndex = 0;
  wordStars = 0;
  goTo('screen-word');
  setupWordProblem();
}

function setupWordProblem(){
  const p = wordProblems[wordIndex];
  document.getElementById('w-round-label').textContent = `(${wordIndex+1} of ${wordProblems.length})`;
  document.getElementById('w-emoji').textContent = p.emoji;
  document.getElementById('w-text').textContent = p.text;
  document.getElementById('w-feedback').textContent = '';
  document.getElementById('w-feedback').className = 'feedback';
  document.getElementById('w-check').style.display = 'inline-block';
  document.getElementById('w-continue').style.display = 'none';

  wMode = p.hintMode;
  wN1 = 2; wN2 = 2;

  const toggle = document.getElementById('w-mode-toggle');
  toggle.innerHTML = '';
  const btnGroups = document.createElement('button');
  btnGroups.className = 'mode-btn' + (wMode==='groups' ? ' selected' : '');
  btnGroups.textContent = 'Equal Groups';
  btnGroups.onclick = () => { wMode='groups'; refreshWordProblem(); updateModeButtons(); };
  const btnArray = document.createElement('button');
  btnArray.className = 'mode-btn' + (wMode==='array' ? ' selected' : '');
  btnArray.textContent = 'Array (Rows × Cols)';
  btnArray.onclick = () => { wMode='array'; refreshWordProblem(); updateModeButtons(); };
  toggle.append(btnGroups, btnArray);

  function updateModeButtons(){
    btnGroups.classList.toggle('selected', wMode==='groups');
    btnArray.classList.toggle('selected', wMode==='array');
  }

  buildWordSteppers();
  refreshWordProblem();
}

function buildWordSteppers(){
  const wrap = document.getElementById('w-steppers');
  wrap.innerHTML = '';
  const label1 = wMode === 'groups' ? 'Number of Groups' : 'Rows';
  const label2 = wMode === 'groups' ? 'Size of Each Group' : 'Columns';
  const s1 = makeStepper(label1, 1, 8, wN1, (v)=>{ wN1=v; refreshWordProblem(); });
  const s2 = makeStepper(label2, 1, 8, wN2, (v)=>{ wN2=v; refreshWordProblem(); });
  wrap.append(s1.el, s2.el);
}

function refreshWordProblem(){
  buildWordSteppers();
  const p = wordProblems[wordIndex];
  const visual = document.getElementById('w-visual');
  if(wMode === 'groups'){
    visual.className = '';
    visual.innerHTML = '';
    renderGroupsVisual(visual, wN1, wN2, p.emoji);
    document.getElementById('w-eq').textContent = `${wN1} groups of ${wN2} = ${wN1} × ${wN2} = ${wN1*wN2}`;
  } else {
    visual.innerHTML = '';
    const holder = document.createElement('div');
    holder.className = 'array-area-rows';
    visual.appendChild(holder);
    renderArrayRows(holder, wN1, wN2, p.emoji);
    document.getElementById('w-eq').textContent = `${wN1} rows × ${wN2} columns = ${wN1} × ${wN2} = ${wN1*wN2}`;
  }
}

function checkWordProblem(){
  const p = wordProblems[wordIndex];
  const wantTotal = p.a * p.b;
  const gotTotal = wN1 * wN2;
  const fb = document.getElementById('w-feedback');
  if(gotTotal === wantTotal){
    wordStars++;
    totalStars++;
    fb.textContent = `🎉 That's right! ${wN1} × ${wN2} = ${gotTotal}.`;
    fb.className = 'feedback good';
    document.getElementById('w-check').style.display = 'none';
    document.getElementById('w-continue').style.display = 'inline-block';
  } else {
    fb.textContent = `Your total is ${gotTotal}. The story needs ${wantTotal}. Try changing the numbers!`;
    fb.className = 'feedback bad';
  }
}

function nextWordProblem(){
  wordIndex++;
  if(wordIndex < wordProblems.length){
    setupWordProblem();
  } else {
    showEnd();
  }
}

  document.getElementById('end-badge').textContent = badge;
}
