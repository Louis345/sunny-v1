    speechSynthesis.cancel();
    const u=new SpeechSynthesisUtterance(text.replace(/<[^>]+>/g,''));
    u.rate=.94;u.pitch=1.02;speechSynthesis.speak(u);
  }

  function setRace(p=state.playerProgress,r=state.rivalProgress){
    state.playerProgress=clamp(p,0,100); state.rivalProgress=clamp(r,0,100);
    const lane = Math.max(620, $('#playerTrack').clientWidth - 110);

  function clearRace(){ if(state.raceTimer){clearInterval(state.raceTimer);state.raceTimer=null} }
  function startRace(duration){
    clearRace();
    state.playerProgress=0;state.rivalProgress=0;state.rivalFinished=false;state.legWon=false;
    state.raceDuration=duration;state.raceStart=performance.now();setRace();
    state.raceTimer=setInterval(()=>{
      const elapsed=(performance.now()-state.raceStart)/1000;
      const wobble=Math.sin(elapsed*1.8)*.35;
      state.rivalProgress=clamp(elapsed/duration*100+wobble,0,100);
      setRace(state.playerProgress,state.rivalProgress);
      if(state.rivalProgress>=100){
        clearRace(); state.rivalFinished=true;
        $('#gate').classList.add('closed');
        rivalSays("Gate first! Those bonus goods ride with me.");
        toast("Bonus goods lost — finish the order and keep racing.",2600);
      }
    },120);
  }
  function advancePlayer(amount){
    if(state.stage==='teach') return;

    setRace(Math.min(cap,state.playerProgress+amount),state.rivalProgress);
  }

  function makeOrder(stage, garbled=false){
    const [r,c]=pick(combos);
    const icon=pick(icons);
    const actual=r*c;
    let wrong=actual+pick([-5,-4,-3,3,4,5]);
    if(wrong<2) wrong=actual+4;
    return {
      rows:r, cols:c, icon, actual, garbled, wrong,
      text: stage==='guided'
        ? `The tavern needs <b>${r} barrels</b>, with <b>${c} potions ${icon} in each barrel</b>.`
        : pick(wordTemplates)(r,c,icon),
      challenged:false, falseChallenge:false
    };
  }

  function makeBoard(spec,index){
    return {
      rows:spec.rows,cols:spec.cols,targetRows:spec.targetRows,targetCols:spec.targetCols,
      label:spec.label||'',index,filled:new Set(),doneRows:new Set(),resolving:false
    };
  }
  function createBoards(specs){
    state.boards=specs.map(makeBoard);
    renderBoards();
  }
  function renderBoards(){
    gridZone.innerHTML='';
    state.boards.forEach((b,bi)=>{
      const wrap=document.createElement('div');
      wrap.className='boardWrap'+(state.double?' doubleBoard':'');
      wrap.dataset.board=bi;
      if(b.label){
        const label=document.createElement('div');label.className='boardLabel';label.textContent=b.label;wrap.appendChild(label);
      }
      const grid=document.createElement('div');grid.className='crateGrid';grid.dataset.board=bi;
      for(let r=0;r<b.rows;r++){
        const row=document.createElement('div');row.className='crateRow';row.dataset.row=r;
        for(let c=0;c<b.cols;c++){
          const slot=document.createElement('div');slot.className='slot';slot.dataset.board=bi;slot.dataset.row=r;slot.dataset.col=c;
          const key=`${r}-${c}`;
          if(b.filled.has(key)){
            slot.classList.add('filled');
            const cr=document.createElement('div');cr.className='placedCrate';
            cr.textContent=state.double?state.doubleOrders[bi].icon:state.icon;slot.appendChild(cr);
          }
          slot.addEventListener('pointerdown',e=>{
            if(state.tapSelected && !slot.classList.contains('filled')){
              e.preventDefault();placeNext(bi);state.tapSelected=false;$$('.trayCrate').forEach(x=>x.classList.remove('selected'));
            }
          });
          row.appendChild(slot);
        }
        const badge=document.createElement('div');badge.className='rowBadge';
        if(b.doneRows.has(r)){badge.classList.add('live');badge.textContent=(r+1)*b.targetCols}
        row.appendChild(badge);grid.appendChild(row);
      }
      wrap.appendChild(grid);gridZone.appendChild(wrap);
    });
    markNext();
    $('#rowsValue').textContent=state.boards[0]?.rows||0;
    $('#colsValue').textContent=state.boards[0]?.cols||0;
  }
  function markNext(){
    $$('.slot.next').forEach(s=>s.classList.remove('next'));
    if(state.stage!=='teach') return;
    const b=state.boards[0];
    for(let r=0;r<b.rows;r++)for(let c=0;c<b.cols;c++){
      if(!b.filled.has(`${r}-${c}`)){
        const slot=$(`.slot[data-board="0"][data-row="${r}"][data-col="${c}"]`);
        if(slot) slot.classList.add('next');return;
      }
    }
  }
  function nextEmpty(b){
    for(let r=0;r<b.rows;r++)for(let c=0;c<b.cols;c++){
      if(!b.filled.has(`${r}-${c}`)) return {r,c};
    }
    return null;
  }
  function placeNext(boardIndex){
    const b=state.boards[boardIndex];if(!b||b.resolving)return;
    const spot=nextEmpty(b);if(!spot)return;
    b.filled.add(`${spot.r}-${spot.c}`);
    sound('drop');renderBoards();
    const rowFull=[...Array(b.cols)].every((_,c)=>b.filled.has(`${spot.r}-${c}`));
    if(rowFull && !b.doneRows.has(spot.r)){
      b.resolving=true;setTimeout(()=>evaluateRow(b,spot.r),180);
    }
  }
  function evaluateRow(b,row){
    b.resolving=false;
    if(b.cols!==b.targetCols){
      spillRow(b,row,`That row holds ${b.cols}. The order needs ${b.targetCols} in each row.`);
      return;
    }
    b.doneRows.add(row);state.streak++;
    const total=(row+1)*b.targetCols;
    updateTally(total,b.index);
    renderBoards();
    const totalTargetRows=state.boards.reduce((s,x)=>s+x.targetRows,0);
    advancePlayer(86/totalTargetRows);
    if(state.streak>0 && state.streak%2===0){
      $('#secondWind').classList.add('ready');
      if(state.stage!=='teach' && state.playerProgress<state.rivalProgress){
        setTimeout(()=>{
          advancePlayer(10);toast('SECOND WIND! Perfect rows pull your caravan forward.');sound('win');
        },220);
      }
    } else $('#secondWind').classList.remove('ready');

    const selectedComplete=b.doneRows.size===b.rows && b.filled.size===b.rows*b.cols;
    if(selectedComplete){
      if(b.rows!==b.targetRows){
        spillRow(b,b.rows-1,`The order needs ${b.targetRows} rows. Adjust the row count.`);
        return;
      }
      if(!state.double && state.current.garbled && !state.current.challenged){
        spillRow(b,b.rows-1,"Ink trap! The written total is wrong. Challenge the order.");
        rivalSays("You packed my false total without checking the ink!");
        $('#challengeBtn').style.display='inline-block';
        return;
      }
      if(state.double){
        if(state.boards.every(x=>x.doneRows.size===x.targetRows)) completeOrder();
      } else completeOrder();
    }
  }
  function spillRow(b,row,message){
    const slotEls=$$(`.slot[data-board="${b.index}"][data-row="${row}"]`);
    slotEls.forEach((s,i)=>{
      if(!s.classList.contains('filled'))return;
      const rect=s.getBoundingClientRect(),sp=document.createElement('div');
      sp.className='spill';sp.textContent=state.double?state.doubleOrders[b.index].icon:state.icon;
      sp.style.left=(rect.left+8)+'px';sp.style.top=(rect.top+2)+'px';
      sp.style.setProperty('--dx',`${(i-(slotEls.length-1)/2)*34}px`);
      sp.style.setProperty('--rot',`${(i%2?1:-1)*(80+i*20)}deg`);
      document.body.appendChild(sp);setTimeout(()=>sp.remove(),800);
    });
    for(let c=0;c<b.cols;c++)b.filled.delete(`${row}-${c}`);
    b.doneRows.delete(row);state.streak=0;b.resolving=false;
    playerCaravan.classList.add('stumble');setTimeout(()=>playerCaravan.classList.remove('stumble'),600);
    sound('spill');renderBoards();updateTally();toast(message,2500);rivalSays(pick(smugLines));
    if(state.stage!=='teach' && !state.rivalFinished)setRace(state.playerProgress,Math.min(98,state.rivalProgress+4));
  }
  function updateTally(latest=null,boardIndex=0){
    const trail=$('#skipTrail');trail.innerHTML='';
    let max=0;
    state.boards.forEach((b,bi)=>{
      [...b.doneRows].sort((a,c)=>a-c).forEach(r=>{
        const n=(r+1)*b.targetCols;max=Math.max(max,n);
        const t=document.createElement('span');t.className='tick';
        t.textContent=(state.double?(bi?'B:':'A:'):'')+n;trail.appendChild(t);
      });
    });
    $('#runningTotal').textContent=latest ?? max;
    $('#streak').textContent=state.streak;
  }
  function changeShape(kind,delta){
    if(state.stage==='teach'||state.double)return;
    const b=state.boards[0];
    let nr=b.rows,nc=b.cols;
    if(kind==='rows')nr=clamp(nr+delta,1,6);else nc=clamp(nc+delta,1,6);
    if(nr===b.rows&&nc===b.cols)return;
    if(nc!==b.cols){b.filled.clear();b.doneRows.clear();state.streak=0}
    else{
      [...b.filled].forEach(k=>{if(+k.split('-')[0]>=nr)b.filled.delete(k)});
      [...b.doneRows].forEach(r=>{if(r>=nr)b.doneRows.delete(r)});
    }
    b.rows=nr;b.cols=nc;renderBoards();updateTally();
    if(b.rows===b.targetRows&&b.cols===b.targetCols&&b.doneRows.size===b.targetRows)completeOrder();
  }

  function completeOrder(){
    clearRace();
    const o=state.double?null:state.current;
    if(state.stage!=='teach')setRace(100,state.rivalProgress);
    sound('win');
    const won=state.stage==='teach'?true:!state.rivalFinished;
    if(state.stage!=='teach'){
      const value=(state.stage==='quest'&&state.questIndex===3)?2:1;
      if(won){state.routePlayer+=value;state.loot++;}else state.routeRival+=value;
      updateScore();
    }
    const rec=state.double
      ? {name:'Double order',array:state.doubleOrders.map(q=>`${q.rows} × ${q.cols} = ${q.actual}`).join(' & '),won}
      : {name:state.stage,array:`${o.rows} × ${o.cols} = ${o.actual}`,won};
    state.records.push(rec);
    if(state.stage==='quest')state.questRecords.push(rec);

    setTimeout(()=>{
      if(state.stage==='teach'){
        showOverlay(`
          <h1>Cart Packed!</h1>
          <p>Each finished row counted by <b>4</b>: 4, 8, 12. Your 3 × 4 array loaded <b>12 crates</b>.</p>
          <p>Broker Brass has stopped watching. Now he wants a race.</p>
          <div class="overlayButtons"><button class="actionBtn" data-action="guided">Enter the Market Lane</button></div>
        `);
      } else if(state.stage==='guided'){
        showOverlay(`
          <h1>${won?'You Reached the Gate First!':'Order Complete — Race Continues'}</h1>
          <p>You translated the request into <b>${o.rows} rows of ${o.cols}</b>: ${o.actual} total.</p>
          <p>${won?'Bonus goods secured.':'Brass took this leg’s bonus goods, but nothing else was lost.'}</p>
          <div class="overlayButtons"><button class="actionBtn" data-action="quest">Begin the Grand Market Run</button></div>
        `);
      } else if(state.stage==='quest'){
        state.questPace=clamp(state.questPace+(won?-2.5:4),29,50);
        if(state.questIndex<3){
          showOverlay(`
            <h2>Leg ${state.questIndex+1} Complete</h2>
            <p><b>${o.rows} × ${o.cols} = ${o.actual}</b></p>
            <p>${won?'You claimed the bonus cargo.':'Brass reached the gate first, but your caravan stays in the run.'}</p>
            <div><span class="loot">Route score ${state.routePlayer}–${state.routeRival}</span></div>
            <div class="overlayButtons"><button class="actionBtn" data-action="nextQuest">Race the Next Leg</button></div>
          `);

        showOverlay(`

          <p>Two arrays, one clock:</p>
          <p><b>${state.doubleOrders[0].rows} × ${state.doubleOrders[0].cols} = ${state.doubleOrders[0].actual}</b><br>
          <b>${state.doubleOrders[1].rows} × ${state.doubleOrders[1].cols} = ${state.doubleOrders[1].actual}</b></p>
          <p>${won?'Both customers beat the gate.':'Brass got the bonus, but both arrays are correctly packed.'}</p>
          <div class="overlayButtons">
            <button class="actionBtn" data-action="double">New Double Order</button>
            <button class="paperBtn secondary" data-action="restart">New Full Run</button>
          </div>
        `);
      }
    },650);
  }

      }
    }else{
      o.challenged=true;o.falseChallenge=true;renderOrder();
      toast('This order is clean. Brass gains a small lead.');
      rivalSays("Suspicious of honest ink? How flattering.");
      if(!state.rivalFinished)setRace(state.playerProgress,Math.min(95,state.rivalProgress+8));
    }
  });

  // Pointer drag with tap fallback.
  let drag=null;
  $$('.trayCrate').forEach(crate=>{
    crate.addEventListener('pointerdown',e=>{
      e.preventDefault();initAudio();
      const ghost=document.createElement('div');ghost.className='dragGhost';
      let icon=crate.textContent||state.icon;ghost.textContent=icon;document.body.appendChild(ghost);
      ghost.style.left=e.clientX+'px';ghost.style.top=e.clientY+'px';
      drag={ghost,startX:e.clientX,startY:e.clientY,x:e.clientX,y:e.clientY,icon};
      crate.setPointerCapture?.(e.pointerId);
    });
    crate.addEventListener('pointermove',e=>{
      if(!drag)return;drag.x=e.clientX;drag.y=e.clientY;
      drag.ghost.style.left=e.clientX+'px';drag.ghost.style.top=e.clientY+'px';
    });
    crate.addEventListener('pointerup',e=>{
      if(!drag)return;
      const moved=Math.hypot(e.clientX-drag.startX,e.clientY-drag.startY)>8;
      drag.ghost.remove();
      if(!moved){
        state.tapSelected=!state.tapSelected;
        $$('.trayCrate').forEach(x=>x.classList.toggle('selected',state.tapSelected));
      }else{
        const hit=document.elementFromPoint(e.clientX,e.clientY);
        const grid=hit?.closest('.crateGrid');
        if(grid)placeNext(+grid.dataset.board);
      }
      drag=null;
    });
    crate.addEventListener('pointercancel',()=>{if(drag){drag.ghost.remove();drag=null}});
  });

  window.addEventListener('resize',()=>setRace());
  document.addEventListener('pointerdown',initAudio,{once:true});

  prepareTeach();
  // The rival visibly inches forward during the silent opening.
  let openingStart=performance.now();
  const opening=setInterval(()=>{
    const t=(performance.now()-openingStart)/3000;
    setRace(0,Math.min(5,t*5));
    if(t>=1){
      clearInterval(opening);setRace(0,5);
      app.classList.remove('cold');
    }
  },70);
})();
