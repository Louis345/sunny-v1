function makeOrder(){
 const node=S.node,cfg=NODES[node],item=pick(ITEMS);let R,C;
 if(node===0){R=rnd(2,4);C=rnd(2,5);}
 else if(node===1){R=rnd(2,5);C=rnd(2,5);}
 else{R=rnd(2,6);C=rnd(2,6);}
 const o={R,C,item,id:++S.orderNo,garbled:false};
 if(!cfg.word){o.text=`Pack <span class="ic">${item.ic}</span> ${R} rows of ${C} ${item.n}.`;o.say=`Pack ${R} rows of ${C} ${item.n}.`;}
 else{
   let t=pick(TPL).replace(/{R}/g,R).replace(/{C}/g,C).replace(/{U}/g,item.u).replace(/{N}/g,item.n);
   o.text=`<span class="ic">${item.ic}</span> `+t;o.say=t;
 }
 if(cfg.garble){
   const canGarble=S.ordersDone>0&&S.garbleCount<2&&Math.random()<.4;
   if(canGarble){o.garbled=true;S.garbleCount++;
     let wrong=R*C;const opts=[R*C+C,R*C-C,R*C+R,(R+1)*C,R*(C+1)];while(wrong===R*C)wrong=pick(opts);
     o.stated=Math.max(2,wrong);}
   else o.stated=R*C;
   o.say+=` The note claims ${o.stated} in all.`;
 }
 return o;
}

function shapeOf(g){
 const counts=g.map(r=>r.filter(Boolean).length);
 const total=counts.reduce((a,b)=>a+b,0);
 let R=0;for(let i=0;i<6;i++){if(counts[i]>0)R=i+1;}
 let contiguous=true;for(let i=0;i<R;i++)if(counts[i]===0)contiguous=false;
 const C=counts[0]||0;
 let uniform=contiguous&&R>0;for(let i=0;i<R;i++)if(counts[i]!==C)uniform=false;
 return{R,C,total,uniform,counts};
}
