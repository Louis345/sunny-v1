<script>
const $=id=>document.getElementById(id);
const rand=(a,b)=>a+Math.floor(Math.random()*(b-a+1));
const ICONS=["🍎","🧪","🍯","🍇","🥖","🫖","🍊"];
const TRACKW=1240;
let AC;
function beep(f,d=.1,t="square",v=.15){try{AC=AC||new (window.AudioContext||webkitAudioContext)();const o=AC.createOscillator(),g=AC.createGain();o.type=t;o.frequency.value=f;g.gain.setValueAtTime(v,AC.currentTime);g.gain.exponentialRampToValueAtTime(.001,AC.currentTime+d);o.connect(g);g.connect(AC.destination);o.start();o.stop(AC.currentTime+d)}catch(e){}}
function thud(){beep(140,.12,"triangle",.3);beep(90,.15,"sine",.25)}
function fanfare(){[523,659,784,1046].forEach((f,i)=>setTimeout(()=>beep(f,.25,"triangle",.2),i*120))}

let node=0, order=null, grid=[], gRows=5,gCols=6, playerX=0,rivalX=0, rivalTimer=null;
let ordersLeft=0, garbledIdx=-1, orderIdx=0, streak=0, secondWind=false, hintTimer=null, ghostOn=false;
let legWon=[], bonusGoods=0;

function buildBoard(){
 const b=$("board");b.innerHTML="";b.style.gridTemplateColumns=`repeat(${gCols},72px)`;grid=[];
 for(let r=0;r<gRows;r++){grid.push([]);for(let c=0;c<gCols;c++){const s=document.createElement("div");s.className="slot";s.dataset.r=r;s.dataset.c=c;b.appendChild(s);grid[r].push(null);}}
}
function slotEl(r,c){return $("board").children[r*gCols+c]}
function crateEl(icon){const d=document.createElement("div");d.className="crate";d.textContent=icon;return d}

function setTray(icon){
 const t=$("tray");t.innerHTML='<span id="trayLabel">CRATE SUPPLY →</span>';
 for(let i=0;i<5;i++){const c=crateEl(icon);t.appendChild(c);}
}

/* DRAG */
let dragging=null;
document.addEventListener("pointerdown",e=>{
 const c=e.target.closest(".crate");if(!c)return;
 if(c.parentElement.classList.contains("slot")){ // remove
   const s=c.parentElement;grid[s.dataset.r][s.dataset.c]=null;c.remove();updateTicker();return;
 }
 if(c.closest("#tray")){
  dragging=$("drag");dragging.innerHTML="";const cl=crateEl(order.icon);dragging.appendChild(cl);dragging.style.display="block";moveDrag(e);beep(600,.05,"sine",.1);
 }
});
function moveDrag(e){const st=$("stage").getBoundingClientRect();dragging.style.left=(e.clientX-st.left-33)+"px";dragging.style.top=(e.clientY-st.top-33)+"px";}
document.addEventListener("pointermove",e=>{if(dragging)moveDrag(e)});
document.addEventListener("pointerup",e=>{
 if(!dragging)return;dragging.style.display="none";dragging=null;
 const el=document.elementFromPoint(e.clientX,e.clientY);const s=el&&el.closest(".slot");
 if(s&&!grid[s.dataset.r][s.dataset.c]){
   const c=crateEl(order.icon);s.appendChild(c);grid[s.dataset.r][s.dataset.c]=1;thud();
   c.animate([{transform:"scale(1.3) translateY(-8px)"},{transform:"scale(1)"}],{duration:220,easing:"cubic-bezier(.3,1.6,.5,1)"});
   userActed();updateTicker();
 }
});

/* TICKER */
function rowCounts(){return grid.map(r=>r.reduce((a,v)=>a+(v?1:0),0))}
function updateTicker(){
 const tgt=order?order.cols:0;let tot=0;const el=$("tickNums");el.innerHTML="";
 rowCounts().forEach(c=>{if(c===tgt&&tgt>0){tot+=tgt;const s=document.createElement("span");s.className="tick";s.textContent=tot;el.appendChild(s);}});
 if(order&&hintNext)pulseNext();
}
let hintNext=false;
function pulseNext(){
 document.querySelectorAll(".slot.pulse").forEach(s=>s.classList.remove("pulse"));
 for(let r=0;r<order.rows;r++)for(let c=0;c<order.cols;c++)if(!grid[r][c]){slotEl(r,c).classList.add("pulse");return;}
}
function showGhost(){for(let r=0;r<order.rows;r++)for(let c=0;c<order.cols;c++)slotEl(r,c).classList.add("ghost");ghostOn=true;}
function clearGhost(){document.querySelectorAll(".ghost").forEach(s=>s.classList.remove("ghost"));ghostOn=false;}

/* SPEECH */
function say(txt,ms=2600){const s=$("speech");s.textContent=txt;s.style.display="block";clearTimeout(s._t);s._t=setTimeout(()=>s.style.display="none",ms);}
function speak(t){try{const u=new SpeechSynthesisUtterance(t);u.rate=.95;speechSynthesis.cancel();speechSynthesis.speak(u)}catch(e){}}

/* CARTS */
function setCarts(){$("cartP").style.left=(10+Math.min(playerX,1)*TRACKW*0)+"px";}
function drawCarts(){
 $("cartP").style.left=(10+Math.min(playerX/100,1)*(TRACKW-90))+"px";
 $("cartR").style.left=(10+Math.min(rivalX/100,1)*(TRACKW-90))+"px";
}
function resetLeg(){playerX=0;rivalX=0;drawCarts();}

/* ORDERS */
function makeOrder(rows,cols,text,garbled){return{rows,cols,text,garbled,icon:ICONS[rand(0,ICONS.length-1)],fixed:false}}
function wordProblem(rows,cols,icon){
 const shops=["The tavern","The apothecary","The baker","A merchant lord","The stable master","The tea house"];
 const goods={"🍎":"apples","🧪":"potions","🍯":"honey jars","🍇":"grapes","🥖":"loaves","🫖":"teapots","🍊":"oranges"};
 return `${shops[rand(0,shops.length-1)]} needs <b>${rows} rows</b> with <b>${cols} ${goods[icon]||"crates"}</b> in each row.`;
}
function nextOrder(){
 buildBoard();$("challengeBox").style.display="none";$("flagBtn").style.display="none";clearGhost();clearTimeout(hintTimer);
 if(node===1){
   order=makeOrder(rand(2,4),rand(3,5));order.text=`Pack <b>${order.rows} rows of ${order.cols}</b> crates.`;
   hintNext=true;
 }else if(node===2){
   order=makeOrder(rand(2,5),rand(3,6));order.text=wordProblem(order.rows,order.cols,order.icon=ICONS[rand(0,6)]);
   hintNext=false;$("readBtn").style.display="inline-block";
   hintTimer=setTimeout(showGhost,9000);
 }else{
   const rows=rand(2,5),cols=rand(3,6);order=makeOrder(rows,cols);order.icon=ICONS[rand(0,6)];
   if(orderIdx===garbledIdx){
     const wrong=cols+rand(1,2);
     order.garbled=true;order.shownCols=wrong;
     order.text=`⚠️ Rival's scribbled note: "<b>${rows} rows of ${wrong}</b>… total <b>${rows*cols}</b> crates."<br><i>Something doesn't add up…</i>`;
     $("flagBtn").style.display="inline-block";
   }else order.text=wordProblem(rows,cols,order.icon);
   hintNext=false;$("readBtn").style.display=node===3?"none":"inline-block";
 }
 $("orderText").innerHTML=order.text;setTray(order.icon);updateTicker();
 $("legInfo").textContent=legLabel();
}
function legLabel(){
 const names=["","Quiet Stall — Practice","Market Street — Race Leg","GRAND MARKET RUN"];
 return `${names[node]}  •  Order ${orderIdx+1}/${ordersLeft}`;
}
function userActed(){if(node===2&&!ghostOn){clearTimeout(hintTimer);hintTimer=setTimeout(showGhost,9000);}}

/* CHALLENGE garbled */
$("flagBtn").onclick=()=>{
 const box=$("challengeBox");box.style.display="block";
 const correct=order.cols,r=order.rows;
 const opts=[correct,order.shownCols,correct+1].sort(()=>Math.random()-.5);
 box.innerHTML=`If the total is <b>${r*correct}</b> in <b>${r}</b> rows, each row needs:`+
  opts.map(o=>`<button data-v="${o}">${o}</button>`).join("");
 box.querySelectorAll("button").forEach(b=>b.onclick=()=>{
   if(+b.dataset.v===correct){
     order.garbled=false;order.fixed=true;
     order.text=`✅ Order corrected: <b>${r} rows of ${correct}</b>.`;$("orderText").innerHTML=order.text;
     box.style.display="none";$("flagBtn").style.display="none";
     say("Rival: \"Hmph. Sharp eyes...\"");fanfare();
   }else{say("Rival: \"Heh, guess again, broker!\"");beep(180,.2);}
 });
};

$("readBtn").onclick=()=>{const t=$("orderText").textContent.replace(/⚠️|✅/g,"");speak(t);beep(880,.1,"sine",.1)};

/* SHIP */
$("shipBtn").onclick=()=>{
 if(!order)return;
 const counts=rowCounts();
 // contiguous check
 let ok=true;
 for(let r=0;r<gRows;r++){let seen=false;for(let c=gCols-1;c>=0;c--){if(grid[r][c])seen=true;else if(seen){ok=false}}}
 const filledRows=counts.filter(c=>c>0).length;
 const target=order.garbled?{rows:order.rows,cols:order.cols}:{rows:order.rows,cols:order.cols};
 if(ok&&filledRows===target.rows&&counts.slice(0,target.rows).every(c=>c===target.cols)&&counts.slice(target.rows).every(c=>c===0)&&!(order.garbled)){
   shipSuccess();
 }else if(order.garbled){
   // built the wrong (garbled) array or any array without fixing
   spill(true);
 }else spill(false);
};

function shipSuccess(){
 const n=order.rows*order.cols;streak++;
 let dist=n*(node===3?1.15:1.6);
 if(secondWind){dist*=1.5;secondWind=false;$("wind").textContent="";say("💨 SECOND WIND! Bonus speed!");}
 else if(streak>=2){secondWind=true;$("wind").textContent="💨 Second Wind ready!";}
 playerX+=dist;drawCarts();fanfare();
 const led=document.createElement("div");led.textContent=`✓ ${order.rows} × ${order.cols} = ${n} ${order.icon}${order.fixed?" (caught the trick!)":""}`;
 $("ledger").appendChild(led);$("ledger").scrollTop=9999;
 say(["Nice packing!","The caravan surges forward!","Customer delighted!","Wheels rolling!"][rand(0,3)]);
 orderIdx++;
 setTimeout(checkLeg,700);
}
function checkLeg(){
 if(playerX>=100){endLeg(true);return;}
 if(orderIdx>=ordersLeft){ if(playerX>rivalX)endLeg(true);else endLeg(false);return;}
 nextOrder();
}
function spill(garbledFail){
 streak=0;secondWind=false;$("wind").textContent="";
 beep(160,.3,"sawtooth",.2);
 // animate crates flying off wrong rows
 const counts=rowCounts();
 for(let r=0;r<gRows;r++)for(let c=0;c<gCols;c++){
   if(grid[r][c]){
     const s=slotEl(r,c),cr=s.querySelector(".crate");
     const rect=cr.getBoundingClientRect(),st=$("stage").getBoundingClientRect();
     const f=cr.cloneNode(true);f.className="crate spill";f.style.left=(rect.left-st.left)+"px";f.style.top=(rect.top-st.top)+"px";$("stage").appendChild(f);
     requestAnimationFrame(()=>{f.style.transform=`translate(${rand(-200,200)}px,${rand(120,300)}px) rotate(${rand(-360,360)}deg)`;f.style.opacity=0});
     setTimeout(()=>f.remove(),900);
     cr.remove();grid[r][c]=null;
   }
 }
 updateTicker();
 if(node>=2){rivalX+=6;drawCarts();}
 say(garbledFail? 'Rival: "HA! You packed my fake order! Read carefully next time!"':
   ["Rival: \"Sloppy stacking, broker!\"","Rival: \"That's not the order! Ha!\"","Rival: \"My grandmother packs better!\""][rand(0,2)],3200);
 if(node>=2&&rivalX>=100)endLeg(false);
}

/* RIVAL PACE */
function startRival(){
 stopRival();
 rivalTimer=setInterval(()=>{
   if(node===2)rivalX+=0.55;
   else if(node===3){const gap=playerX-rivalX;rivalX+=0.75+(gap>15?0.35:gap<-10?-0.15:0);}
   drawCarts();
   if(rivalX>=100){endLeg(false);}
 },400);
}
function stopRival(){clearInterval(rivalTimer);rivalTimer=null}

/* NODES / LEGS */
function startNode(n){
 node=n;resetLeg();orderIdx=0;streak=0;secondWind=false;$("wind").textContent="";
 $("readBtn").style.display="none";
 if(n===1){ordersLeft=2;garbledIdx=-1;}
 if(n===2){ordersLeft=3;garbledIdx=-1;startRival();}
 if(n===3){ordersLeft=5;garbledIdx=rand(1,3);startRival();$("dusk").style.opacity=1;}
 nextOrder();
}
function endLeg(won){
 stopRival();order=null;legWon.push(won);
 if(won)bonusGoods+=node*3;
 const ov=$("overlay");ov.style.display="flex";
 if(node===3){
   const grand=won;
   ov.innerHTML=`<h1>${grand?"🏆 GRAND MARKET RUN — WON!":"Gate Closed…"}</h1>
   <p>${grand?'Rival: "Fine, FINE! You\'re the sharpest packer on this route. This time."':'Rival: "Better luck tomorrow, broker! The bonus goods are MINE."'}</p>
   <p>Bonus goods earned today: <b>${bonusGoods} 💰</b></p>
   <div id="ovLedger">${$("ledger").innerHTML}</div>
   <button id="replayBtn">🔄 New Market Day (new orders!)</button>`;
   if(grand){$("banner").style.display="block";fanfare();setTimeout(fanfare,500);}
   $("replayBtn").onclick=()=>{$("ledger").innerHTML="<b>— Day Ledger —</b>";bonusGoods=0;legWon=[];$("dusk").style.opacity=0;ov.style.display="none";startNode(1)};
 }else{
   ov.innerHTML=`<h1>${won?"Leg Won! ✨":"Rival took this leg…"}</h1>
   <p>${won?`Bonus goods secured! (+${node*3} 💰) The market street grows busier ahead.`:"The bonus goods slip away — but the campaign rolls on. Nothing is lost for good."}</p>
   <p>${node===1?"Next: real customers with word-problem orders — and the rival starts moving!":"Next: the GRAND MARKET RUN. Back-to-back orders, an adapting rival, a closing gate at dusk… and one order the rival has tampered with. Stay sharp."}</p>
   <button id="nextBtn">${node===1?"Enter Market Street →":"Begin the Grand Run →"}</button>`;
   $("nextBtn").onclick=()=>{ov.style.display="none";startNode(node+1)};
 }
}
$("startBtn").onclick=()=>{$("overlay").style.display="none";startNode(1);beep(660,.15,"triangle",.15);};
buildBoard();
