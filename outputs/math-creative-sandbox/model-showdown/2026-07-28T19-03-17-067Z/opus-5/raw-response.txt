function makeOrder(node, idx){
  let rows,cols;
  if(node===0){ const set=[[3,4],[2,5],[4,3],[3,3],[2,6],[5,2]]; [rows,cols]=pick(set); }
  else if(node===1){ rows=rnd(2,6); cols=rnd(2,6); if(rows*cols<6){rows=3;cols=3;} }
  else { rows=rnd(3,7); cols=rnd(3,7); if(rows*cols>36){cols=clamp(cols,3,5);} }
  const good=pick(GOODS);
  const cust=pick(CUSTOMERS);
  const o={rows,cols,good,cust,total:rows*cols,garbled:false,mode:'plain'};
  if(node===0){
    o.text=`Pack <b>${rows} rows</b> of <b>${cols} ${good.u}</b>.`;
    o.spoken=`Pack ${rows} rows of ${cols} ${good.u}.`;
  } else {
    const forms=[
      ()=>{o.text=`${cust} needs <b>${rows} crates</b>, <b>${cols} ${good.u}</b> in each.`;o.spoken=`${cust} needs ${rows} crates, ${cols} ${good.u} in each.`;},
      ()=>{o.text=`Load <b>${rows} shelves</b> — every shelf holds <b>${cols} ${good.u}</b>.`;o.spoken=`Load ${rows} shelves. Every shelf holds ${cols} ${good.u}.`;},
      ()=>{o.text=`${cust} orders <b>${cols} ${good.u}</b> per crate for <b>${rows} crates</b>.`;o.spoken=`${cust} orders ${cols} ${good.u} per crate, for ${rows} crates.`;}
    ];
    pick(forms)();
  }
  return o;
}
function garble(o){
  // Rival tampers: stated total contradicts rows×cols
  const wrong = o.total + pick([-o.cols, o.cols, -1, 1, o.rows, -o.rows]).valueOf();
  const bad = (wrong===o.total|| wrong<1) ? o.total+o.cols : wrong;
  o.garbled=true; o.claimedTotal=bad;
  o.text=`${o.cust} needs <b>${o.rows} crates</b>, <b>${o.cols} ${o.good.u}</b> each — <b>total ${bad} ${o.good.u}</b>. <i>(stamped by V.)</i>`;
  o.spoken=`${o.cust} needs ${o.rows} crates, ${o.cols} ${o.good.u} each. Total ${bad} ${o.good.u}. Stamped by V.`;
  return o;
}
function buildNodeOrders(node){
  const list=[];
  if(node===0){ for(let i=0;i<3;i++) list.push(makeOrder(0,i)); }
  else if(node===1){ for(let i=0;i<4;i++) list.push(makeOrder(1,i)); }
  else {
    const n=6;
    for(let i=0;i<n;i++) list.push(makeOrder(2,i));
    // 1-2 garbled orders placed randomly (not first)
    const gcount = Math.random()<.45?2:1;
    const idxs=[]; while(idxs.length<gcount){const k=rnd(1,n-1); if(!idxs.includes(k))idxs.push(k);}
    idxs.forEach(k=>garble(list[k]));
    // one double-order leg
    const dbl=rnd(2,n-1);
    if(!list[dbl].garbled){
      list[dbl].mode='double';
      const b=makeOrder(2,dbl); b.rows=clamp(b.rows,2,4); b.cols=clamp(b.cols,2,4);
      list[dbl].second={rows:b.rows,cols:b.cols,good:b.good};
      list[dbl].text=`DOUBLE ORDER — pack <b>${list[dbl].rows}×${list[dbl].cols} ${list[dbl].good.u}</b>, then <b>${b.rows}×${b.cols} ${b.good.u}</b>, both before the gate.`;
      list[dbl].spoken=`Double order. Pack ${list[dbl].rows} rows of ${list[dbl].cols} ${list[dbl].good.u}, then ${b.rows} rows of ${b.cols} ${b.good.u}. Both before the gate.`;
    }
  }
  return list;
}

let drag=null;
function attachDrag(el){
  el.addEventListener('pointerdown',e=>{
    if(G.locked||el.classList.contains('placed'))return;
    ac();
    el.setPointerCapture(e.pointerId);
    const r=el.getBoundingClientRect(), sr=S.stage.getBoundingClientRect();
    drag={el,dx:e.clientX-r.left,dy:e.clientY-r.top,sr};
    el.classList.add('dragging');
    tone(660,.04,'sine',.05);
  });
  el.addEventListener('pointermove',e=>{
    if(!drag||drag.el!==el)return;
    el.style.left=(e.clientX-drag.sr.left-drag.dx)+'px';
    el.style.top=(e.clientY-drag.sr.top-drag.dy)+'px';
  });
  const end=e=>{
    if(!drag||drag.el!==el)return;
    el.classList.remove('dragging');
    dropCrate(el, e.clientX-drag.sr.left, e.clientY-drag.sr.top);
    drag=null;
  };
  el.addEventListener('pointerup',end);
  el.addEventListener('pointercancel',end);
}
function dropCrate(el,px,py){
  const sp=currentSpec();
  // find slot under center
  const cx=px+el.offsetWidth/2 - 0, cy=py+el.offsetHeight/2;
  let best=null,bd=1e9;
  for(let r=0;r<sp.rows;r++)for(let c=0;c<sp.cols;c++){
    if(G.placed[r][c])continue;
    const s=G.gridEls[r][c];
    const sx=376+parseFloat(s.style.left)+CELL/2, sy=206+parseFloat(s.style.top)+CELL/2;
    const d=Math.hypot(sx-cx,sy-cy);
    if(d<bd){bd=d;best={r,c,sx,sy};}
  }
  if(best && bd<CELL*0.95){
    snapIn(el,best,sp);
  } else {
    const [hx,hy]=el.dataset.home.split(',').map(Number);
    animTo(el,hx,hy,180);
    if(el.dataset.slot){ // was placed then removed
      const [r,c]=el.dataset.slot.split(',').map(Number);
      G.placed[r][c]=false; delete el.dataset.slot; el.classList.remove('placed');
      updateTallies();refreshPulse();updateHUD();
    }
  }
}
function snapIn(el,best,sp){
  if(el.dataset.slot){const [pr,pc]=el.dataset.slot.split(',').map(Number);G.placed[pr][pc]=false;}
  G.placed[best.r][best.c]=true;
  el.dataset.slot=best.r+','+best.c;
  el.classList.add('placed');
  const tx=376+parseFloat(G.gridEls[best.r][best.c].style.left)+(CELL-el.offsetWidth)/2;
  const ty=206+parseFloat(G.gridEls[best.r][best.c].style.top)+(CELL-el.offsetHeight)/2;

  clickSnap();

  updateTallies();refreshPulse();updateHUD();
  // auto-complete detection
  const sp2=currentSpec();
  let full=true;for(let r=0;r<sp2.rows;r++)for(let c=0;c<sp2.cols;c++)if(!G.placed[r][c])full=false;

function animTo(el,x,y,ms,bounce){
  const sx=parseFloat(el.style.left),sy=parseFloat(el.style.top),t0=performance.now();
  function step(t){
    let k=clamp((t-t0)/ms,0,1); let e=ease(k);
    let ox=sx+(x-sx)*e, oy=sy+(y-sy)*e;
    if(bounce&&k>0.7){ oy += Math.sin((k-0.7)/0.3*Math.PI)* -7; }
    el.style.left=ox+'px';el.style.top=oy+'px';
    if(k<1)requestAnimationFrame(step);
  }
  requestAnimationFrame(step);
}

function shipIt(){
  if(G.locked)return;
  const o=G.order, sp=currentSpec();
  if(o.garbled){
    // shipping a garbled order = failure
    fail("You shipped Vosk's tampered slip. The buyer counted "+ (sp.rows*sp.cols) +", the slip said "+o.claimedTotal+".", true);
    return;
  }
  const n=totalPlaced();
  const need=sp.rows*sp.cols;
  // check every row equal & full
  let ok = (n===need);
  if(ok){for(let r=0;r<sp.rows;r++)if(G.placed[r].filter(Boolean).length!==sp.cols)ok=false;}
  if(!ok){
    const rowsDone=G.placed.filter(r=>r.filter(Boolean).length===sp.cols).length;
    fail(`The cart holds ${n}. The order needs ${sp.rows} × ${sp.cols} = ${need}.`,false,rowsDone);
    return;
  }
  succeed();
}
function rejectOrder(){
  if(G.locked)return;
  const o=G.order;
  if(o.garbled){
    G.locked=true;
    G.ledger.push(`⚠ Caught bad slip: ${o.rows}×${o.cols}=${o.rows*o.cols}, not ${o.claimedTotal}`);
    G.streak++; if(G.streak>=2)G.secondWind=true;
    taunt("Tsk. You actually checked the total.",2400);
    fanfare();
    bigBanner(`SLIP REJECTED — ${o.rows} × ${o.cols} = ${o.rows*o.cols}, not ${o.claimedTotal}`, '#2d6b2d');
    advancePlayer((o.rows*o.cols)*1.15, true);
    setTimeout(()=>{G.locked=false;nextOrder();},1500);
  } else {
    G.locked=true;
    taunt("That slip was honest, trader. You just wasted a turn.",2600);
    tone(180,.4,'sawtooth',.1,-80);
    G.streak=0;G.secondWind=false;
    G.ledger.push(`✗ Rejected a good order (${o.rows}×${o.cols})`);
    flashRect('rgba(180,60,30,.35)');
    advanceRival(1.3);
    setTimeout(()=>{G.locked=false;updateHUD();},1100);
  }
  updateHUD();
}

let rivalDrip=0;
function rivalTick(dt){
  if(G.node<2||G.locked||G.finished)return;
  // adaptive: if player leads, rival speeds up slightly; if behind, eases
  const lead=G.playerX-G.rivalX;
  const rate=0.0085 + clamp(lead,0,.4)*0.02 - clamp(-lead,0,.3)*0.012;
  G.rivalX=clamp(G.rivalX+rate*dt,0,1);
  rivalDrip+=dt;
  if(rivalDrip>9){rivalDrip=0; if(G.rivalX>G.playerX && Math.random()<.6) taunt(pick(TAUNT_LEAD),2000);}
  checkGate();
}

  document.getElementById('btnReject').style.display=G.node===2?'inline-block':'none';
  drawBG();
}
function startNode(n){
  G.node=n;G.finished=false;G.sub=0;G.ghostShown=false;
  G.playerX=0.06;G.rivalX= n===0?0.06:0.06;
  G.orders=buildNodeOrders(n);G.legIndex=0;
  G.streak=0;G.secondWind=false;
  setNodeChrome();
  nextOrder(true);
}
function nextOrder(first){
  if(G.finished)return;
  if(G.legIndex>=G.orders.length){ G.finished=true; setTimeout(endNode,400); return; }
  G.order=G.orders[G.legIndex++];
  G.sub=0;
  clearBoardDom();
  buildGrid();
  const ot=document.getElementById('orderText');
  ot.innerHTML=G.order.text;
  document.getElementById('doubleTag').textContent = G.order.mode==='double'?'DOUBLE':'';
  const meta=document.getElementById('orderMeta');
  meta.innerHTML = G.node===0
    ? `Order ${G.legIndex} of ${G.orders.length} · take your time — no clock yet.`
    : `Order ${G.legIndex} of ${G.orders.length} · Vosk is packing too.`;
  document.getElementById('tickerBig').textContent='—';
  if(G.node>=1) speak(G.order.spoken);
  const t=G.node===0?0:(G.node===1?46:34 - Math.min(8,G.legIndex*1.2));
  startTimer(t);
  updateHUD();
}
function showGhost(){
  if(G.node!==1)return;
  const sp=currentSpec();
  for(let r=0;r<sp.rows;r++)for(let c=0;c<sp.cols;c++){
    const s=G.gridEls[r][c];
    s.style.transition='background .3s';
    s.style.background='rgba(255,230,150,.42)';
    setTimeout(()=>{s.style.background='';},1800);
  }
  bigBanner(`Ghost: ${sp.rows} rows × ${sp.cols} in each row`,'#1d6ea8');
  tone(700,.2,'sine',.07);
}

   END OF NODE
   ============================================================ */
function endNode(){
  const won = G.playerX>=G.rivalX;
  if(won)G.legsWon++;else G.legsLost++;
  if(G.node<2){
    const bonus= won? '+1 crate of bonus goods' : 'bonus goods lost this leg';
    overlay(`
      <h1>${won?'Leg Won — Gate reached first!':'Vosk reached the gate first.'}</h1>
      <p>${won?'Your caravan rolls through with room to spare. '+bonus+'.':'Vosk claims the leg bonus. Your caravan still rolls on — the campaign continues.'}</p>
      <h2>What you packed</h2>
      <ul>${G.ledger.slice(-6).map(x=>'<li>'+x+'</li>').join('')}</ul>
      <p class="center"><button class="mk good" id="nx" style="font-size:19px;padding:12px 26px;">Next Market Day →</button></p>
    `,()=>{document.getElementById('nx').onclick=()=>{closeOverlay();startNode(G.node+1);};});
  } else {
    G.masterBanner = won;
    if(won)fanfare();
    const arrays=G.ledger.filter(x=>x.startsWith('✓'));
    overlay(`
      <h1>${won?'MASTER TRADER':'The Gate Closes'}</h1>
      <p>${won
        ? 'Vosk hauls his reins and calls out: <i>“Fine. Fine! You out-counted me, trader. Rows and columns — I underestimated the arithmetic in your cart. The banner is yours.”</i>'
        : 'Vosk slides through the gate first: <i>“Bragging rights, mine. Bonus goods, mine. Come back tomorrow — I like a rematch.”</i> Nothing is lost but pride.'}</p>
      <h2>Day Ledger — every array you built</h2>
      <div id="ledgerFinal">${G.ledger.map(x=>'<div>'+x+'</div>').join('')}</div>
      <p style="margin-top:10px;">Legs won: <b>${G.legsWon}</b> · Perfect arrays: <b>${arrays.length}</b></p>
      <p class="center">
        <button class="mk good" id="again" style="font-size:19px;padding:12px 26px;">Run the Route Again (new orders)</button>
        <button class="mk" id="fromStart" style="font-size:19px;padding:12px 26px;">Back to Day 1</button>
      </p>
    `,()=>{
      document.getElementById('again').onclick=()=>{closeOverlay();G.ledger=[];G.masterBanner=won;startNode(2);};
      document.getElementById('fromStart').onclick=()=>{closeOverlay();G.ledger=[];G.masterBanner=false;G.legsWon=0;G.legsLost=0;startNode(0);};
    });
  }
}

  if(performance.now()<spinUntil) G.wheelSpin+=dt*7; else G.wheelSpin+=dt*0.6;
  tickTimer(dt); rivalTick(dt);
  drawRace();

  requestAnimationFrame(loop);
}
