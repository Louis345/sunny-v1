<script>
function showScreen(id){
  document.querySelectorAll('.screen').forEach(s=>s.classList.add('hidden'));
  document.getElementById(id).classList.remove('hidden');
}

// ---- game state ----
// crateCounts represent gears currently in each crate. Crate index 3 (row 4)
// starts short (rushed load) — demonstrates the incorrect-consequence beat.
let crateCounts = [7,7,7,6,7,7];
let poolCount = 1; // the "missing" gear sits loose in the pool
let selectedGearFromPool = false;
const NEEDED = 7;

function startLoad(){
  showScreen('scr-load');
  renderAll();
}

function renderAll(){
  renderCrates();
  renderPool();
  document.getElementById('flagMsg').textContent = '';
}

function renderCrates(){
  const wrap = document.getElementById('crates');
  wrap.innerHTML = '';
  crateCounts.forEach((count, idx)=>{
    const div = document.createElement('div');
    div.className = 'crate';
    div.id = 'crate'+idx;
    div.innerHTML = `<div class="crate-label">Crate ${idx+1}</div>
      <div class="crate-count" id="cnt${idx}">${count} / ${NEEDED} ⚙️</div>
      <div class="gearRow" id="gr${idx}"></div>`;
    div.addEventListener('click', (e)=>{
      if(e.target.classList.contains('gear')) return; // handled separately
      onCrateClick(idx);
    });
    wrap.appendChild(div);
    const gr = div.querySelector('.gearRow');
    for(let i=0;i<count;i++){
      const g = document.createElement('span');
      g.className='gear';
      g.textContent='⚙️';
      g.title='Click to send this gear back to the pool';
      g.addEventListener('click', (ev)=>{
        ev.stopPropagation();
        crateCounts[idx]--; poolCount++; renderAll();
      });
      gr.appendChild(g);
    }
  });
}

function renderPool(){
  const p = document.getElementById('pool');
  p.innerHTML='';
  for(let i=0;i<poolCount;i++){
    const g = document.createElement('span');
    g.className='gear';
    g.textContent='⚙️';
    g.addEventListener('click', ()=>{
      document.querySelectorAll('#pool .gear').forEach(el=>el.classList.remove('selected'));
      g.classList.add('selected');
      selectedGearFromPool = true;
    });
    p.appendChild(g);
  }
  if(poolCount===0){
    const s=document.createElement('span');
    s.style.color='var(--sub)';
    s.textContent='(none loose right now)';
    p.appendChild(s);
  }
}

function onCrateClick(idx){
  if(!selectedGearFromPool || poolCount<=0) return;
  crateCounts[idx]++;
  poolCount--;
  selectedGearFromPool = false;
  renderAll();
}

function tryShip(){
  const badIdx = crateCounts.findIndex(c => c !== NEEDED);
  if(badIdx === -1){
    document.getElementById('shipBtn').disabled = true;
    runSuccessSequence();
    return;
  }
  // ---- INCORRECT CONSEQUENCE ----
  const crateEl = document.getElementById('crate'+badIdx);
  crateEl.classList.add('toppled');
  const had = crateCounts[badIdx];
  document.getElementById('flagMsg').textContent =
    `📻 Dispatch: "Whoa — hold up. Row ${badIdx+1} only has ${had} gears, not 7. `+
    `Don't send the whole load out — just rebuild that row."`;
  setTimeout(()=>{
    poolCount += crateCounts[badIdx];
    crateCounts[badIdx] = 0;
    crateEl.classList.remove('toppled');
    renderAll();
    document.getElementById('flagMsg').textContent =
      `Row ${badIdx+1} reset — add gears from the pool until it reads 7 / 7, then ship it again.`;
  }, 700);
}

// ---- SUCCESS SEQUENCE: skip-count + world reaction + preview ----
function runSuccessSequence(){
  showScreen('scr-success');
  const skipEl = document.getElementById('skipLine');
  const totalEl = document.getElementById('totalLine');
  skipEl.textContent = '';
  totalEl.textContent = '';
  document.getElementById('reactionWrap').style.display='none';
  document.getElementById('roadmap').innerHTML='';

  let running = 0;
  let step = 0;
  const timer = setInterval(()=>{
    running += 7;
    step++;
    skipEl.textContent += (step>1 ? ', ' : '') + running;
    if(step === 6){
      clearInterval(timer);
      setTimeout(()=>{
        totalEl.textContent = `Radioing total... 6 crates confirmed on screen → 42 gears ✅`;
        setTimeout(()=>{
          document.getElementById('reactionWrap').style.display='block';
          buildRoadmap();
        }, 500);
      }, 400);
    }
  }, 550);
}

function buildRoadmap(){
  const data = [
    {t:'Tier 1', d:'Equal groups, single crates.', cur:true},
    {t:'Tier 2', d:'Arrays — same crates, rows & columns swapped.', cur:false},
    {t:'Tier 3', d:'A quiet audit: check a vendor\'s manifest, no rival watching.', cur:false},
    {t:'Tier 4', d:'Two-step riddles: "4 pallets of 6 rows of 3."', cur:false},
    {t:'Finale', d:'Rival\'s total is correct — but his grouping lies. Prove it.', cur:false},
  ];
  const wrap = document.getElementById('roadmap');
  data.forEach(item=>{
    const div = document.createElement('div');
    div.className = 'tier' + (item.cur ? ' current' : '');
    div.innerHTML = `<h3>${item.t} ${item.cur? '✅' : '<span class="lockicon">🔒</span>'}</h3><div>${item.d}</div>`;
    wrap.appendChild(div);
  });
}
