function playTone(freq,dur,type,delay){
  type=type||'sine'; delay=delay||0;
  try{
    ensureAudio(); if(!actx) return;
    const t=actx.currentTime+delay;
    const osc=actx.createOscillator(); const gain=actx.createGain();
    osc.type=type; osc.frequency.value=freq;
    gain.gain.setValueAtTime(0.0001,t);
    gain.gain.exponentialRampToValueAtTime(0.22,t+0.01);
    gain.gain.exponentialRampToValueAtTime(0.0001,t+dur);
    osc.connect(gain); gain.connect(actx.destination);
    osc.start(t); osc.stop(t+dur+0.02);
  }catch(e){}
}

function newTaskForStage(idx){
  if(idx===0){
    const groups=rand(3,5), perGroup=rand(3,6);
    return {type:'groups', groups, perGroup, total:groups*perGroup, filled:Array(groups).fill(0)};
  }
  if(idx===1){
    const rows=rand(3,5), cols=rand(3,6);
    return {type:'array', rows, cols, total:rows*cols, grid:Array(rows).fill(0).map(()=>Array(cols).fill(false))};
  }
  if(idx===2){
    const groups=rand(3,6), perGroup=rand(3,7);
    return {type:'missingFactor', groups, total:groups*perGroup, layers:0, trail:[]};
  }
}

/* ============ MATH READOUT ============ */
function updateMathReadout(task){
  const l1=document.getElementById('readoutLine1'), l2=document.getElementById('readoutLine2');
  if(task.type==='groups'||task.type==='finale'){
    const sum=task.filled.reduce((a,b)=>a+b,0);
    l1.textContent = 'Total energy cells: '+sum;
    const unit=task.filled[0];
    const allEq = task.filled.every(c=>c===unit) && unit>0;
    l2.textContent = allEq ? (task.filled.length+' × '+unit+' = '+sum) : '';
  } else if(task.type==='array'){
    let filled=0, rowsFull=0;
    task.grid.forEach(row=>{ const c=row.filter(Boolean).length; filled+=c; if(c===task.cols) rowsFull++; });
    l1.textContent = 'Filled cells: '+filled+' / '+task.total;
    l2.textContent = rowsFull>0 ? (rowsFull+' row'+(rowsFull>1?'s':'')+' × '+task.cols+' = '+(rowsFull*task.cols)) : '';
  } else if(task.type==='missingFactor'){
    l1.textContent = 'Layers added: '+task.layers+'  (target '+task.total+' cells)';
    l2.textContent = task.groups+' × '+task.layers+' = '+(task.groups*task.layers);
  }
}
function sentenceFor(task){
  if(task.type==='groups'||task.type==='finale') return task.groups+' × '+task.perGroup+' = '+task.total;
  if(task.type==='array') return task.rows+' × '+task.cols+' = '+task.total;
  if(task.type==='missingFactor') return task.groups+' × '+task.layers+' = '+task.total;
}

/* ============ FINALE TIMER ============ */
function startFinaleTimer(){
  clearInterval(finaleInterval);
  if(state.task.timeLeft==null) state.task.timeLeft=40;
  const bar=document.getElementById('timerBarInner'); if(bar) bar.style.width='100%';
  finaleInterval=setInterval(()=>{
    if(!state.task || state.task.type!=='finale'){ clearInterval(finaleInterval); return; }
    state.task.timeLeft--;
    const pct=Math.max(0,state.task.timeLeft/40*100);
    const b=document.getElementById('timerBarInner'); if(b) b.style.width=pct+'%';
    if(state.task.timeLeft<=0){
      onFail('Core stability depleted! The rival surges forward!', null);
      if(state.task && state.task.type==='finale') state.task.timeLeft=40;
    }
  },1000);
}

/* ============ SUBMIT / VALIDATION ============ */
function submit(){
  playClick();
  const task=state.task;
  let ok=false, msg='', highlight=null;
  if(task.type==='groups'||task.type==='finale'){
    const bad=[];
    task.filled.forEach((c,i)=>{ if(c!==task.perGroup) bad.push(i); });
    ok = bad.length===0;
    if(!ok){
      msg = bad.map(i=>'Row '+(i+1)+' has '+task.filled[i]+' cell'+(task.filled[i]!==1?'s':'')+', needs '+task.perGroup).join('  •  ');
      highlight={type:'pods', indices:bad};
    }
  } else if(task.type==='array'){
    let filled=0; const bad=[];
    task.grid.forEach((row,ri)=>{ const c=row.filter(Boolean).length; filled+=c; if(c!==task.cols) bad.push(ri); });
    ok = bad.length===0 && filled===task.total;
    if(!ok){
      if(bad.length){
        msg = bad.map(ri=>{ const c=task.grid[ri].filter(Boolean).length; return 'Row '+(ri+1)+' has '+c+' of '+task.cols; }).join('  •  ');
      } else { msg = 'Filled '+filled+', need '+task.total+'.'; }
      highlight={type:'rows', indices:bad};
    }
  } else if(task.type==='missingFactor'){
    const total = task.layers*task.groups;
    ok = total===task.total;
    if(!ok){
      msg = task.groups+' rows × '+task.layers+' layers = '+total+', but the shield needs '+task.total+'. '+(total<task.total?'Add more layers!':'Remove a few layers!');
    }
  }
  if(ok) onSuccess(); else onFail(msg, highlight);
}

function onSuccess(){
  const task=state.task, idx=state.stageIndex;
  playSuccessChime();
  feedbackEl.textContent = '✅ Shield charged! '+sentenceFor(task);
  feedbackEl.className='feedback good';
  burstConfettiSmall();
  if(idx<3){
    flashShieldSuccess(idx);
    state.shieldsCharged[idx]=true;
    updateShieldIcons();
  } else {
    state.lastFinaleSentence = sentenceFor(task);
    const core=document.getElementById('coreIcon');
    core.classList.add('pulse'); setTimeout(()=>core.classList.remove('pulse'),800);
    clearInterval(finaleInterval);
  }
  state.rivalPos = Math.max(0, state.rivalPos-1);
  updateRivalTrack();
  if(idx<3){
    setTimeout(()=>{ state.stageIndex++; loadStage(); }, 1500);
  } else {
    setTimeout(()=> triggerVictory(), 1200);
  }
}

function onFail(msg, highlight){
  playFailBuzz();
  feedbackEl.textContent = '⚠ '+msg;
  feedbackEl.className='feedback bad';
  crackShieldEffect();
  applyHighlight(highlight);
  state.rivalPos = Math.min(state.maxRival, state.rivalPos+1);
  updateRivalTrack();
  if(state.rivalPos >= state.maxRival) triggerBreach();
}
