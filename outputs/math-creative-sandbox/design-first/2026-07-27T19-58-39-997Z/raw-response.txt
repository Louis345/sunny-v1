(function(){
  "use strict";

  // ---------- AUDIO ----------
  let audioReady = false;
  let successSynth, errorSynth, chordSynth;
  function initAudio(){
    if(audioReady || typeof Tone === 'undefined') return;
    try{
      Tone.start();
      successSynth = new Tone.Synth({oscillator:{type:'triangle'}, envelope:{attack:0.005,decay:0.15,sustain:0.05,release:0.2}}).toDestination();
      errorSynth = new Tone.MembraneSynth({pitchDecay:0.03, octaves:3}).toDestination();
      chordSynth = new Tone.PolySynth(Tone.Synth).toDestination();
      audioReady = true;
    }catch(e){ audioReady=false; }
  }
  function playNote(freq, time){
    if(!audioReady) return;
    try{ successSynth.triggerAttackRelease(freq, '8n', time); }catch(e){}
  }
  function playError(){
    if(!audioReady) return;
    try{ errorSynth.triggerAttackRelease('C2','8n'); }catch(e){}
  }
  function playVictoryChord(){
    if(!audioReady) return;
    try{
      chordSynth.triggerAttackRelease(['C4','E4','G4','C5'],'2n');
    }catch(e){}
  }

  // ---------- RIVAL PERSONAS ----------
  const PERSONAS = [
    {name:'Vex the Breaker', color:'#ff5f6d'},
    {name:'Nyx the Shade', color:'#b266ff'},
    {name:'Torque', color:'#ff9d3b'},
  ];
  const ADVANCE_LINES = ["The rival surges forward!","A crack forms in the wall!","The rival pushes closer!"];
  const RETREAT_LINES = ["The rival recalculates.","Systems interference pushes it back!","The rival falls back, confused."];

  // ---------- STATE ----------
  const state = {
    levels: [], levelIndex:0, shieldsRestored:0,
    rivalPos:0, rivalMax:5, persona:null, attempted:false
  };

  function rand(min,max){ return Math.floor(Math.random()*(max-min+1))+min; }

  function generateLevels(){
    const l1 = {type:'groups', bins: rand(3,4), per: rand(3,5)};
    const l2 = {type:'array',  bins: rand(3,5), per: rand(4,6)};
    const l3 = {type:'missing',bins: rand(3,5), per: rand(4,7)};
    const l4 = {type:'finale', bins: rand(4,6), per: rand(5,7)};
    return [l1,l2,l3,l4].map(l => ({...l, total: l.bins*l.per}));
  }

  function storyFor(level){
    switch(level.type){
      case 'groups': return `Scout Alert! Split the energy cells into ${level.bins} equal groups of ${level.per} to charge Shield Alpha.`;
      case 'array':  return `Array Sequence! Build ${level.bins} rows with ${level.per} cells in every row — a perfect rectangle — to charge Shield Beta.`;
      case 'missing':return `Power Puzzle! Shield Gamma needs exactly ${level.total} energy cells spread EVENLY across ${level.bins} rows. Figure out how many belong in each row!`;
      case 'finale': return `Core Ignition! Form the grand array — ${level.bins} rows with ${level.per} cells in each — to fully activate the Arena Core and win!`;
    }
  }

  function revealSentence(level, required){
    if(level.type==='missing'){
      return `${level.bins} × ${required} = ${level.total} — you found the missing factor!`;
    }
    if(level.type==='finale'){
      return `${level.bins} × ${level.per} = ${level.total}. THE CORE IS ACTIVE!`;
    }
    return `${level.bins} × ${level.per} = ${level.total}. Shield charged!`;
  }

  // ---------- DOM refs ----------
  const el = id => document.getElementById(id);
  const supplyTray = () => el('supplyTray');
  const lanesArea = () => el('lanesArea');

  // ---------- BACKGROUND DECOR ----------
  function buildStars(){
    const bg = el('bgFx');
    for(let i=0;i<70;i++){
      const s = document.createElement('div');
      s.className='star';
      const size = Math.random()*2+1;
      s.style.width=size+'px'; s.style.height=size+'px';
      s.style.left=Math.random()*100+'%';
      s.style.top=Math.random()*70+'%';
      s.style.animationDelay=(Math.random()*3)+'s';
      bg.appendChild(s);
    }
  }

  // ---------- RIVAL TRACK ----------
  function buildTrackNodes(){
    const track = el('rivalTrack');
    [...track.querySelectorAll('.trackNode')].forEach(n=>n.remove());
    for(let i=0;i<=state.rivalMax;i++){
      const n = document.createElement('div');
      n.className='trackNode';
      n.style.left = (i/state.rivalMax*100)+'%';
      track.appendChild(n);
    }
  }
  function renderRivalPosition(instant){
    const pct = (state.rivalPos/state.rivalMax*100);
    const icon = el('rivalIcon');
    if(instant){ icon.style.left = pct+'%'; }
    else if(window.gsap){ gsap.to(icon, {left: pct+'%', duration:0.6, ease:'power2.out'}); }
    else icon.style.left = pct+'%';
  }
  function setPersona(p){
    state.persona = p;
    el('rivalName').textContent = p.name;
    el('rivalIcon').style.background = `radial-gradient(circle,${p.color},#3a0000 80%)`;
    el('rivalIcon').style.color = p.color;
  }

  function advanceRival(delta){
    state.rivalPos = Math.max(0, Math.min(state.rivalMax, state.rivalPos+delta));
    renderRivalPosition(false);
    const msg = delta>0 ? ADVANCE_LINES[rand(0,ADVANCE_LINES.length-1)] : RETREAT_LINES[rand(0,RETREAT_LINES.length-1)];
    el('rivalLabel').lastChild; // noop
    flashRivalLine(msg);
    if(state.rivalPos>=state.rivalMax){
      setTimeout(triggerBreach, 500);
    }
  }
  let rivalLineTimeout;
  function flashRivalLine(text){
    clearTimeout(rivalLineTimeout);
    el('rivalLabel').innerHTML = `Rival: <span id="rivalName" style="color:${state.persona.color}">${state.persona.name}</span> — <em style="color:#ffd">${text}</em>`;
    rivalLineTimeout = setTimeout(()=>{
      el('rivalLabel').innerHTML = `Rival: <span id="rivalName" style="color:${state.persona.color}">${state.persona.name}</span>`;
    }, 2200);
  }

  // ---------- CELL CREATION & DRAG ----------
  function createCell(){
    const c = document.createElement('div');
    c.className='cell';
    return c;
  }

  function onPointerDown(e){
    const cell = e.target.closest('.cell');
    if(!cell) return;
    e.preventDefault();
    const rect = cell.getBoundingClientRect();
    const offsetX = e.clientX - rect.left;
    const offsetY = e.clientY - rect.top;
    cell.classList.add('dragging');
    cell.style.left = rect.left+'px';
    cell.style.top = rect.top+'px';
    cell.style.width = rect.width+'px';
    cell.style.height = rect.height+'px';
    document.body.appendChild(cell);

    function move(ev){
      cell.style.left = (ev.clientX - offsetX)+'px';
      cell.style.top = (ev.clientY - offsetY)+'px';
    }
    function up(ev){
      document.removeEventListener('pointermove', move);
      document.removeEventListener('pointerup', up);
      cell.classList.remove('dragging');
      cell.style.left=''; cell.style.top=''; cell.style.width=''; cell.style.height='';
      const target = document.elementFromPoint(ev.clientX, ev.clientY);
      const laneHolder = target && target.closest('.lane-holder');
      const tray = target && target.closest('#supplyTray');
      if(laneHolder){ laneHolder.appendChild(cell); pulseCell(cell); }
      else if(tray){ tray.appendChild(cell); }
      else { supplyTray().appendChild(cell); }
      updateLiveCounter();
    }
    document.addEventListener('pointermove', move);
    document.addEventListener('pointerup', up);
  }
  el2Listen();
  function el2Listen(){
    document.getElementById('gameContainer').addEventListener('pointerdown', onPointerDown);
  }

  function pulseCell(cell){
    if(window.gsap){
      gsap.fromTo(cell, {scale:1.35}, {scale:1, duration:0.28, ease:'back.out(3)'});
    }
  }

  // ---------- LEVEL RENDERING ----------
  function renderLevel(idx){
    state.levelIndex = idx;
    state.attempted = false;
    const level = state.levels[idx];
    supplyTray().innerHTML='';
    lanesArea().innerHTML='';
    el('feedbackMsg').textContent='';
    el('storyText').textContent = storyFor(level);
    el('liveCounter').textContent='';

    for(let i=0;i<level.bins;i++){
      const lane = document.createElement('div');
      lane.className='lane';
      lane.dataset.index = i;
      const label = document.createElement('div');
      label.className='lane-label';
      label.textContent = (level.type==='groups' ? 'Group ' : 'Row ') + (i+1);
      const holder = document.createElement('div');
      holder.className='lane-holder';
      const badge = document.createElement('div');
      badge.className='lane-badge';
      badge.textContent='0';
      lane.appendChild(label);
      lane.appendChild(holder);
      lane.appendChild(badge);
      lanesArea().appendChild(lane);
    }
    for(let i=0;i<level.total;i++){
      supplyTray().appendChild(createCell());
    }
    el('chargeBtn').textContent = level.type==='finale' ? '🔆 Activate Core' : '⚡ Charge Shield';
    updateLiveCounter();
  }

  function updateLiveCounter(){
    const level = state.levels[state.levelIndex];
    if(!level) return;
    const lanes = [...lanesArea().querySelectorAll('.lane')];
    const counts = lanes.map(l => l.querySelector('.lane-holder').children.length);
    const required = level.total / level.bins;
    lanes.forEach((l,i)=>{
      l.querySelector('.lane-badge').textContent = counts[i];
      if(state.attempted){
        l.classList.toggle('mismatch', counts[i] !== required);
      }
    });
    const filled = counts.filter(c=>c>0).length;
    const allEqualNonZero = counts.every(c=>c===counts[0]) && counts[0]>0;
    let txt = '';
    if(allEqualNonZero){
      const seq = [];
      for(let i=1;i<=filled;i++) seq.push(counts[0]*i);
      txt = 'Skip counting: ' + seq.join(' → ');
    }
    el('liveCounter').textContent = txt;
  }

  // ---------- VALIDATION ----------
  function onCharge(){
    initAudio();
    const level = state.levels[state.levelIndex];
    const lanes = [...lanesArea().querySelectorAll('.lane')];
    const counts = lanes.map(l => l.querySelector('.lane-holder').children.length);
    const supplyLeft = supplyTray().children.length;
    const required = level.total / level.bins;
    const allEqual = counts.every(c => c === counts[0]) && counts[0] > 0;
    const isValid = supplyLeft === 0 && allEqual && counts[0] === required;

    if(isValid){
      handleSuccess(level, lanes, counts, required);
    } else {
