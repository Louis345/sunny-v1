function rand(min,max){ return Math.floor(Math.random()*(max-min+1))+min; }
function shuffle(arr){ for(let i=arr.length-1;i>0;i--){const j=rand(0,i);[arr[i],arr[j]]=[arr[j],arr[i]];} return arr; }

function updateScorePill(){ document.getElementById('scorePill').textContent = '⭐ ' + state.stars; }
function addStar(n=1){ state.stars += n; updateScorePill(); }

function updateBeltbar(){
  const bar = document.getElementById('beltbar');

  }
}

/* ============ ROUND 1: GROUPS ============ */
let groupState = { size:0, groups:[], targetSize:0, targetCount:0, targetTotal:0, round:0 };

function initGroups(){
  groupState = { size:0, groups:[], targetSize:0, targetCount:0, targetTotal:0, round:0 };
  document.getElementById('groupQuizCard').style.display='none';
  document.getElementById('groupsNextBtn').style.display='none';
  const sizes = [2,3,4,5];
  const wrap = document.getElementById('groupSizeButtons');
  wrap.innerHTML='';
  sizes.forEach(s=>{
    const b = document.createElement('button');
    b.className='btn secondary';
    b.textContent = 'Team of ' + s;
    b.onclick = ()=>selectGroupSize(s);
    wrap.appendChild(b);
  });
  renderGroups();
  startGroupRound();
}
function selectGroupSize(s){ groupState.size = s; addGroup(); }
function addGroup(){
  if(!groupState.size){ speak('wrong'); document.getElementById('bubble').textContent='Pick a team size first!'; return; }
  groupState.groups.push(groupState.size);
  renderGroups();
}
function undoGroup(){ groupState.groups.pop(); renderGroups(); }
function resetGroups(){ groupState.groups = []; renderGroups(); }
function renderGroups(){
  const area = document.getElementById('groupsArea');
  area.innerHTML='';
  groupState.groups.forEach(sz=>{
    const box = document.createElement('div');
    box.className='group-box';
    for(let i=0;i<sz;i++){ const s=document.createElement('span'); s.textContent='🤼'; box.appendChild(s); }
    area.appendChild(box);
  });
  const n = groupState.groups.length;
  const size = groupState.groups[0] || 0;
  const total = groupState.groups.reduce((a,b)=>a+b,0);
  if(n===0){
    document.getElementById('groupSentence').textContent = 'Add a team to begin!';
  } else {
    const skip = [];
    for(let i=1;i<=n;i++) skip.push(size*i);
    document.getElementById('groupSentence').textContent =
      `${n} × ${size} = ${total}   (skip count: ${skip.join(', ')})`;
  }
}
function startGroupRound(){
  groupState.round++;
  if(groupState.round > 3){
    document.getElementById('groupQuizCard').style.display='none';
    document.getElementById('groupsNextBtn').style.display='inline-block';
    speak('levelup');
    return;
  }
  const size = rand(2,6);
  const count = rand(2,5);
  groupState.targetSize = size; groupState.targetCount = count; groupState.targetTotal = size*count;
  document.getElementById('groupQuizCard').style.display='block';
  document.getElementById('groupQTitle').textContent = `Challenge ${groupState.round} of 3`;
  const scenarios = [
    `Coach Rex lines up ${count} wrestling teams with ${size} wrestlers on each team. How many wrestlers in all?`,
    `There are ${count} training mats, and ${size} wrestlers practice on each mat. How many wrestlers total?`,
    `${count} gyms send ${size} athletes each to the tournament. How many athletes come altogether?`
  ];
  document.getElementById('groupQText').textContent = scenarios[rand(0,scenarios.length-1)];
  document.getElementById('groupAnswerInput').value='';
  document.getElementById('groupFeedback').textContent='';
  document.getElementById('groupFeedback').className='feedback';
}
function checkGroupAnswer(){
  const v = parseInt(document.getElementById('groupAnswerInput').value,10);
  const fb = document.getElementById('groupFeedback');
  if(v === groupState.targetTotal){
    fb.textContent = `Correct! ${groupState.targetCount} × ${groupState.targetSize} = ${groupState.targetTotal} 🎉`;
    fb.className='feedback good';
    addStar(1);
    speak('correct');
    setTimeout(startGroupRound, 1300);
  } else {
    fb.textContent = `Not quite — try skip counting by ${groupState.targetSize}, ${groupState.targetCount} times.`;
    fb.className='feedback bad';
    speak('wrong');
  }
}

/* ============ ROUND 2: ARRAYS ============ */
let arrayState = { rows:3, cols:4, round:0, targetRows:0, targetCols:0 };
function initArrays(){
  arrayState = { rows:3, cols:4, round:0, targetRows:0, targetCols:0 };
  document.getElementById('arrayQuizCard').style.display='none';
  document.getElementById('arraysNextBtn').style.display='none';
  renderArray();
  startArrayRound();
}
function changeRows(d){ arrayState.rows = Math.min(9, Math.max(1, arrayState.rows+d)); renderArray(); }
function changeCols(d){ arrayState.cols = Math.min(9, Math.max(1, arrayState.cols+d)); renderArray(); }
function renderArray(){
  document.getElementById('rowsVal').textContent = arrayState.rows;
  document.getElementById('colsVal').textContent = arrayState.cols;
  const grid = document.getElementById('seatGrid');
  grid.style.gridTemplateColumns = `repeat(${arrayState.cols}, 24px)`;
  grid.innerHTML='';
  for(let i=0;i<arrayState.rows*arrayState.cols;i++){
    const d = document.createElement('div');
    d.className='seat'; d.textContent='💺';
    grid.appendChild(d);
  }
  document.getElementById('arraySentence').textContent =
    `${arrayState.rows} rows × ${arrayState.cols} seats = ${arrayState.rows*arrayState.cols} seats`;
}
function startArrayRound(){
  arrayState.round++;
  if(arrayState.round > 3){
    document.getElementById('arrayQuizCard').style.display='none';
    document.getElementById('arraysNextBtn').style.display='inline-block';
    speak('levelup');
    return;
  }
  const r = rand(2,7), c = rand(2,7);
  arrayState.targetRows = r; arrayState.targetCols = c;
  document.getElementById('arrayQuizCard').style.display='block';
  document.getElementById('arrayQText').textContent =
    `Build an arena section with ${r} rows and ${c} seats in each row. Set the steppers to match, then check!`;
  document.getElementById('arrayFeedback').textContent='';
  document.getElementById('arrayFeedback').className='feedback';
}
function checkArrayAnswer(){
  const fb = document.getElementById('arrayFeedback');
  if(arrayState.rows === arrayState.targetRows && arrayState.cols === arrayState.targetCols){
    fb.textContent = `Perfect array! ${arrayState.rows} × ${arrayState.cols} = ${arrayState.rows*arrayState.cols} seats! 🎉`;
    fb.className='feedback good';
    addStar(1);
    speak('correct');
    setTimeout(startArrayRound, 1300);
  } else {
    fb.textContent = `Adjust the steppers: you need ${arrayState.targetRows} rows and ${arrayState.targetCols} seats per row.`;
    fb.className='feedback bad';
    speak('wrong');
  }
}

/* ============ ROUND 3: SKIP COUNTING ============ */
let skipState = { step:0, jumps:0, targetJumps:0, position:0, max:60 };
function initSkip(){
  const step = rand(2,6);
  const targetJumps = rand(3,6);
  skipState = { step, jumps:0, targetJumps, position:0, max: step*targetJumps + step };
  document.getElementById('skipPrompt').textContent =
    `Coach Rex's scoreboard jumps by ${step} for every round won. Jump ${targetJumps} times to find the tournament score!`;
  document.getElementById('skipFeedback').textContent='';
  document.getElementById('skipNextBtn').style.display='none';
  drawNumberLine();
  updateSkipSentence();
}
function drawNumberLine(){
  const svg = document.getElementById('numberLineSvg');
  const max = skipState.max;
  const width = Math.max(360, max*16);
  svg.setAttribute('width', width);
  svg.setAttribute('viewBox', `0 0 ${width} 90`);
  svg.innerHTML='';
  const scale = (width-30) / max;
  const y = 50;
  svg.innerHTML += `<line x1="15" y1="${y}" x2="${width-15}" y2="${y}" stroke="#ffd166" stroke-width="3"/>`;
  for(let i=0;i<=max;i+=1){
    const x = 15 + i*scale;
    const isMultiple = i % skipState.step === 0;
    svg.innerHTML += `<line x1="${x}" y1="${y-5}" x2="${x}" y2="${y+5}" stroke="${isMultiple?'#ffd166':'#ffffff55'}" stroke-width="${isMultiple?3:1}"/>`;
    if(isMultiple){
      svg.innerHTML += `<text x="${x}" y="${y+22}" font-size="11" fill="#ffd166" text-anchor="middle">${i}</text>`;
    }
  }
  const px = 15 + skipState.position*scale;
  svg.innerHTML += `<circle cx="${px}" cy="${y}" r="9" fill="#ff8c42" stroke="#fff" stroke-width="2"/>`;
  svg.innerHTML += `<text x="${px}" y="${y-14}" font-size="16" text-anchor="middle">🐾</text>`;
}
function updateSkipSentence(){
  document.getElementById('skipSentence').textContent =
    `Jumps: ${skipState.jumps} of ${skipState.targetJumps}  |  Score so far: ${skipState.position}`;
}
function doJump(){
  if(skipState.jumps >= skipState.targetJumps){
    checkSkipAnswer();
    return;
  }
  skipState.jumps++;
  skipState.position += skipState.step;
  drawNumberLine();
  updateSkipSentence();
  if(skipState.jumps === skipState.targetJumps) checkSkipAnswer();
}
function resetSkip(){
  skipState.jumps=0; skipState.position=0;
  drawNumberLine(); updateSkipSentence();
  document.getElementById('skipFeedback').textContent='';
}
function checkSkipAnswer(){
  const fb = document.getElementById('skipFeedback');
  fb.textContent = `${skipState.targetJumps} × ${skipState.step} = ${skipState.position}. Rex's team scored ${skipState.position} points! 🏆`;
  fb.className='feedback good';
  addStar(1);
  speak('correct');
  document.getElementById('skipNextBtn').style.display='inline-block';
}

function initBoss(){
  state.bossHp = 5; state.playerHp = 5;
  state.bossCorrect = 0; state.bossTotal = 0;
  updateHealthUI();
  speak('boss');
  nextBossQuestion();
}
function updateHealthUI(){
  document.getElementById('playerHp').style.width = (state.playerHp/5*100)+'%';
  document.getElementById('bossHp').style.width = (state.bossHp/5*100)+'%';
  document.getElementById('playerHpText').textContent = state.playerHp+'/5';
  document.getElementById('bossHpText').textContent = state.bossHp+'/5';
}
function genBossQuestion(){
  const type = rand(0,2);
  if(type===0){
    const a = rand(2,9), b = rand(2,9);
    return { text: `${a} × ${b} = ?`, answer: a*b, wrong:[a*b+a, a*b-b, a*b+ b===0?1:1] };
  } else if(type===1){
    const rows = rand(2,8), cols = rand(2,8);
    return { text: `An array has ${rows} rows and ${cols} columns. How many total?`, answer: rows*cols, wrong:[rows+cols, rows*cols+rows, rows*cols-cols] };
  } else {
    const g = rand(2,8), n = rand(2,8);
    return { text: `${n} teams of ${g} wrestlers each. Total wrestlers?`, answer: n*g, wrong:[n+g, n*g+n, n*g-g] };
  }
}
function nextBossQuestion(){
  if(state.bossHp<=0 || state.playerHp<=0) return;
  clearInterval(state.bossTimer);
  const q = genBossQuestion();
  state.currentBossQ = q;
  document.getElementById('bossQuestion').textContent = q.text;
  let choices = shuffle([q.answer, ...q.wrong.filter(w=>w!==q.answer).slice(0,3)]);
  while(choices.length<4) choices.push(q.answer + rand(1,4));
  choices = shuffle([...new Set(choices)]);
  while(choices.length<4) choices.push(q.answer + rand(5,9));
  const wrap = document.getElementById('bossChoices');
  wrap.innerHTML='';
  choices.slice(0,4).forEach(c=>{
    const b = document.createElement('button');
    b.className='choice-btn'; b.textContent=c;
    b.onclick = ()=>answerBoss(b, c, q.answer);
    wrap.appendChild(b);
  });
  document.getElementById('bossFeedback').textContent='';
  state.bossTimeLeft = 10;
  document.getElementById('timerFill').style.width='100%';
  state.bossTimer = setInterval(()=>{
    state.bossTimeLeft -= 0.1;
    const pct = Math.max(0, state.bossTimeLeft/10*100);
    document.getElementById('timerFill').style.width = pct+'%';
    if(state.bossTimeLeft<=0){
      clearInterval(state.bossTimer);
      answerBoss(null, null, q.answer);
    }
  }, 100);
}
function answerBoss(btn, chosen, correct){
  clearInterval(state.bossTimer);
  document.querySelectorAll('#bossChoices .choice-btn').forEach(b=>b.onclick=null);
  state.bossTotal++;
  const fb = document.getElementById('bossFeedback');
  if(chosen === correct){
    if(btn) btn.classList.add('correct');
    fb.textContent = 'Direct hit! 🥊'; fb.className='feedback good';
    state.bossHp--; state.bossCorrect++; addStar(2);
    speak('correct');
  } else {
    if(btn) btn.classList.add('wrong');
    document.querySelectorAll('#bossChoices .choice-btn').forEach(b=>{
      if(parseInt(b.textContent,10)===correct) b.classList.add('correct');
    });
    fb.textContent = chosen===null ? `Time's up! Answer was ${correct}.` : `Rex blocks it! Answer was ${correct}.`;
    fb.className='feedback bad';
    state.playerHp--;
    speak('wrong');
  }
  updateHealthUI();
  setTimeout(()=>{
    if(state.bossHp<=0){ endBoss(true); }
    else if(state.playerHp<=0){ endBoss(false); }
    else nextBossQuestion();
  }, 1100);
}
function endBoss(won){
  document.getElementById('victoryTitle').textContent = won ? '🏆 CHAMPION! 🏆' : '💪 Great Fight!';
  document.getElementById('victoryText').textContent = won
    ? "You out-trained Coach Rex and won the Golden Belt! Your equal groups, arrays, and skip-counting skills are championship level!"
    : "Coach Rex held on this time — but you fought hard! Train again and you'll take that belt for sure.";
  document.getElementById('statsText').textContent =
    `⭐ Stars earned: ${state.stars}  |  Final match accuracy: ${state.bossCorrect}/${state.bossTotal}  |  Word problems: ${state.wordCorrect}/${state.wordTotal}`;
  speak(won ? 'win' : 'lose');
  goTo('screen-victory');
  if(won) launchConfetti();
}

/* ============ CONFETTI ============ */
function launchConfetti(){
  const canvas = document.getElementById('confetti-canvas');
  canvas.width = window.innerWidth; canvas.height = window.innerHeight;
  const ctx = canvas.getContext('2d');
  const colors = ['#ffd166','#ff8c42','#e6531f','#3ba55c','#3a6ea5'];
  const pieces = Array.from({length:120}, ()=>({
    x: Math.random()*canvas.width, y: -20 - Math.random()*canvas.height,
    r: 4+Math.random()*6, c: colors[rand(0,colors.length-1)],
    vy: 2+Math.random()*3, vx: -2+Math.random()*4, rot: Math.random()*360, vr: -6+Math.random()*12
  }));
  let frame = 0;
  function anim(){
    frame++;
    ctx.clearRect(0,0,canvas.width,canvas.height);
    pieces.forEach(p=>{
      p.x += p.vx; p.y += p.vy; p.rot += p.vr;
      ctx.save();
      ctx.translate(p.x,p.y); ctx.rotate(p.rot*Math.PI/180);
      ctx.fillStyle = p.c;
      ctx.fillRect(-p.r/2,-p.r/2,p.r,p.r*1.6);
      ctx.restore();
    });
    if(frame < 220) requestAnimationFrame(anim);
    else ctx.clearRect(0,0,canvas.width,canvas.height);
  }
  anim();
}
