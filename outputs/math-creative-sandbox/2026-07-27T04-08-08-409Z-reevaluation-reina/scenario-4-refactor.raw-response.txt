/* ---------------- Generic drag-and-drop token system ---------------- */
window.onDropUpdate = null;
function makeDraggable(el){
  el.style.touchAction = 'none';
  el.addEventListener('pointerdown', (e)=>{
    e.preventDefault();
    const startParent = el.parentElement;
    const rect = el.getBoundingClientRect();
    const offsetX = e.clientX - rect.left;
    const offsetY = e.clientY - rect.top;
    try{ el.setPointerCapture(e.pointerId); }catch(err){}
    el.classList.add('dragging-token');
    document.body.appendChild(el);
    el.style.position = 'fixed';
    el.style.left = (e.clientX - offsetX) + 'px';
    el.style.top = (e.clientY - offsetY) + 'px';
    el.style.zIndex = 1000;

    function move(ev){
      el.style.left = (ev.clientX - offsetX) + 'px';
      el.style.top = (ev.clientY - offsetY) + 'px';
    }
    function up(ev){
      el.removeEventListener('pointermove', move);
      el.removeEventListener('pointerup', up);
      el.classList.remove('dragging-token');
      el.style.position = '';
      el.style.left = ''; el.style.top = ''; el.style.zIndex = '';
      el.style.display = 'none';
      const target = document.elementFromPoint(ev.clientX, ev.clientY);
      el.style.display = '';
      const zone = target ? target.closest('.drop-zone') : null;
      if(zone){ zone.appendChild(el); } else { startParent.appendChild(el); }
      if(typeof window.onDropUpdate === 'function') window.onDropUpdate();
    }
    el.addEventListener('pointermove', move);
    el.addEventListener('pointerup', up);
  });
}

/* ---------------- Resizable array widget ---------------- */
function createResizableArray(container, initRows, initCols, emoji, opts){
  opts = opts || {};
  const minR = opts.minR||1, maxR=opts.maxR||6, minC=opts.minC||1, maxC=opts.maxC||6;
  let rows = initRows, cols = initCols;
  container.innerHTML = '';
  const outer = document.createElement('div'); outer.className='resize-outer';
  const gridAndCol = document.createElement('div'); gridAndCol.className='grid-and-col';
  const grid = document.createElement('div'); grid.className='array-grid';
  const colHandle = document.createElement('div'); colHandle.className='col-handle'; colHandle.textContent='↔ cols';
  const rowHandle = document.createElement('div'); rowHandle.className='row-handle'; rowHandle.textContent='↕ drag rows';
  gridAndCol.append(grid, colHandle);
  outer.append(gridAndCol, rowHandle);
  container.appendChild(outer);

  function render(){
    grid.innerHTML = '';
    grid.style.gridTemplateColumns = `repeat(${cols}, 32px)`;
    for(let r=0;r<rows;r++){
      for(let c=0;c<cols;c++){
        const cell = document.createElement('div');
        cell.className = 'array-cell';
        cell.textContent = emoji;
        cell.dataset.r = r; cell.dataset.c = c;
        grid.appendChild(cell);
      }
    }
    if(opts.onChange) opts.onChange(rows, cols);
  }
  render();

  function attachHandleDrag(handle, isRow){
    handle.style.touchAction = 'none';
    handle.addEventListener('pointerdown', (e)=>{
      e.preventDefault();
      try{ handle.setPointerCapture(e.pointerId); }catch(err){}
      const startX = e.clientX, startY = e.clientY;
      const startRows = rows, startCols = cols;
      function move(ev){
        if(isRow){
          const delta = ev.clientY - startY;
          const newRows = clamp(startRows + Math.round(delta/36), minR, maxR);
          if(newRows !== rows){ rows = newRows; render(); }
        } else {
          const delta = ev.clientX - startX;
          const newCols = clamp(startCols + Math.round(delta/36), minC, maxC);
          if(newCols !== cols){ cols = newCols; render(); }
        }
      }
      function up(){
        handle.removeEventListener('pointermove', move);
        handle.removeEventListener('pointerup', up);
      }
      handle.addEventListener('pointermove', move);
      handle.addEventListener('pointerup', up);
    });
  }
  attachHandleDrag(rowHandle, true);
  attachHandleDrag(colHandle, false);

  return {
    getRows: ()=>rows, getCols: ()=>cols,
    setRows: (v)=>{ rows = clamp(v,minR,maxR); render(); },
    setCols: (v)=>{ cols = clamp(v,minC,maxC); render(); },
    gridEl: grid
  };
}

/* ---------------- steppers / static visuals (still used by build1 & word problems) ---------------- */
function makeStepper(label, min, max, value, onChange){
  const wrap = document.createElement('div'); wrap.className = 'stepper';
  const lbl = document.createElement('div'); lbl.className = 'stepper-label'; lbl.textContent = label;
  const row = document.createElement('div'); row.className = 'stepper-row';
  const minus = document.createElement('button'); minus.className = 'step-btn'; minus.textContent = '−';
  const val = document.createElement('div'); val.className = 'stepper-value'; val.textContent = value;
  const plus = document.createElement('button'); plus.className = 'step-btn'; plus.textContent = '+';
  minus.onclick = () => { if(value > min){ value--; val.textContent = value; onChange(value); } };
  plus.onclick = () => { if(value < max){ value++; val.textContent = value; onChange(value); } };
  row.append(minus, val, plus); wrap.append(lbl, row);
  return { el: wrap, get: () => value, set: (v) => { value = v; val.textContent = v; } };
}
function renderGroupsVisual(container, groups, size, emoji){

    cum += size;
    const cLabel = document.createElement('div'); cLabel.className = 'cum-label'; cLabel.textContent = cum;
    box.appendChild(cLabel);
    container.appendChild(box);
  }
}

}

  }

  document.querySelectorAll('#t1-pots .pot-drop').forEach(p=>{

function checkTeach1(){
  const shelf = document.getElementById('t1-shelf');
  const pots = [...document.querySelectorAll('#t1-pots .pot-drop')];
  const counts = pots.map(p=>p.querySelector('.pot-flowers').children.length);
  const fb = document.getElementById('t1-feedback');
  if(shelf.children.length > 0){
    fb.textContent = `You still have ${shelf.children.length} flower(s) on the shelf. Drag them all into pots!`;
    fb.className = 'feedback bad';
    return;
  }
  const allEqual = counts.every(c=>c===counts[0]) && counts[0] > 0;
  if(allEqual){
    fb.textContent = `🎉 All pots equal — that's ${pots.length} groups of ${counts[0]}!`;
    fb.className = 'feedback good';
    document.getElementById('t1-eq').textContent =
      `${pots.length} × ${counts[0]} = ${pots.length*counts[0]}  (skip count: ${skipList(pots.length,counts[0]).join(', ')})`;
    document.getElementById('t1-next').disabled = false;
    carryFromTeach1 = { n1: pots.length, n2: counts[0] };
  } else {
    fb.textContent = `Not equal yet — pots have ${counts.join(', ')} flowers. Drag flowers between pots to make them match!`;
    fb.className = 'feedback bad';
  }
}

/* ================= SPOT THE EQUAL GROUPS ================= */
function setupSpot(){
  const correctIdx = randInt(0,2);
  const equalSize = randInt(2,5);
  const correctCounts = [equalSize, equalSize, equalSize];
  let wrongA = [equalSize+1, equalSize, Math.max(1,equalSize-1)];
  let wrongB = [Math.max(1,equalSize-1), equalSize+2, equalSize];
  const arrangement = [null,null,null];
  arrangement[correctIdx] = correctCounts;
  const others = [wrongA, wrongB];
  let oi = 0;
  for(let i=0;i<3;i++) if(i !== correctIdx) arrangement[i] = others[oi++];

  const container = document.getElementById('spot-options');
  container.innerHTML = '';
  arrangement.forEach((counts, idx)=>{
    const card = document.createElement('div');
    card.className = 'spot-card';
    const potsWrap = document.createElement('div');
    potsWrap.className = 'spot-pots';
    counts.forEach(c=>{
      const pot = document.createElement('div');
      pot.className = 'mini-pot';
      let flowers = '';
      for(let k=0;k<c;k++) flowers += '🌼';
      pot.innerHTML = `<div class="mini-pot-icon">🪴</div><div class="mini-flowers">${flowers}</div>`;
      potsWrap.appendChild(pot);
    });
    card.appendChild(potsWrap);
    card.onclick = ()=> selectSpot(idx === correctIdx, card);
    container.appendChild(card);
  });
  document.getElementById('spot-feedback').textContent = '';
  document.getElementById('spot-feedback').className = 'feedback';
  document.getElementById('spot-next').disabled = true;
}
function selectSpot(isCorrect, card){
  document.querySelectorAll('.spot-card').forEach(c=>c.classList.remove('correct','wrong'));
  const fb = document.getElementById('spot-feedback');
  if(isCorrect){
    card.classList.add('correct');
    fb.textContent = "🎉 Yes! Every pot has the exact same number — that's equal groups, ready for multiplication!";
    fb.className = 'feedback good';
    document.getElementById('spot-next').disabled = false;
  } else {
    card.classList.add('wrong');
    fb.textContent = 'Look closely — count each pot. Are they really all the same? Try another one.';
    fb.className = 'feedback bad';
  }
}
