<script>
(function(){
  "use strict";

  // ---------- Audio (tiny beeps, no assets) ----------
  let actx;
  function beep(freq, dur, type){
    try{
      actx = actx || new (window.AudioContext||window.webkitAudioContext)();
      const o = actx.createOscillator();
      const g = actx.createGain();
      o.type = type||'sine';
      o.frequency.value = freq;
      g.gain.value = 0.08;
      o.connect(g); g.connect(actx.destination);
      o.start();
      g.gain.exponentialRampToValueAtTime(0.001, actx.currentTime + dur);
      o.stop(actx.currentTime + dur);
    }catch(e){}
  }
  function playCorrect(){ beep(660,0.15); setTimeout(()=>beep(880,0.2),120); }
  function playWrong(){ beep(200,0.25,'sawtooth'); }
  function playClick(){ beep(440,0.06); }

  // ---------- Companion ----------
  const speechBubble = document.getElementById('speechBubble');
  const mouthPath = document.getElementById('mouthPath');
  const MOUTHS = {
    happy: "M35 60 Q50 78 65 60",
    excited: "M32 58 Q50 82 68 58",
    neutral: "M35 64 Q50 68 65 64",
    sad: "M35 68 Q50 56 65 68",
    idle: "M38 64 Q50 66 62 64"
  };
  function setMood(mood){ mouthPath.setAttribute('d', MOUTHS[mood]||MOUTHS.neutral); }
  function say(text, mood){
    speechBubble.textContent = text;
    setMood(mood||'neutral');
  }

  // idle detection
  let idleTimer;
  const IDLE_MS = 45000;
  function resetIdle(){
    clearTimeout(idleTimer);
    idleTimer = setTimeout(()=>{
      say("Still with me, champ? Take your time — tap when ready! 🤔", "idle");
    }, IDLE_MS);
  }
  document.addEventListener('pointerdown', resetIdle);
  resetIdle();

  // ---------- Confetti ----------
  const canvas = document.getElementById('confettiCanvas');
  const ctx = canvas.getContext('2d');
  function resizeCanvas(){canvas.width = innerWidth; canvas.height = innerHeight;}
  resizeCanvas();
  window.addEventListener('resize', resizeCanvas);
  let confettiParticles = [];
  function launchConfetti(){
    confettiParticles = [];
    const colors = ['#2563eb','#f59e0b','#dc2626','#16a34a','#a855f7'];
    for(let i=0;i<130;i++){
      confettiParticles.push({
        x: Math.random()*canvas.width,
        y: -20 - Math.random()*canvas.height*0.5,
        vy: 2+Math.random()*3,
        vx: -1+Math.random()*2,
        size: 4+Math.random()*6,
        color: colors[i%colors.length],
        rot: Math.random()*360
      });
    }
    requestAnimationFrame(animateConfetti);
  }
  let confettiRunning=false;
  function animateConfetti(){
    confettiRunning = true;
    ctx.clearRect(0,0,canvas.width,canvas.height);
    let alive=false;
    confettiParticles.forEach(p=>{
      p.y += p.vy; p.x += p.vx; p.rot += 6;
      if(p.y < canvas.height+20) alive=true;
      ctx.save();
      ctx.translate(p.x,p.y);
      ctx.rotate(p.rot*Math.PI/180);
      ctx.fillStyle = p.color;
      ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size);
      ctx.restore();
    });
    if(alive) requestAnimationFrame(animateConfetti);
    else { ctx.clearRect(0,0,canvas.width,canvas.height); confettiRunning=false; }
  }

  // ---------- Game data ----------
  const TEMPLATES = [
    {emoji:'🤼', group:'Team', text:(r,c)=>`Coach Kuma lines up ${r} tag-teams with ${c} wrestlers on each team. How many wrestlers are ready to battle?`},
    {emoji:'♟️', group:'Row', text:(r,c)=>`The dojo sets out ${r} rows of chess pieces with ${c} pieces in each row. How many pieces are on the board?`},
    {emoji:'🥇', group:'Box', text:(r,c)=>`The trophy shelf has ${r} boxes with ${c} medals in each box. How many medals are there?`},
    {emoji:'🎟️', group:'Section', text:(r,c)=>`The arena has ${r} sections with ${c} fans cheering in each section. How many fans came to watch?`},
    {emoji:'🍫', group:'Tray', text:(r,c)=>`${r} snack trays each hold ${c} energy bars for the wrestlers. How many energy bars in all?`},
    {emoji:'🥋', group:'Group', text:(r,c)=>`${r} practice groups train together, ${c} wrestlers in each group. How many wrestlers are training?`}
  ];
  const TIERS = [
    {minR:2,maxR:3,minC:2,maxC:4},
    {minR:3,maxR:4,minC:3,maxC:5},
    {minR:3,maxR:5,minC:3,maxC:5},
    {minR:4,maxR:5,minC:4,maxC:6},
    {minR:4,maxR:6,minC:3,maxC:6}
  ];
  const TOTAL_MATCHES = TIERS.length;

  function randInt(a,b){ return a + Math.floor(Math.random()*(b-a+1)); }
  function pick(arr){ return arr[randInt(0,arr.length-1)]; }

  function createRound(tier){
    const rows = randInt(tier.minR, tier.maxR);
    const cols = randInt(tier.minC, tier.maxC);
    const tpl = pick(TEMPLATES);
    const answer = rows*cols;
    const distractSet = new Set([answer]);
    const distractors = [];
    const tries = [rows*(cols+1), rows*(cols-1), (rows+1)*cols, (rows-1)*cols, answer+cols, answer-cols, answer+rows];
    tries.forEach(v=>{
      if(v>0 && !distractSet.has(v) && distractors.length<3){ distractSet.add(v); distractors.push(v); }
    });
    while(distractors.length<3){
      const v = answer + randInt(-6,6);
      if(v>0 && !distractSet.has(v)){ distractSet.add(v); distractors.push(v); }
    }
    const choices = [answer, ...distractors.slice(0,3)];
    // shuffle
    for(let i=choices.length-1;i>0;i--){ const j=randInt(0,i); [choices[i],choices[j]]=[choices[j],choices[i]]; }
    return {rows, cols, tpl, answer, choices, text: tpl.text(rows,cols)};
  }

  function buildMatches(){
    return TIERS.map(createRound);
  }

  // ---------- Game state ----------
  let matches = [];
  let currentIndex = 0;
  let filledRows = 0;
  let score = 0;
  let streak = 0;
  let bestStreak = 0;
  let answered = false;
  let bonusActive = false;
  let bonusIndex = 0;
  let bonusQuestions = [];
  let bonusTimerHandle = null;

  const scenarioText = document.getElementById('scenarioText');
  const arrayContainer = document.getElementById('arrayContainer');
  const skipStrip = document.getElementById('skipStrip');
  const sentenceEl = document.getElementById('sentence');
  const questionText = document.getElementById('questionText');
  const choicesEl = document.getElementById('choices');
  const nextBtn = document.getElementById('nextBtn');
  const roundLabel = document.getElementById('roundLabel');
  const streakBadge = document.getElementById('streakBadge');
  const scoreLabel = document.getElementById('scoreLabel');
  const progressBar = document.getElementById('progressBar');
  const timerBarWrap = document.getElementById('timerBarWrap');
  const timerFill = document.getElementById('timerFill');

  const startScreen = document.getElementById('startScreen');
  const gameScreen = document.getElementById('gameScreen');
  const finaleScreen = document.getElementById('finaleScreen');

  function updateHud(){
    scoreLabel.textContent = '⭐ ' + score;
    streakBadge.textContent = '🔥 Streak: ' + streak;
    const pct = bonusActive
      ? 100
      : Math.round((currentIndex/(TOTAL_MATCHES+1))*100);
    progressBar.style.width = pct + '%';
  }

  function renderRound(){
    const m = matches[currentIndex];
    filledRows = 0;
    answered = false;
    roundLabel.textContent = `Match ${currentIndex+1} of ${TOTAL_MATCHES}`;
    scenarioText.textContent = m.text;
    skipStrip.innerHTML = '';
    sentenceEl.textContent = `Tap Group 1 to begin building the array!`;
    questionText.textContent = '';
    choicesEl.innerHTML = '';
    nextBtn.classList.add('hidden');
    arrayContainer.innerHTML = '';

    for(let r=0;r<m.rows;r++){
      const rowDiv = document.createElement('div');
      rowDiv.className = 'arrayRow' + (r===0 ? ' next' : '');
      rowDiv.dataset.row = r;

      const label = document.createElement('div');
      label.className = 'rowLabel';
      label.textContent = `${m.tpl.group} ${r+1}`;
      rowDiv.appendChild(label);

      for(let c=0;c<m.cols;c++){
        const cell = document.createElement('div');
        cell.className = 'cell';
        cell.dataset.row = r;
        cell.textContent = '';
        cell.addEventListener('click', onCellClick);
        rowDiv.appendChild(cell);
      }
      arrayContainer.appendChild(rowDiv);
    }
    updateHud();
  }

  function onCellClick(e){
    const m = matches[currentIndex];
    const row = parseInt(e.currentTarget.dataset.row,10);
    if(row !== filledRows) return; // must fill in order
    playClick();
    const cells = arrayContainer.querySelectorAll(`.cell[data-row="${row}"]`);
    cells.forEach(c=>{ c.classList.add('filled'); c.textContent = m.tpl.emoji; });
    filledRows++;

    // update next-row highlight
    arrayContainer.querySelectorAll('.arrayRow').forEach(rw=>{
      rw.classList.toggle('next', parseInt(rw.dataset.row,10)===filledRows);
    });

    const running = filledRows * m.cols;
    const num = document.createElement('div');
    num.className = 'skipNum';
    num.textContent = running;
    skipStrip.appendChild(num);

    sentenceEl.textContent = `${filledRows} ${m.tpl.group.toLowerCase()}${filledRows>1?'s':''} of ${m.cols} = ${running}`;

    if(filledRows === m.rows){
      questionText.textContent = `So altogether, how many ${m.tpl.text(m.rows,m.cols).match(/how many (.*)\?/i) ? m.tpl.text(m.rows,m.cols).match(/how many (.*)\?/i)[1] : 'in all'}?`;
      showChoices(m);
    }
  }

  function showChoices(m){
    choicesEl.innerHTML = '';
    m.choices.forEach(val=>{
      const btn = document.createElement('button');
      btn.className = 'choiceBtn';
      btn.textContent = val;
      btn.addEventListener('click', ()=>onChoice(btn, val, m));
      choicesEl.appendChild(btn);
    });
  }

  function onChoice(btn, val, m){
    if(answered) return;
    resetIdle();
    if(val === m.answer){
      answered = true;
      btn.classList.add('correct');
      score += 10 + streak*2;
      streak++;
      bestStreak = Math.max(bestStreak, streak);
      playCorrect();
      const specialLine = Math.random() < 0.3;
      const lines = ["Ippon! You pinned that problem!", "Strong counting, champion!", "That's the power of arrays!", "You out-strategized that one!"];
      say(specialLine ? "🏆 Incredible! Championship-level thinking!" : pick(lines), 'excited');
      document.querySelectorAll('.choiceBtn').forEach(b=>b.disabled=true);
      nextBtn.classList.remove('hidden');
    } else {
      streak = 0;
      btn.classList.add('wrong');
      btn.disabled = true;
      playWrong();
      say("Not quite — check the skip-count trail above and try again!", 'sad');
    }
    updateHud();
  }

  nextBtn.addEventListener('click', ()=>{
    resetIdle();
    currentIndex++;
    if(currentIndex < TOTAL_MATCHES){
      renderRound();
      say("New challenger approaches! Build your array!", 'happy');
    } else {
      startBonusRound();
    }
  });

  // ---------- Bonus Round ----------
  function startBonusRound(){
    bonusActive = true;
    bonusIndex = 0;
    bonusQuestions = [createRound({minR:2,maxR:6,minC:2,maxC:6}),
                      createRound({minR:2,maxR:6,minC:2,maxC:6}),
                      createRound({minR:2,maxR:6,minC:2,maxC:6})];
    roundLabel.textContent = "⚡ Strategy Bonus Round!";
    say("Bonus Round! Quick strategy thinking — beat the clock for extra points!", 'excited');
    renderBonusQuestion();
  }

  function renderBonusQuestion(){
    if(bonusIndex >= bonusQuestions.length){
      finishGame();
      return;
    }
    const m = bonusQuestions[bonusIndex];
    filledRows = m.rows; // pre-filled for speed round; show full array immediately
    answered = false;
    scenarioText.textContent = "⏱ " + m.text;
    skipStrip.innerHTML = '';
    for(let i=1;i<=m.rows;i++){
      const n = document.createElement('div');
      n.className='skipNum';
      n.textContent = i*m.cols;
      skipStrip.appendChild(n);
    }
    sentenceEl.textContent = `${m.rows} groups of ${m.cols}`;
    questionText.textContent = "What's the total?";
    arrayContainer.innerHTML = '';
    for(let r=0;r<m.rows;r++){
      const rowDiv = document.createElement('div');
      rowDiv.className = 'arrayRow';
      const label = document.createElement('div');
      label.className='rowLabel';
      label.textContent = `${m.tpl.group} ${r+1}`;
      rowDiv.appendChild(label);
      for(let c=0;c<m.cols;c++){
        const cell = document.createElement('div');
        cell.className='cell filled';
        cell.textContent = m.tpl.emoji;
        rowDiv.appendChild(cell);
      }
      arrayContainer.appendChild(rowDiv);
    }
    showChoices(m);
    nextBtn.classList.add('hidden');

    timerBarWrap.classList.remove('hidden');
    let timeLeft = 100;
    timerFill.style.width = '100%';
    clearInterval(bonusTimerHandle);
    bonusTimerHandle = setInterval(()=>{
      timeLeft -= 2;
      timerFill.style.width = Math.max(timeLeft,0) + '%';
      if(timeLeft<=0){
        clearInterval(bonusTimerHandle);
        if(!answered){
          say("Time's up! Keep that chin up, champion!", 'sad');
          document.querySelectorAll('.choiceBtn').forEach(b=>b.disabled=true);
          setTimeout(()=>{ bonusIndex++; renderBonusQuestion(); }, 1400);
        }
      }
    }, 200);

    // override choice handler for bonus scoring/timing
    document.querySelectorAll('.choiceBtn').forEach(b=>{
      b.onclick = ()=>{
        if(answered) return;
        answered = true;
        clearInterval(bonusTimerHandle);
        resetIdle();
        if(parseInt(b.textContent,10) === m.answer){
          b.classList.add('correct');
          const bonusPts = 15 + Math.round(timeLeft/5);
          score += bonusPts;
          streak++; bestStreak=Math.max(bestStreak,streak);
          playCorrect();
          say(`Bonus points! +${bonusPts} ⭐`, 'excited');
        } else {
          b.classList.add('wrong');
          streak = 0;
          playWrong();
          say("Close! Speed comes with practice.", 'sad');
        }
        updateHud();
        document.querySelectorAll('.choiceBtn').forEach(bb=>bb.disabled=true);
        setTimeout(()=>{ bonusIndex++; renderBonusQuestion(); }, 1200);
      };
    });
  }

  // ---------- Finale ----------
  function finishGame(){
    timerBarWrap.classList.add('hidden');
    gameScreen.classList.add('hidden');
    finaleScreen.classList.remove('hidden');
